   63 impl<'mir, 'tcx: 'mir> RustBorrowCheckerData<'mir, 'tcx> {
   64     pub(crate) fn borrows_in_scope_at(&self, location: Location, before: bool) -> InScopeBorrows {
   65         if before {
   66             self.in_scope_borrows
   67                 .borrow_mut()
   68                 .seek_before_primary_effect(location);
   69         } else {
   70             self.in_scope_borrows
   71                 .borrow_mut()
   72                 .seek_after_primary_effect(location);
   73         }
   74         self.in_scope_borrows.borrow().get().clone()
   75     }
   76     pub(crate) fn new<T: BodyAndBorrows<'tcx>>(tcx: ty::TyCtxt<'tcx>, body: &'mir T) -> Self {
   77         let location_table = body.location_table();
   78         let region_cx = body.region_inference_context();
   79         let borrows = body.borrow_set();
   80         Self {
   81             in_scope_borrows: Rc::new(RefCell::new(
   82                 compute_fixpoint(
   83                     Borrows::new(tcx, body.body(), region_cx, borrows),
   84                     tcx,
   85                     body.body(),
   86                 )
   87                 .into_results_cursor(body.body()),
   88             )),
   89             input_facts: body.input_facts(),
   90             location_table,
   91             body: body.body(),
   92             tcx,
   93             region_cx,
   94             borrows,
   95             #[cfg(feature = "visualization")]
   96             pretty_printer: RegionPrettyPrinter::new(region_cx),
   97         }
   98     }
   99 }

  109 impl<'mir, 'tcx: 'mir> PoloniusBorrowChecker<'mir, 'tcx> {
  110     fn ctxt(&self) -> CompilerCtxt<'mir, 'tcx, &Self> {
  111         CompilerCtxt::new(
  112             self.borrow_checker_data.body,
  113             self.borrow_checker_data.tcx,
  114             self,
  115         )
  116     }
  117 
  118     pub fn region_of_borrow(&self, index: BorrowIndex) -> RegionVid {
  119         self.borrow_checker_data.borrows[index].region()
  120     }
  121 
  122     pub fn new<T: BodyAndBorrows<'tcx>>(tcx: ty::TyCtxt<'tcx>, body: &'mir T) -> Self {
  123         let borrow_checker_data = RustBorrowCheckerData::new(tcx, body);
  124         let output_facts = Output::compute(
  125             body.input_facts(),
  126             polonius_engine::Algorithm::DatafrogOpt,
  127             true,
  128         );
  129         let place_liveness = PlaceLiveness::new(CompilerCtxt::new(body.body(), tcx, ()));
  130         Self {
  131             output_facts,
  132             place_liveness,
  133             borrow_checker_data,
  134         }
  135     }
  136 
  137     pub fn origin_live_on_entry(&self, location: RichLocation) -> Option<BTreeSet<RegionVid>> {
  138         let origins = match location {
  139             RichLocation::Start(location) => self.output_facts.origin_live_on_entry.get(
  140                 &self
  141                     .borrow_checker_data
  142                     .location_table
  143                     .start_index(location),
  144             )?,
  145             RichLocation::Mid(location) => self
  146                 .output_facts
  147                 .origin_live_on_entry
  148                 .get(&self.borrow_checker_data.location_table.mid_index(location))?,
  149         };
  150         Some(origins.iter().map(|r| (*r).into()).collect())
  151     }
  152 
  153     pub fn loans_live_at(&self, location: RichLocation) -> BTreeSet<RegionVid> {
  154         let loans = match location {
  155             RichLocation::Start(location) => self.output_facts.loans_in_scope_at(
  156                 self.borrow_checker_data
  157                     .location_table
  158                     .start_index(location),
  159             ),
  160             RichLocation::Mid(location) => self
  161                 .output_facts
  162                 .loans_in_scope_at(self.borrow_checker_data.location_table.mid_index(location)),
  163         };
  164         loans
  165             .iter()
  166             .map(|r| self.borrow_index_to_region(*r))
  167             .collect()
  168     }
  169 
  170     pub fn origin_contains_loan_at_map(
  171         &self,
  172         location: RichLocation,
  173     ) -> Option<BTreeMap<RegionVid, BTreeSet<BorrowIndex>>> {
  174         let location_table = self.location_table();
  175         let loans = match location {
  176             RichLocation::Start(location) => self
  177                 .output_facts
  178                 .origin_contains_loan_at
  179                 .get(&location_table.start_index(location))?,
  180             RichLocation::Mid(location) => self
  181                 .output_facts
  182                 .origin_contains_loan_at
  183                 .get(&location_table.mid_index(location))?,
  184         };
  185         Some(
  186             loans
  187                 .iter()
  188                 .map(|(region, indices)| ((*region).into(), indices.clone()))
  189                 .collect(),
  190         )
  191     }
  192 }

  200 impl<'mir, 'tcx: 'mir> RustBorrowCheckerInterface<'tcx> for PoloniusBorrowChecker<'mir, 'tcx> {
  201     fn tcx(&self) -> ty::TyCtxt<'tcx> {
  202         self.borrow_checker_data.tcx
  203     }
  204     fn is_live(&self, node: PcgNode<'tcx>, location: Location) -> bool {
  205         let ctxt = self.ctxt();
  206         let regions: Vec<GeneralizedLifetime<'tcx>> = match node {
  207             PcgNode::Place(place) => ctxt.extract_regions(place).into_iter().collect(),
  208             PcgNode::LifetimeProjection(lifetime_projection) => {
  209                 vec![lifetime_projection.region(ctxt)]
  210             }
  211         };
  212 
  213         let start_index = self
  214             .borrow_checker_data
  215             .location_table
  216             .start_index(location);
  217 
  218         let live_loans = self.output_facts.loans_in_scope_at(start_index);
  219 
  220         let live_origins: BTreeSet<RegionVid> = self
  221             .output_facts
  222             .origins_live_at(start_index)
  223             .iter()
  224             .map(|r| (*r).into())
  225             .collect::<BTreeSet<_>>();
  226         regions.iter().any(|region| match region {
  227             GeneralizedLifetime::Region(PcgRegion::RegionVid(region_vid)) => {
  228                 live_origins.contains(region_vid)
  229                     || live_loans.iter().any(|loan| {
  230                         self.borrow_checker_data
  231                             .region_cx
  232                             .eval_outlives(*region_vid, self.borrow_index_to_region(*loan))
  233                     })
  234             }
  235             // `RegionsIn` has no region of its own, so it is live while its place is.
  236             GeneralizedLifetime::RegionsIn(_) => node
  237                 .related_maybe_labelled_place()
  238                 .is_none_or(|p| self.place_liveness.is_live(p.place(), location)),
  239             _ => todo!(),
  240         })
  241     }
  242 
  243     fn region_infer_ctxt(&self) -> &RegionInferenceContext<'tcx> {
  244         self.borrow_checker_data.region_cx
  245     }
  246 
  247     fn location_table(&self) -> &LocationTable {
  248         self.borrow_checker_data.location_table
  249     }
  250 
  251     fn polonius_output(&self) -> Option<&PoloniusOutput> {
  252         Some(&self.output_facts)
  253     }
  254 
  255     fn borrow_set(&self) -> &BorrowSet<'tcx> {
  256         self.borrow_checker_data.borrows
  257     }
  258 
  259     fn input_facts(&self) -> &PoloniusInput {
  260         self.borrow_checker_data.input_facts
  261     }
  262 
  263     fn origin_contains_loan_at(
  264         &self,
  265         region: PcgRegion<'tcx>,
  266         loan: BorrowIndex,
  267         location: Location,
  268     ) -> bool {
  269         self.origin_contains_loan_at_map(RichLocation::Start(location))
  270             .is_some_and(|map| {
  271                 map.get(&region.vid().unwrap())
  272                     .is_some_and(|set| set.contains(&loan))
  273             })
  274     }
  275 
  276     fn borrows_in_scope_at(&self, location: Location, before: bool) -> InScopeBorrows {
  277         self.borrow_checker_data
  278             .borrows_in_scope_at(location, before)
  279     }
  280 }

  317     fn borrow_in_scope_at(&self, borrow_index: BorrowIndex, location: Location) -> bool {
  318         self.borrow_checker_data
  319             .in_scope_borrows
  320             .borrow_mut()
  321             .seek_before_primary_effect(location);
  322         let result = (*self.borrow_checker_data.in_scope_borrows)
  323             .borrow()
  324             .get()
  325             .contains(borrow_index);
  326         tracing::debug!(
  327             "borrow_in_scope_at({:?}, {:?}) = {}",
  328             borrow_index,
  329             location,
  330             result
  331         );
  332         result
  333     }

  335     fn is_live(&self, node: PcgNode<'tcx>, location: Location) -> bool {
  336         let Some(local) = node.related_maybe_labelled_place().map(|p| p.local()) else {
  337             return true;
  338         };
  339         if local == RETURN_PLACE {
  340             return true;
  341         }
  342         self.place_liveness.is_live(local.into(), location)
  343     }
