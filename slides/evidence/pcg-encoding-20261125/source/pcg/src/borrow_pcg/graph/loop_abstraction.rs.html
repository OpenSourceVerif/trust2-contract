  104     fn lifetime_projections<'a>(
  105         self,
  106         ctxt: impl HasCompilerCtxt<'a, 'tcx>,
  107     ) -> Vec<LifetimeProjection<'tcx>>
  108     where
  109         'tcx: 'a,
  110     {
  111         match self {
  112             MaybeRemoteCurrentPlace::Local(place) => place
  113                 .lifetime_projections(ctxt)
  114                 .into_iter()
  115                 .map(|rp| rp.to_pcg_node(ctxt).expect_lifetime_projection())
  116                 .collect(),
  117             MaybeRemoteCurrentPlace::Remote(place) => place
  118                 .generalized_lifetime_projections(ctxt)
  119                 .into_iter()
  120                 .map(|rp| rp.to_pcg_node(ctxt).expect_lifetime_projection())
  121                 .collect(),
  122         }
  123     }

  125 impl<'tcx> BorrowsGraph<'tcx> {
  126     pub(crate) fn get_loop_abstraction_graph<'mir>(
  127         &self,
  128         loop_blocked_places: &PlaceUsages<'tcx>,
  129         root_places: &HashSet<MaybeRemoteCurrentPlace<'tcx>>,
  130         candidate_blockers: &PlaceUsages<'tcx>,
  131         loop_head: mir::BasicBlock,
  132         validity_conditions: &ValidityConditions,
  133         ctxt: AnalysisCtxt<'mir, 'tcx>,
  134     ) -> ConstructAbstractionGraphResult<'tcx> {
  135         let mut graph = BorrowsGraph::default();
  136         let mut capability_updates = HashMap::default();
  137         let mut to_label = HashSet::default();
  138         let mut blocked_root_places: HashSet<Place<'tcx>> = HashSet::default();
  139 
  140         let loop_head_location = mir::Location {
  141             block: loop_head,
  142             statement_index: 0,
  143         };
  144 
  145         let mut expander = AbsExpander {
  146             loop_head_block: loop_head,
  147             graph: &mut graph,
  148             validity_conditions,
  149             ctxt,
  150         };
  151 
  152         tracing::debug!(
  153             "loop blocked places: {}",
  154             loop_blocked_places.display_string(ctxt)
  155         );
  156 
  157         for blocker in candidate_blockers.iter_places() {
  158             for root in root_places {
  159                 let relevant_root = root.relevant_place_for_blocking();
  160                 if blocker == relevant_root
  161                     || ctxt.ctxt.borrow_checker.blocks(
  162                         blocker,
  163                         relevant_root,
  164                         loop_head_location,
  165                         ctxt.bc_ctxt(),
  166                     )
  167                 {
  168                     tracing::debug!(
  169                         "{} blocks root {}",
  170                         blocker.display_string(ctxt),
  171                         relevant_root.display_string(ctxt)
  172                     );
  173                     add_block_edges(&mut expander, *root, blocker, ctxt);
  174                     if let MaybeRemoteCurrentPlace::Local(root) = root {
  175                         blocked_root_places.insert(*root);
  176                         for rp in root.lifetime_projections(ctxt) {
  177                             to_label.insert(LabelNodePredicate::all_non_future(
  178                                 (*root).into(),
  179                                 rp.region_idx,
  180                             ));
  181                         }
  182                     }
  183                 } else if root.is_remote() {
  184                     add_rp_block_edges(&mut expander, *root, blocker, ctxt);
  185                 }
  186             }
  187         }
  188 
  189         for blocked_place_usage in loop_blocked_places.iter() {
  190             let blocked_place = blocked_place_usage.place;
  191             let blockers = candidate_blockers
  192                 .iter_places()
  193                 .filter(|blocker| {
  194                     blocker.local != blocked_place.local
  195                         && ctxt.bc().blocks(
  196                             *blocker,
  197                             blocked_place,
  198                             loop_head_location,
  199                             ctxt.bc_ctxt(),
  200                         )
  201                 })
  202                 .collect::<Vec<_>>();
  203             if !blockers.is_empty() {
  204                 for blocker in &blockers {
  205                     add_block_edges(&mut expander, blocked_place.into(), *blocker, ctxt);
  206                 }
  207                 if blocked_place_usage.usage.is_read() {
  208                     capability_updates.insert(blocked_place, Some(CapabilityKind::Read));
  209                 } else {
  210                     capability_updates.insert(blocked_place, None);
  211                 }
  212                 for rp in blocked_place.lifetime_projections(ctxt) {
  213                     to_label.insert(LabelNodePredicate::all_non_future(
  214                         blocked_place.into(),
  215                         rp.region_idx,
  216                     ));
  217                 }
  218             }
  219         }
  220         expander.render_debug_graph(
  221             Some(DebugImgcat::JoinLoop),
  222             "Abstraction graph after connections expansion",
  223         );
  224 
  225         expander.expand_to_places(
  226             loop_blocked_places,
  227             &loop_blocked_places.joined_with(candidate_blockers),
  228         );
  229 
  230         expander.render_debug_graph(
  231             Some(DebugImgcat::JoinLoop),
  232             "Abstraction graph after expansion",
  233         );
  234 
  235         let abs_graph_roots = expander
  236             .graph
  237             .roots(ctxt)
  238             .into_iter()
  239             .filter_map(|graph_root| {
  240                 if let Some(PcgNode::LifetimeProjection(rp)) = graph_root.try_to_local_node(ctxt)
  241                     && let MaybeLabelledPlace::Current(place) = rp.base
  242                     && !loop_blocked_places.contains(place)
  243                 {
  244                     Some(rp)
  245                 } else {
  246                     None
  247                 }
  248             })
  249             .collect::<HashSet<_>>();
  250 
  251         for graph_root in abs_graph_roots {
  252             let mut candidate_root_nodes = self.nodes(ctxt);
  253             candidate_root_nodes.retain(|node| match node {
  254                 PcgNode::LifetimeProjection(region_projection)
  255                     if let Some(related_place) =
  256                         region_projection.base.maybe_remote_current_place() =>
  257                 {
  258                     root_places.contains(&related_place)
  259                 }
  260                 _ => false,
  261             });
  262             for node in candidate_root_nodes {
  263                 if self
  264                     .nodes_blocked_by(graph_root.into(), ctxt)
  265                     .contains(&node)
  266                 {
  267                     add_block_edge(&mut expander, node, graph_root.into(), ctxt);
  268                 }
  269             }
  270         }
  271 
  272         expander.render_debug_graph(
  273             Some(DebugImgcat::JoinLoop),
  274             "Abstraction graph after root reconnect",
  275         );
  276         let loop_head_label = LifetimeProjectionLabel::Location(SnapshotLocation::Loop(loop_head));
  277         let frozen_graph = graph.frozen_graph();
  278         tracing::debug!(
  279             "leaf edges: {}",
  280             frozen_graph.leaf_edges(ctxt).display_string(ctxt)
  281         );
  282         tracing::debug!(
  283             "leaf nodes: {}",
  284             frozen_graph.leaf_nodes(ctxt).display_string(ctxt)
  285         );
  286         for rp in &to_label {
  287             tracing::debug!("labeling {rp:?}");
  288             graph.label_lifetime_projections(rp, Some(loop_head_label), ctxt);
  289         }
  290         // Labeling can unify a loop abstraction edge's input and output onto
  291         // the same node: `all_non_future` (used to build `to_label`) matches
  292         // every non-future snapshot of a place's region, including ones that
  293         // were already labelled (e.g. a prior `Before(..)` snapshot), not
  294         // just the still-"current" one. When an edge's input was such an
  295         // older snapshot of the same place+region as its (then-current)
  296         // output, both endpoints get relabelled to the same loop-head
  297         // snapshot and the edge becomes a self-loop (`X --* X`). Such an
  298         // edge is a trivial/tautological wand: drop it, since it carries no
  299         // information and a self-blocking edge cannot be topologically
  300         // sorted by `UnblockGraph::actions`.
  301         let self_loop_edges: Vec<BorrowPcgEdgeKind<'tcx>> = graph
  302             .edges()
  303             .filter_map(|edge| {
  304                 let BorrowPcgEdgeKind::Abstraction(AbstractionEdge::Loop(loop_edge)) = edge.kind()
  305                 else {
  306                     return None;
  307                 };
  308                 let blocked_by: HashSet<_> = loop_edge
  309                     .blocked_by_nodes(ctxt)
  310                     .map(|n| n.to_pcg_node(ctxt))
  311                     .collect();
  312                 loop_edge
  313                     .blocked_nodes(ctxt)
  314                     .any(|node| blocked_by.contains(&node))
  315                     .then(|| edge.kind().clone())
  316             })
  317             .collect();
  318         for kind in &self_loop_edges {
  319             graph.remove(kind);
  320         }
  321         tracing::debug!("Completed loop abstraction");
  322         for (place, capability) in &capability_updates {
  323             logging::log!(
  324                 &LogPredicate::DebugBlock,
  325                 ctxt,
  326                 "capability update for {}: {:?}",
  327                 place.display_string(ctxt),
  328                 capability
  329             );
  330         }
  331         ConstructAbstractionGraphResult::new(
  332             graph,
  333             to_label,
  334             capability_updates,
  335             blocked_root_places,
  336         )
  337     }

  339     pub(crate) fn get_borrow_roots(
  340         &self,
  341         node: Place<'tcx>,
  342         loop_head_block: mir::BasicBlock,
  343         ctxt: CompilerCtxt<'_, 'tcx>,
  344     ) -> HashSet<PcgNode<'tcx>> {
  345         let mut result = HashSet::default();
  346         let mut queue: Vec<LocalNode<'tcx>> = node
  347             .lifetime_projections(ctxt)
  348             .into_iter()
  349             .flat_map(|rp| {
  350                 vec![
  351                     rp.to_local_node(ctxt),
  352                     rp.with_label(
  353                         Some(LifetimeProjectionLabel::Location(SnapshotLocation::Loop(
  354                             loop_head_block,
  355                         ))),
  356                         ctxt,
  357                     )
  358                     .to_local_node(ctxt),
  359                 ]
  360             })
  361             .collect();
  362         let mut seen = HashSet::default();
  363         while let Some(node) = queue.pop() {
  364             if seen.contains(&node) {
  365                 continue;
  366             }
  367             seen.insert(node);
  368             let mut blocked_edges = self.edges_blocked_by(node, ctxt).peekable();
  369             if blocked_edges.peek().is_none() {
  370                 result.insert(node.to_pcg_node(ctxt));
  371                 continue;
  372             }
  373             for edge in blocked_edges {
  374                 for blocked_by in edge.blocked_nodes(ctxt) {
  375                     match blocked_by.try_to_local_node(ctxt) {
  376                         Some(local_node) => {
  377                             if local_node.related_current_place().is_some() {
  378                                 result.insert(local_node.into());
  379                             } else {
  380                                 queue.push(local_node);
  381                             }
  382                         }
  383                         None => {
  384                             result.insert(blocked_by);
  385                         }
  386                     }
  387                 }
  388             }
  389         }
  390         tracing::debug!(
  391             "immediate live ancestors of {}: {}",
  392             node.display_string(ctxt),
  393             result.display_string(ctxt)
  394         );
  395         result
  396     }

  398     /// Unpacks the PCG at the loop head so that each place of the loop
  399     /// invariant capabilities `U` is accessible with the capability required by
  400     /// its usage.
  401     ///
  402     /// The places of `U` requiring write or exclusive access are unpacked
  403     /// before the read-only ones, so that an exclusive unpack is never applied
  404     /// to a place that has already been downgraded for a read.
  405     pub(crate) fn unpack_for_loop_invariant_capabilities<'a>(
  406         &mut self,
  407         loop_blocked_places: &PlaceUsages<'tcx>,
  408         invariant_capabilities: &PlaceUsages<'tcx>,
  409         validity_conditions: &ValidityConditions,
  410         args: &mut JoinBorrowsArgs<'_, 'a, 'tcx>,
  411         ctxt: AnalysisCtxt<'a, 'tcx>,
  412     ) {
  413         self.obtain_places_at_loop_head(
  414             loop_blocked_places,
  415             &invariant_capabilities.loop_head_unpack_order(),
  416             validity_conditions,
  417             args,
  418             ctxt,
  419         );
  420         for (place, capability) in args.capabilities.iter_mut() {
  421             if capability.is_exclusive()
  422                 && invariant_capabilities
  423                     .iter()
  424                     .any(|usage| usage.usage.is_read() && usage.place.is_prefix_of(*place))
  425             {
  426                 *capability = CapabilityKind::Read;
  427             }
  428         }
  429     }

  431     pub(crate) fn expand_places_for_abstraction<'a>(
  432         &mut self,
  433         loop_blocked_places: &PlaceUsages<'tcx>,
  434         to_expand: &PlaceUsages<'tcx>,
  435         validity_conditions: &ValidityConditions,
  436         args: &mut JoinBorrowsArgs<'_, 'a, 'tcx>,
  437         ctxt: AnalysisCtxt<'a, 'tcx>,
  438     ) {
  439         let mut to_obtain: Vec<PlaceUsage<'tcx>> = vec![];
  440         for place_usage in to_expand.iter() {
  441             if to_obtain
  442                 .iter()
  443                 .any(|p| place_usage.place.is_prefix_of(p.place))
  444             {
  445                 continue;
  446             }
  447             to_obtain.retain(|p| !place_usage.place.is_prefix_of(p.place));
  448             to_obtain.push(place_usage);
  449         }
  450         self.obtain_places_at_loop_head(
  451             loop_blocked_places,
  452             &to_obtain,
  453             validity_conditions,
  454             args,
  455             ctxt,
  456         );
  457     }

  459     fn obtain_places_at_loop_head<'a>(
  460         &mut self,
  461         loop_blocked_places: &PlaceUsages<'tcx>,
  462         to_obtain: &[PlaceUsage<'tcx>],
  463         validity_conditions: &ValidityConditions,
  464         args: &mut JoinBorrowsArgs<'_, 'a, 'tcx>,
  465         ctxt: AnalysisCtxt<'a, 'tcx>,
  466     ) {
  467         let borrow = BorrowStateMutRef::new(self, validity_conditions);
  468         let pcg = PcgMutRef::new(borrow, args.capabilities, args.owned);
  469         let snapshot_location = SnapshotLocation::Loop(args.self_block);
  470         let mut obtainer = PlaceObtainer::new(
  471             pcg,
  472             None,
  473             ctxt,
  474             mir::Location {
  475                 block: args.self_block,
  476                 statement_index: 0,
  477             },
  478             snapshot_location,
  479         );
  480         obtainer.render_debug_graph(Some(DebugImgcat::JoinLoop), "Before obtaining (self)");
  481         for place_usage in to_obtain {
  482             let obtain_type = ObtainType::LoopInvariant {
  483                 is_blocked: loop_blocked_places.contains(place_usage.place),
  484                 usage_type: place_usage.usage,
  485             };
  486             obtainer.obtain(place_usage.place, obtain_type).unwrap();
  487             obtainer.render_debug_graph(
  488                 Some(DebugImgcat::JoinLoop),
  489                 &format!(
  490                     "After obtaining (self) {}",
  491                     place_usage.display_string(ctxt.ctxt)
  492                 ),
  493             );
  494         }
  495     }

  497     /// Cutting a hyperedge can disconnect outputs outside the abstraction,
  498     /// including sibling fields introduced by an expansion. Preserve the
  499     /// paths to these outputs from the boundary of the removed subgraph.
  500     pub(crate) fn reconnect_cut_outputs<'a>(
  501         &self,
  502         to_cut: &Self,
  503         abstraction: &mut Self,
  504         loop_head: mir::BasicBlock,
  505         validity_conditions: &ValidityConditions,
  506         ctxt: AnalysisCtxt<'a, 'tcx>,
  507     ) {
  508         let abstraction_nodes = abstraction.nodes(ctxt);
  509         let outputs = to_cut
  510             .nodes(ctxt)
  511             .into_iter()
  512             .filter(|node| !abstraction_nodes.contains(node))
  513             .filter_map(|node| node.try_to_local_node(ctxt))
  514             .filter(|node| {
  515                 to_cut.edges_blocking((*node).into(), ctxt).next().is_none()
  516                     || self
  517                         .edges_blocking((*node).into(), ctxt)
  518                         .any(|edge| !to_cut.edges.contains_key(edge.kind()))
  519             });
  520         let mut expander = AbsExpander {
  521             loop_head_block: loop_head,
  522             graph: abstraction,
  523             validity_conditions,
  524             ctxt,
  525         };
  526         for output in outputs {
  527             let mut pending = to_cut.nodes_blocked_by(output, ctxt);
  528             let mut seen = HashSet::default();
  529             while let Some(node) = pending.pop() {
  530                 if !seen.insert(node) {
  531                     continue;
  532                 }
  533                 let predecessors = node
  534                     .try_to_local_node(ctxt)
  535                     .map(|node| to_cut.nodes_blocked_by(node, ctxt))
  536                     .unwrap_or_default();
  537                 let in_abstraction = abstraction_nodes.contains(&node);
  538                 let has_retained_input = node.try_to_local_node(ctxt).is_some_and(|node| {
  539                     self.edges_blocked_by(node, ctxt)
  540                         .any(|edge| !to_cut.edges.contains_key(edge.kind()))
  541                 });
  542                 if in_abstraction || predecessors.is_empty() || has_retained_input {
  543                     add_block_edge(&mut expander, node, output, ctxt);
  544                 }
  545                 if !in_abstraction {
  546                     pending.extend(predecessors);
  547                 }
  548             }
  549         }
  550     }

  552     /// Identifies the edges of `self` on paths from a node of
  553     /// `abstraction_graph` to one of its outputs; these are the edges that the
  554     /// abstraction replaces.
  555     ///
  556     /// A path that reaches an input of the abstraction is not cut: that input
  557     /// remains in the graph, so the edges producing it must remain as well.
  558     pub(crate) fn identify_subgraph_to_cut<'mir: 'graph, 'graph>(
  559         &'graph self,
  560         block: mir::BasicBlock,
  561         abstraction_graph: &BorrowsGraph<'tcx>,
  562         ctxt: impl HasSettings<'mir> + HasBorrowCheckerCtxt<'mir, 'tcx>,
  563     ) -> BorrowsGraph<'tcx>
  564     where
  565         'tcx: 'mir,
  566     {
  567         type Path<'tcx, 'graph> = Vec<BorrowPcgEdgeRef<'tcx, 'graph>>;
  568         let abstraction_graph_nodes = abstraction_graph.nodes(ctxt);
  569         let abstraction_outputs = abstraction_graph
  570             .edges()
  571             .flat_map(|edge| edge.blocked_by_nodes(ctxt).collect::<Vec<_>>())
  572             .map(|node| node.to_pcg_node(ctxt))
  573             .collect::<HashSet<_>>();
  574         let mut to_cut = HashSet::default();
  575         let mut paths: Vec<Path<'tcx, 'graph>> = abstraction_graph_nodes
  576             .iter()
  577             .flat_map(|node| self.edges_blocking(*node, ctxt))
  578             .map(|edge| vec![edge])
  579             .collect::<Vec<_>>();
  580         'outer: while let Some(path) = paths.pop() {
  581             let last_edge = *path.last().unwrap();
  582             if to_cut.contains(&last_edge) {
  583                 to_cut.extend(path);
  584                 continue;
  585             }
  586             let blocked_by_nodes = last_edge.blocked_by_nodes(ctxt).collect::<Vec<_>>();
  587             if blocked_by_nodes
  588                 .iter()
  589                 .any(|node| abstraction_outputs.contains(&node.to_pcg_node(ctxt)))
  590             {
  591                 to_cut.extend(path);
  592                 continue;
  593             }
  594             if blocked_by_nodes
  595                 .iter()
  596                 .any(|node| abstraction_graph_nodes.contains(&node.to_pcg_node(ctxt)))
  597             {
  598                 continue;
  599             }
  600             for blocked_by_node in blocked_by_nodes {
  601                 for edge in self.edges_blocking(blocked_by_node.into(), ctxt) {
  602                     if path.contains(&edge) {
  603                         let pc = PlaceCapabilities::default();
  604                         self.render_debug_graph(
  605                             block,
  606                             Some(DebugImgcat::JoinLoop),
  607                             &pc,
  608                             "Invalid abstraction graph",
  609                             ctxt,
  610                         );
  611                         pcg_validity_assert!(false, [ctxt], "edge already in path");
  612                         // panic!("edge already in path");
  613                         // For debugging, just stop here and we can try to visualize the graph
  614                         break 'outer;
  615                     }
  616                     let mut next_path = path.clone();
  617                     next_path.push(edge);
  618                     paths.push(next_path);
  619                 }
  620             }
  621         }
  622         let mut graph = BorrowsGraph::default();
  623         for edge in to_cut {
  624             graph.insert(edge.to_owned_edge(), ctxt);
  625         }
  626         graph
  627     }
  628 }

  673 impl<'tcx> ActionApplier<'tcx> for AbsExpander<'_, '_, 'tcx> {
  674     fn apply_action(&mut self, action: PcgAction<'tcx>) -> Result<(), crate::error::PcgError> {
  675         tracing::debug!("applying action: {}", action.debug_line(self.ctxt));
  676         match action {
  677             PcgAction::Borrow(action) => match action.kind {
  678                 BorrowPcgActionKind::AddEdge { edge } => {
  679                     self.graph.insert(edge, self.ctxt);
  680                 }
  681                 BorrowPcgActionKind::LabelLifetimeProjection(action) => {
  682                     self.graph.label_lifetime_projections(
  683                         action.predicate(),
  684                         action.label(),
  685                         self.ctxt,
  686                     );
  687                 }
  688                 BorrowPcgActionKind::RemoveEdge(borrow_pcg_edge) => {
  689                     self.graph.remove(borrow_pcg_edge.kind());
  690                 }
  691                 _ => unreachable!(),
  692             },
  693             _ => unreachable!(),
  694         }
  695         Ok(())
  696     }
  697 }

  750 fn add_block_edge<'tcx, 'mir, Ctxt: HasBorrowCheckerCtxt<'mir, 'tcx> + HasSettings<'mir>>(
  751     expander: &mut AbsExpander<'_, 'mir, 'tcx>,
  752     long: PcgNode<'tcx>,
  753     short: LocalNode<'tcx>,
  754     ctxt: Ctxt,
  755 ) {
  756     let long_edge = AbstractionBlockEdge::new_checked(long.into(), short.into(), ctxt);
  757     let loop_edge = LoopAbstraction::new(long_edge, expander.loop_head_block);
  758     expander.graph.insert(
  759         BorrowPcgEdge::new(
  760             BorrowPcgEdgeKind::Abstraction(loop_edge.into()),
  761             expander.validity_conditions.clone(),
  762         ),
  763         ctxt,
  764     );
  765 }

  767 fn add_rp_block_edges<'mir, 'tcx, Ctxt: HasBorrowCheckerCtxt<'mir, 'tcx> + HasSettings<'mir>>(
  768     expander: &mut AbsExpander<'_, 'mir, 'tcx>,
  769     blocked_place: MaybeRemoteCurrentPlace<'tcx>,
  770     blocker: Place<'tcx>,
  771     ctxt: Ctxt,
  772 ) {
  773     let blocker_rps = blocker.lifetime_projections(ctxt);
  774     for blocked_rp in blocked_place.lifetime_projections(ctxt) {
  775         let flow_rps = blocker_rps
  776             .iter()
  777             .filter(|blocker_rp| {
  778                 ctxt.bc().outlives_generalized(
  779                     blocked_rp.region(ctxt),
  780                     blocker_rp.region(ctxt),
  781                     expander.loop_head_location(),
  782                 )
  783             })
  784             .copied()
  785             .collect::<Vec<_>>();
  786         if let Some(blocked_node) = blocked_rp.try_to_local_node(ctxt) {
  787             let blocked_rp = blocked_node.expect_lifetime_projection();
  788             let mut_rps = flow_rps
  789                 .iter()
  790                 .filter_map(|rp| {
  791                     if rp.is_invariant_in_type(ctxt)
  792                         && ctxt.bc().outlives_generalized(
  793                             rp.region(ctxt),
  794                             blocked_rp.region(ctxt),
  795                             expander.loop_head_location(),
  796                         )
  797                     {
  798                         Some(rp.rebase())
  799                     } else {
  800                         None
  801                     }
  802                 })
  803                 .collect::<Vec<_>>();
  804             if !mut_rps.is_empty() {
  805                 tracing::debug!(
  806                     "Mutable rps: {} -> {}",
  807                     blocked_rp.display_string(ctxt),
  808                     mut_rps.display_string(ctxt)
  809                 );
  810                 expander
  811                     .add_and_update_placeholder_edges(blocked_rp, &mut_rps, "mut rps", ctxt)
  812                     .unwrap();
  813                 expander.render_debug_graph(
  814                     Some(DebugImgcat::JoinLoop),
  815                     "Abstraction graph after adding mut rps",
  816                 );
  817             }
  818         }
  819         for flow_rp in flow_rps {
  820             add_block_edge(
  821                 expander,
  822                 blocked_rp.to_pcg_node(ctxt),
  823                 flow_rp.to_local_node(ctxt),
  824                 ctxt,
  825             );
  826         }
  827     }
  828 }

  830 fn add_block_edges<'mir, 'tcx, Ctxt: HasBorrowCheckerCtxt<'mir, 'tcx> + HasSettings<'mir>>(
  831     expander: &mut AbsExpander<'_, 'mir, 'tcx>,
  832     blocked_place: MaybeRemoteCurrentPlace<'tcx>,
  833     blocker: Place<'tcx>,
  834     ctxt: Ctxt,
  835 ) {
  836     assert_ne!(
  837         MaybeRemoteCurrentPlace::Local(blocker),
  838         blocked_place,
  839         "blocker {} and blocked_place {} are the same",
  840         blocker.display_string(ctxt),
  841         blocked_place.display_string(ctxt)
  842     );
  843     if let MaybeRemoteCurrentPlace::Local(blocked_place) = blocked_place {
  844         let blocker_rps = blocker.lifetime_projections(ctxt);
  845         add_block_edge(
  846             expander,
  847             blocked_place.to_pcg_node(ctxt),
  848             blocker_rps[LifetimeProjectionIdx::from(0)].to_local_node(ctxt),
  849             ctxt,
  850         );
  851     }
  852     add_rp_block_edges(expander, blocked_place, blocker, ctxt);
  853 }
