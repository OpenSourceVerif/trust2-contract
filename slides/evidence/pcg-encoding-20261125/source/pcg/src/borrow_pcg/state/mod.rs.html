   61 fn map_label_predicate<'tcx, P: Copy>(
   62     predicate: &LabelNodePredicate<'tcx, P>,
   63 ) -> LabelNodePredicate<'tcx, P> {
   64     match predicate {
   65         LabelNodePredicate::LifetimeProjectionLabelEquals(label) => {
   66             LabelNodePredicate::LifetimeProjectionLabelEquals(*label)
   67         }
   68         LabelNodePredicate::PlaceLabelEquals(location) => {
   69             LabelNodePredicate::PlaceLabelEquals(*location)
   70         }
   71         LabelNodePredicate::ProjectionRegionIdxEquals(region_idx) => {
   72             LabelNodePredicate::ProjectionRegionIdxEquals(*region_idx)
   73         }
   74         LabelNodePredicate::Equals(node) => LabelNodePredicate::Equals(*node),
   75         LabelNodePredicate::PlaceEquals(place) => LabelNodePredicate::PlaceEquals(*place),
   76         LabelNodePredicate::PlaceIsPostfixOf(place) => LabelNodePredicate::PlaceIsPostfixOf(*place),
   77         LabelNodePredicate::NodeType(node_type) => LabelNodePredicate::NodeType(*node_type),
   78         LabelNodePredicate::And(predicates) => {
   79             LabelNodePredicate::And(predicates.iter().map(map_label_predicate).collect())
   80         }
   81         LabelNodePredicate::Or(predicates) => {
   82             LabelNodePredicate::Or(predicates.iter().map(map_label_predicate).collect())
   83         }
   84         LabelNodePredicate::Not(predicate) => {
   85             LabelNodePredicate::Not(Box::new(map_label_predicate(predicate)))
   86         }
   87         LabelNodePredicate::EdgeType(edge_type) => LabelNodePredicate::EdgeType(*edge_type),
   88         LabelNodePredicate::InSourceNodes => LabelNodePredicate::InSourceNodes,
   89         LabelNodePredicate::InTargetNodes => LabelNodePredicate::InTargetNodes,
   90     }
   91 }

  191     fn label_place_and_update_related_capabilities<
  192         'a,
  193         P: PlaceLike<'tcx, Ctxt> + DisplayWithCtxt<Ctxt>,
  194         Ctxt: Copy + HasCompilerCtxt<'a, 'tcx>,
  195         C,
  196     >(
  197         &mut self,
  198         place: P,
  199         reason: LabelPlaceReason,
  200         labeller: &impl PlaceLabeller<'tcx, Ctxt, P>,
  201         capabilities: &mut impl PlaceCapabilitiesInterface<'tcx, C, P>,
  202         ctxt: Ctxt,
  203     ) -> ApplyActionResult
  204     where
  205         'tcx: 'a,
  206         EdgeKind: LabelEdgePlaces<'tcx, Ctxt, P> + Eq + std::hash::Hash,

  208         PcgNodeWithPlace<'tcx, P>: DisplayWithCtxt<Ctxt>,
  209     {
  210         let state = self.as_mut_ref();
  211         let replacements = state.graph.label_place(place, reason, labeller, ctxt);

  213         // essentially be handled by the join logic.
  214         // See 69_http_header_map.rs
  215         if reason != LabelPlaceReason::JoinOwnedReadAndWriteCapabilities {
  216             capabilities.retain(|p, _| !p.projects_indirection_from(place, ctxt));
  217         }
  218         let display = display_node_replacements(&replacements, ctxt, OutputMode::Normal);
  219         ApplyActionResult {
  220             changed: true,
  221             change_summary: display,
  222         }
  223     }

  240     fn remove<'a, P: Copy + Eq + std::hash::Hash, Ctxt: Copy, C>(
  241         &mut self,
  242         edge: &EdgeKind,
  243         capabilities: &mut impl PlaceCapabilitiesInterface<'tcx, C, P>,
  244         ctxt: Ctxt,
  245     ) -> bool
  246     where
  247         'tcx: 'a,
  248         EdgeKind: EdgeData<'tcx, Ctxt, P> + Eq + std::hash::Hash,
  249     {
  250         let state = self.as_mut_ref();
  251         let removed = state.graph.remove(edge).is_some();
  252         if removed {
  253             for node in edge.blocked_by_nodes(ctxt) {
  254                 if !state.graph.contains(node, ctxt)
  255                     && let PcgNode::Place(MaybeLabelledPlace::Current(place)) = node
  256                 {
  257                     let _ = capabilities.remove(place, ctxt);
  258                 }
  259             }
  260         }
  261         removed
  262     }

  264     fn apply_action<
  265         'a,
  266         Ctxt: DebugCtxt + HasCompilerCtxt<'a, 'tcx> + HasSettings<'a>,
  267         P: PlaceLike<'tcx, Ctxt> + DisplayWithCtxt<Ctxt>,
  268     >(
  269         &mut self,
  270         action: BorrowPcgAction<'tcx, EdgeKind, P, VC>,
  271         capabilities: &mut impl PlaceCapabilitiesInterface<'tcx, CapabilityKind, P>,
  272         ctxt: Ctxt,
  273     ) -> ApplyActionResult
  274     where
  275         'tcx: 'a,
  276         VC: ValidityConditionOps<Ctxt>,
  277         EdgeKind: EdgeData<'tcx, Ctxt, P>
  278             + LabelEdgePlaces<'tcx, Ctxt, P>
  279             + LabelEdgeLifetimeProjections<'tcx, Ctxt, P>
  280             + Eq
  281             + std::hash::Hash,
  282         MaybeRemotePlace<'tcx, P>: DisplayWithCtxt<Ctxt>,
  283     {
  284         match action.kind {
  285             BorrowPcgActionKind::Restore(restore) => {
  286                 let restore_place = restore.place();
  287                 if let Some(cap) = capabilities.get(restore_place, ctxt) {
  288                     pcg_validity_assert!(cap < restore.capability());
  289                 }
  290                 if !capabilities.insert(restore_place, restore.capability(), ctxt) {
  291                     // panic!("Capability should have been updated")
  292                 }
  293                 if restore.capability() == CapabilityKind::Exclusive {
  294                     self.label_lifetime_projections(
  295                         &LabelNodePredicate::all_future_postfixes(restore_place),
  296                         None,
  297                         ctxt,
  298                     );
  299                 }
  300                 ApplyActionResult::changed_no_display()
  301             }
  302             BorrowPcgActionKind::Weaken(weaken) => {
  303                 let weaken_place = weaken.place();
  304                 pcg_validity_assert!(
  305                     capabilities.get(weaken_place, ctxt).unwrap() == weaken.from,
  306                     [ctxt],
  307                     "Weakening from {:?} to {:?} is not valid",
  308                     capabilities.get(weaken_place, ctxt).unwrap(),
  309                     weaken.from
  310                 );
  311                 match weaken.to {
  312                     Some(to) => {
  313                         capabilities.insert(weaken_place, to, ctxt);
  314                     }
  315                     None => {
  316                         assert!(capabilities.remove(weaken_place, ctxt).is_some());
  317                     }
  318                 }
  319                 ApplyActionResult::changed_no_display()
  320             }
  321             BorrowPcgActionKind::LabelPlace(action) => self
  322                 .label_place_and_update_related_capabilities(
  323                     action.place,
  324                     action.reason,
  325                     &SetLabel(action.location),
  326                     capabilities,
  327                     ctxt,
  328                 ),
  329             BorrowPcgActionKind::RemoveEdge(edge) => {
  330                 ApplyActionResult::from_changed(self.remove(&edge.value, capabilities, ctxt))
  331             }
  332             BorrowPcgActionKind::AddEdge { edge } => {
  333                 ApplyActionResult::from_changed(self.graph_mut().insert(edge, ctxt))
  334             }
  335             BorrowPcgActionKind::LabelLifetimeProjection(action) => {
  336                 let predicate = map_label_predicate(action.predicate());
  337                 ApplyActionResult::from_changed(self.label_lifetime_projections(
  338                     &predicate,
  339                     action.label(),
  340                     ctxt,
  341                 ))
  342             }
  343         }
  344     }

  422 impl<'a, 'tcx> BorrowsState<'a, 'tcx, BorrowPcgEdgeKind<'tcx>, ValidityConditions> {
  423     fn introduce_initial_borrows(
  424         &mut self,
  425         local: mir::Local,
  426         capabilities: &mut impl PlaceCapabilitiesInterface<'tcx, CapabilityKind, Place<'tcx>>,
  427         ctxt: AnalysisCtxt<'a, 'tcx>,
  428     ) {
  429         let arg_place: Place<'tcx> = local.into();
  430         for lifetime in arg_place.generalized_lifetimes(ctxt) {
  431             let source_projection: LifetimeProjection<'tcx, RemotePlace> =
  432                 LifetimeProjection::new(RemotePlace::new(local), lifetime, None, ctxt)
  433                     .unwrap_or_else(|| {
  434                         panic!(
  435                             "Failed to create region for remote place (for {local:?}).
  436                                     It does not have lifetime {lifetime:?}",
  437                         );
  438                     });
  439             let source_projection: LifetimeProjection<'tcx, PcgLifetimeProjectionBase<'tcx>> =
  440                 source_projection.rebase();
  441             let target_projection = LifetimeProjection::<'tcx, MaybeLabelledPlace<'tcx>>::new(
  442                 MaybeLabelledPlace::Current(arg_place),
  443                 lifetime,
  444                 None,
  445                 ctxt,
  446             )
  447             .unwrap();
  448             assert!(
  449                 self.apply_action(
  450                     BorrowPcgAction::add_edge(
  451                         BorrowPcgEdge::new(
  452                             BorrowFlowEdge::new(
  453                                 source_projection,
  454                                 target_projection,
  455                                 BorrowFlowEdgeKind::InitialBorrows,
  456                             )
  457                             .into(),
  458                             ValidityConditions::new(),
  459                         ),
  460                         "Introduce initial borrows",
  461                     ),
  462                     capabilities,
  463                     ctxt
  464                 )
  465                 .changed
  466             );
  467         }
  468     }

  470     pub(crate) fn start_block(
  471         capabilities: &mut impl PlaceCapabilitiesInterface<'tcx, CapabilityKind, Place<'tcx>>,
  472         ctxt: AnalysisCtxt<'a, 'tcx>,
  473     ) -> Self {
  474         let mut borrow = Self::default();
  475         for arg in ctxt.args_iter() {

  538     #[allow(clippy::too_many_arguments)]
  539     pub(crate) fn add_borrow<Ctxt: HasBorrowCheckerCtxt<'a, 'tcx> + HasSettings<'a>>(
  540         &mut self,
  541         blocked_place: Place<'tcx>,
  542         assigned_place: Place<'tcx>,
  543         kind: BorrowKind,
  544         location: Location,
  545         region: ty::Region<'tcx>,
  546         capabilities: &mut PlaceCapabilities<'tcx>,
  547         ctxt: Ctxt,
  548     ) where
  549         'tcx: 'a,
  550     {
  551         assert!(
  552             assigned_place.ty(ctxt).ty.ref_mutability().is_some(),
  553             "{:?}:{:?} Assigned place {:?} is not a reference. Ty: {:?}",
  554             ctxt.body().source.def_id(),
  555             location,
  556             assigned_place,
  557             assigned_place.ty(ctxt).ty
  558         );
  559         let borrow_edge = BorrowEdge::new(
  560             blocked_place.into(),
  561             assigned_place.into(),
  562             kind,
  563             location,
  564             region,
  565             ctxt,
  566         );
  567         assert!(self.graph.insert(
  568             BorrowPcgEdge::new(
  569                 BorrowPcgEdgeKind::Borrow(borrow_edge),
  570                 self.validity_conditions.clone(),
  571             ),
  572             ctxt.ctxt()
  573         ));
  574 
  575         if let BorrowKind::Mut {
  576             kind: MutBorrowKind::Default | MutBorrowKind::ClosureCapture,
  577         } = kind
  578         {
  579             let _ = capabilities.remove(blocked_place, ctxt);
  580         } else {
  581             let blocked_place_capability = capabilities.get(blocked_place, ctxt);
  582             match blocked_place_capability {
  583                 Some(CapabilityKind::Exclusive) => {
  584                     assert!(capabilities.insert(blocked_place, CapabilityKind::Read, ctxt));
  585                 }
  586                 Some(CapabilityKind::Read) => {
  587                     // Do nothing, this just adds another shared borrow
  588                 }
  589                 other => {
  590                     // Shouldn't be None or Write, due to capability updates
  591                     // based on the TripleWalker analysis
  592                     pcg_validity_assert!(
  593                         false,
  594                         "{:?}: Unexpected capability for borrow blocked place {:?}: {:?}",
  595                         location,
  596                         blocked_place,
  597                         other
  598                     );
  599                 }
  600             }
  601         }
  602     }
  603 }
  604 
  605 impl<'a, 'tcx, EdgeKind: PartialEq + Eq + std::hash::Hash> BorrowsState<'a, 'tcx, EdgeKind> {
  606     pub(crate) fn add_cfg_edge(
  607         &mut self,
  608         from: BasicBlock,
  609         to: BasicBlock,
  610         ctxt: AnalysisCtxt<'a, 'tcx>,
  611     ) -> bool {
  612         pcg_validity_assert!(
  613             !ctxt.ctxt.is_back_edge(from, to),
  614             [ctxt],
  615             "Adding CFG edge from {from:?} to {to:?} is a back edge"
  616         );
  617         let pc = PathCondition::new(from, to);
  618         let mut validity_conditions = self.validity_conditions.clone();
  619         validity_conditions.insert(pc, ctxt.body());
  620         self.validity_conditions = ctxt.arena.alloc(validity_conditions);
  621         self.graph.add_path_condition(pc, ctxt.ctxt)
  622     }
  623 }
