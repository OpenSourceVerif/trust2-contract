  100 impl<'ops, 'tcx: 'ops> FunctionCall<'ops, 'tcx> {
  101     pub(crate) fn defined_fn_call_with_call_tys<'a>(
  102         self,
  103         ctxt: impl HasCompilerCtxt<'a, 'tcx>,
  104     ) -> Option<DefinedFnCallWithCallTys<'tcx>>
  105     where
  106         'tcx: 'a,
  107     {
  108         self.defined.map(|defined| {
  109             DefinedFnCallWithCallTys::new(
  110                 defined,
  111                 self.inputs
  112                     .iter()
  113                     .map(|input| input.ty(ctxt.body(), ctxt.tcx()))
  114                     .collect(),
  115                 self.output.ty(ctxt).ty,
  116             )
  117         })
  118     }
  119 
  120     fn defined_fn_call_shape<'a, Ctxt: HasBorrowCheckerCtxt<'a, 'tcx> + HasSettings<'a>>(
  121         self,
  122         ctxt: Ctxt,
  123     ) -> Option<FunctionShape>
  124     where
  125         'tcx: 'a,
  126     {
  127         let defined = self.defined_fn_call_with_call_tys(ctxt)?;
  128         let data_source = DefinedFnCallShapeDataSource::new(defined, self.location, ctxt).ok()?;
  129         Some(FunctionShape::new(&data_source, ctxt))
  130     }
  131 
  132     /// Computes the shape for this function call. For calls to defined
  133     /// functions, uses the signature-derived call shape: it computes the
  134     /// signature shape using the instantiated signature types, then remaps
  135     /// region indices to the call-site types (handling unnormalized alias types
  136     /// in the sig).  For other calls (e.g. calls to a function pointer), the
  137     /// shape is computed by [`FnCallDataSource`], which queries the borrow
  138     /// checker for outlives constraints between the regions in the call site
  139     /// operands and return place directly.
  140     ///
  141     /// See <https://prusti.github.io/pcg-docs/function-shapes.html> for more info.
  142     pub(crate) fn shape<'a>(
  143         self,
  144         ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx> + HasSettings<'a>,
  145     ) -> FunctionShape
  146     where
  147         'tcx: 'a,
  148     {
  149         let input_tys = self
  150             .inputs
  151             .iter()
  152             .map(|input| input.ty(ctxt.body(), ctxt.tcx()))
  153             .collect();
  154         let output_ty = self.output.ty(ctxt).ty;
  155         self.defined_fn_call_shape(ctxt).unwrap_or_else(|| {
  156             FunctionShape::new(
  157                 &FnCallDataSource::new(self.location, input_tys, output_ty),
  158                 ctxt,
  159             )
  160         })
  161     }
  162     pub(crate) fn new(
  163         location: mir::Location,
  164         inputs: &'ops [&'ops mir::Operand<'tcx>],
  165         output: utils::Place<'tcx>,
  166         defined: Option<DefinedFnCall<'tcx>>,
  167     ) -> Self {
  168         Self {
  169             defined,
  170             location,
  171             inputs,
  172             output,
  173         }
  174     }
  175 }

  177 pub(crate) trait FunctionShapeDataSource<'tcx, Ctxt: HasTyCtxt<'tcx> + Copy> {
  178     /// The lifetime representation used by this data source: [`PcgRegion`] for
  179     /// lifetime projections (call shapes), [`GeneralizedLifetime`] for
  180     /// generalized lifetime projections (signature shapes).
  181     type Lifetime: LifetimeKind<'tcx>;
  182 
  183     fn input_tys(&self, ctxt: Ctxt) -> Vec<ty::Ty<'tcx>>;
  184     fn output_ty(&self, ctxt: Ctxt) -> ty::Ty<'tcx>;
  185     fn outlives(&self, sup: Self::Lifetime, sub: Self::Lifetime, ctxt: Ctxt) -> bool;
  186 
  187     fn input_arg_projections(&self, ctxt: Ctxt) -> Vec<ProjectionData<'tcx, ArgIdx, Self::Lifetime>>
  188     where
  189         Ctxt: ExtractRegionsCtxt<'tcx, ty::Ty<'tcx>, Self::Lifetime>,
  190     {
  191         self.input_tys(ctxt)
  192             .into_iter()
  193             .enumerate()
  194             .flat_map(|(i, ty)| ProjectionData::nodes_for_ty(i.into(), ty, ctxt))
  195             .collect()
  196     }
  197 
  198     fn result_projections(
  199         &self,
  200         ctxt: Ctxt,
  201     ) -> Vec<ProjectionData<'tcx, ArgIdxOrResult, Self::Lifetime>>
  202     where
  203         Ctxt: ExtractRegionsCtxt<'tcx, ty::Ty<'tcx>, Self::Lifetime>,
  204     {
  205         ProjectionData::nodes_for_ty(ArgIdxOrResult::Result, self.output_ty(ctxt), ctxt)
  206     }
  207 
  208     fn inputs(
  209         &self,
  210         ctxt: Ctxt,
  211     ) -> Vec<FunctionShapeInput<<Self::Lifetime as LifetimeKind<'tcx>>::IdxMarker>>
  212     where
  213         Ctxt: ExtractRegionsCtxt<'tcx, ty::Ty<'tcx>, Self::Lifetime>,
  214     {
  215         self.input_arg_projections(ctxt)
  216             .into_iter()
  217             .map(|pd| FunctionShapeInput::from_index(pd.base, pd.region_idx))
  218             .collect()
  219     }
  220 
  221     fn outputs(
  222         &self,
  223         ctxt: Ctxt,
  224     ) -> Vec<FunctionShapeOutput<<Self::Lifetime as LifetimeKind<'tcx>>::IdxMarker>>
  225     where
  226         Ctxt: ExtractRegionsCtxt<'tcx, ty::Ty<'tcx>, Self::Lifetime>,
  227     {
  228         self.result_projections(ctxt)
  229             .into_iter()
  230             .map(|pd| FunctionShapeOutput::from_index(pd.base, pd.region_idx))
  231             .collect()
  232     }
  233 }

  243 impl<'tcx, T: Copy, L: LifetimeKind<'tcx>> ProjectionData<'tcx, T, L> {
  244     fn nodes_for_ty<Ctxt: ExtractRegionsCtxt<'tcx, ty::Ty<'tcx>, L>>(
  245         base: T,
  246         ty: ty::Ty<'tcx>,
  247         ctxt: Ctxt,
  248     ) -> Vec<Self> {
  249         ctxt.extract_regions(ty)
  250             .into_iter()
  251             .enumerate()
  252             .map(|(region_idx, region)| Self {
  253                 base,
  254                 ty,
  255                 region,
  256                 region_idx: region_idx.into(),
  257             })
  258             .collect()
  259     }
  260 
  261     /// Like [`Self::nodes_for_ty`], but uses pre-extracted lifetimes instead of
  262     /// calling [`LifetimeKind::extract`].
  263     pub(crate) fn nodes_for_extracted(
  264         base: T,
  265         ty: ty::Ty<'tcx>,
  266         lifetimes: IndexVec<LifetimeProjectionIdx<L::IdxMarker>, L>,
  267     ) -> Vec<Self> {
  268         lifetimes
  269             .into_iter()
  270             .enumerate()
  271             .map(|(region_idx, region)| Self {
  272                 base,
  273                 ty,
  274                 region,
  275                 region_idx: region_idx.into(),
  276             })
  277             .collect()
  278     }
  279 }

  369 /// A bipartite graph describing the shape of a function. Note that *outputs*
  370 /// include lifetime projections of nested lifetimes in the function arguments.
  371 ///
  372 /// The `Kind` parameter distinguishes between shapes built from lifetime
  373 /// projections ([`Region`], the default, used for call shapes) and generalized
  374 /// lifetime projections ([`Generalized`], used for signature shapes).
  375 #[derive(PartialEq, Eq, Clone, Debug, Hash)]
  376 pub struct FunctionShape<Kind: RegionIdxMarker = Generalized> {
  377     inputs: Vec<FunctionShapeInput<Kind>>,
  378     outputs: Vec<FunctionShapeOutput<Kind>>,
  379     edges: BTreeSet<
  380         AbstractionBlockEdge<'static, FunctionShapeInput<Kind>, FunctionShapeOutput<Kind>>,
  381     >,
  382 }
  383 
  384 impl<Kind: RegionIdxMarker> FunctionShape<Kind> {
  385     pub fn edges(
  386         &self,
  387     ) -> impl Iterator<
  388         Item = AbstractionBlockEdge<'static, FunctionShapeInput<Kind>, FunctionShapeOutput<Kind>>,
  389     > + '_ {
  390         self.edges.iter().copied()
  391     }
  392 
  393     #[must_use]
  394     pub fn take_inputs_and_outputs(
  395         self,
  396     ) -> (
  397         Vec<FunctionShapeInput<Kind>>,
  398         Vec<FunctionShapeOutput<Kind>>,
  399     ) {
  400         (self.inputs, self.outputs)
  401     }
  402 
  403     /// Constructs a `FunctionShape` from explicit inputs, outputs, and edges.
  404     #[must_use]
  405     pub fn from_raw(
  406         inputs: Vec<(ArgIdx, usize)>,
  407         outputs: Vec<(ArgIdxOrResult, usize)>,
  408         edges: HashSet<((ArgIdx, usize), (ArgIdxOrResult, usize))>,
  409     ) -> Self {
  410         let inputs = inputs
  411             .into_iter()
  412             .map(|(base, region)| FunctionShapeInput::from_index(base, region.into()))
  413             .collect();
  414         let outputs = outputs
  415             .into_iter()
  416             .map(|(base, region)| FunctionShapeOutput::from_index(base, region.into()))
  417             .collect();
  418         let edges = edges
  419             .into_iter()
  420             .map(|((in_base, in_region), (out_base, out_region))| {
  421                 AbstractionBlockEdge::new(
  422                     FunctionShapeInput::from_index(in_base, in_region.into()),
  423                     FunctionShapeOutput::from_index(out_base, out_region.into()),
  424                 )
  425             })
  426             .collect();
  427         Self {
  428             inputs,
  429             outputs,
  430             edges,
  431         }
  432     }
  433 }

  566 impl<Kind: RegionIdxMarker> std::fmt::Display for FunctionShape<Kind> {
  567     fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
  568         fn fmt_input<K: RegionIdxMarker>(
  569             f: &mut std::fmt::Formatter<'_>,
  570             input: &FunctionShapeInput<K>,
  571         ) -> std::fmt::Result {
  572             write!(f, "Arg({})↓{}", *input.base, input.region_idx.index())
  573         }
  574 
  575         fn fmt_output<K: RegionIdxMarker>(
  576             f: &mut std::fmt::Formatter<'_>,
  577             output: &FunctionShapeOutput<K>,
  578         ) -> std::fmt::Result {
  579             match output.base {
  580                 ArgIdxOrResult::Argument(arg) => {
  581                     write!(f, "Arg({})↓{}", *arg, output.region_idx.index())
  582                 }
  583                 ArgIdxOrResult::Result => {
  584                     write!(f, "Result↓{}", output.region_idx.index())
  585                 }
  586             }
  587         }
  588 
  589         write!(f, "Inputs: [")?;
  590         for (i, input) in self.inputs.iter().enumerate() {
  591             if i > 0 {
  592                 write!(f, ", ")?;
  593             }
  594             fmt_input(f, input)?;
  595         }
  596         write!(f, "]\nOutputs: [")?;
  597         for (i, output) in self.outputs.iter().enumerate() {
  598             if i > 0 {
  599                 write!(f, ", ")?;
  600             }
  601             fmt_output(f, output)?;
  602         }
  603         write!(f, "]\nEdges: [")?;
  604         for (i, edge) in self.edges.iter().enumerate() {
  605             if i > 0 {
  606                 write!(f, ", ")?;
  607             }
  608             fmt_input(f, &edge.input)?;
  609             write!(f, " -> ")?;
  610             fmt_output(f, &edge.output)?;
  611         }
  612         write!(f, "]")
  613     }
  614 }
  615 
  616 impl<Kind: RegionIdxMarker> FunctionShape<Kind> {
  617     #[allow(unused)]
  618     pub(crate) fn is_specialization_of(&self, other: &Self) -> bool {
  619         self.edges.is_subset(&other.edges)
  620     }
  621 
  622     pub(crate) fn new<
  623         'tcx,
  624         Ctxt,
  625         ShapeData: FunctionShapeDataSource<'tcx, Ctxt, Lifetime: LifetimeKind<'tcx, IdxMarker = Kind>>,
  626     >(
  627         shape_data: &ShapeData,
  628         ctxt: Ctxt,
  629     ) -> Self
  630     where
  631         Ctxt: HasTyCtxt<'tcx>
  632             + Copy
  633             + ExtractRegionsCtxt<'tcx, ty::Ty<'tcx>, ShapeData::Lifetime>
  634             + LifetimeDataCtxt<'tcx, ShapeData::Lifetime>,
  635     {
  636         let mut shape: BTreeSet<
  637             AbstractionBlockEdge<'static, FunctionShapeInput<Kind>, FunctionShapeOutput<Kind>>,
  638         > = BTreeSet::default();
  639         let arg_projections = shape_data.input_arg_projections(ctxt);
  640         let result_projections = shape_data.result_projections(ctxt);
  641         let to_input =
  642             |pd: &ProjectionData<'tcx, ArgIdx, ShapeData::Lifetime>| -> FunctionShapeInput<Kind> {
  643                 FunctionShapeInput::from_index(pd.base, pd.region_idx)
  644             };
  645         let to_output_from_arg =
  646             |pd: &ProjectionData<'tcx, ArgIdx, ShapeData::Lifetime>| -> FunctionShapeOutput<Kind> {
  647                 FunctionShapeOutput::from_index(ArgIdxOrResult::Argument(pd.base), pd.region_idx)
  648             };
  649         let to_output = |pd: &ProjectionData<'tcx, ArgIdxOrResult, ShapeData::Lifetime>| -> FunctionShapeOutput<Kind> {
  650             FunctionShapeOutput::from_index(pd.base, pd.region_idx)
  651         };
  652         for input in &arg_projections {
  653             for output in &arg_projections {
  654                 if ctxt.is_invariant_in_type(output.region, output.ty)
  655                     && shape_data.outlives(input.region, output.region, ctxt)
  656                 {
  657                     tracing::debug!("{input} outlives {output}");
  658                     shape.insert(AbstractionBlockEdge::new(
  659                         to_input(input),
  660                         to_output_from_arg(output),
  661                     ));
  662                 }
  663             }
  664             for rp in &result_projections {
  665                 if shape_data.outlives(input.region, rp.region, ctxt) {
  666                     tracing::debug!("{input} outlives {rp}");
  667                     shape.insert(AbstractionBlockEdge::new(to_input(input), to_output(rp)));
  668                 }
  669             }
  670         }
  671 
  672         let inputs = shape_data.inputs(ctxt);
  673         let outputs = shape_data.outputs(ctxt);
  674         FunctionShape {
  675             inputs,
  676             outputs,
  677             edges: shape,
  678         }
  679     }
  680 
  681     #[must_use]
  682     pub fn coupled_edges(&self) -> FunctionShapeCoupledEdges<Kind> {
  683         CoupledEdgesData::new(self.edges.iter().copied())
  684     }
  685 }
  686 
  687 pub type FunctionShapeCoupledEdges<Kind = Generalized> =
  688     CoupledEdgesData<FunctionShapeInput<Kind>, FunctionShapeOutput<Kind>>;
