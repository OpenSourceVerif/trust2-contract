   31 struct EdgesToRemove<'tcx> {
   32     deref_edges:
   33         HashMap<MaybeLabelledPlace<'tcx>, WithReason<HashSet<Conditioned<DerefEdge<'tcx>>>>>,
   34     other_edges: Vec<WithReason<BorrowPcgEdge<'tcx>>>,
   35 }
   36 
   37 #[allow(clippy::large_enum_variant)]
   38 enum RemovalAction<'tcx> {
   39     RemoveDerefEdgesTo(
   40         MaybeLabelledPlace<'tcx>,
   41         HashSet<Conditioned<DerefEdge<'tcx>>>,
   42         Reason,
   43     ),
   44     RemoveOtherEdge(BorrowPcgEdge<'tcx>, Reason),
   45 }
   46 
   47 impl<'tcx> EdgesToRemove<'tcx> {
   48     fn new() -> Self {
   49         Self {
   50             deref_edges: HashMap::default(),
   51             other_edges: Vec::new(),
   52         }
   53     }
   54 
   55     fn push(&mut self, edge: &BorrowPcgEdge<'tcx>, reason: Reason) {
   56         match edge.kind() {
   57             BorrowPcgEdgeKind::Deref(deref) => {
   58                 if let Some(deref_edges) = self.deref_edges.get_mut(&deref.deref_place) {
   59                     deref_edges
   60                         .value
   61                         .insert(Conditioned::new(*deref, edge.conditions().clone()));
   62                 } else {
   63                     self.deref_edges.insert(
   64                         deref.deref_place,
   65                         WithReason {
   66                             value: vec![Conditioned::new(*deref, edge.conditions().clone())]
   67                                 .into_iter()
   68                                 .collect(),
   69                             reason,
   70                         },
   71                     );
   72                 }
   73             }
   74             _ => self.other_edges.push(WithReason {
   75                 value: edge.clone(),
   76                 reason,
   77             }),
   78         }
   79     }
   80 
   81     fn is_empty(&self) -> bool {
   82         self.deref_edges.is_empty() && self.other_edges.is_empty()
   83     }
   84 
   85     fn removal_actions(self) -> Vec<RemovalAction<'tcx>> {
   86         let mut actions = Vec::new();
   87         for (place, deref_edges) in self.deref_edges {
   88             actions.push(RemovalAction::RemoveDerefEdgesTo(
   89                 place,
   90                 deref_edges.value,
   91                 deref_edges.reason,
   92             ));
   93         }
   94         for edge in self.other_edges {
   95             actions.push(RemovalAction::RemoveOtherEdge(edge.value, edge.reason));
   96         }
   97         actions
   98     }
   99 }

  104     /// Removes leaves that are old or dead (based on the borrow checker). This
  105     /// function should called prior to evaluating the effect of the statement
  106     /// at `location`.
  107     ///
  108     /// Note that the liveness calculation is performed based on what happened
  109     /// at the end of the *previous* statement.
  110     ///
  111     /// For example when evaluating:
  112     /// ```text
  113     /// bb0[1]: let x = &mut y;
  114     /// bb0[2]: *x = 2;
  115     /// bb0[3]: ... // x is dead
  116     /// ```
  117     /// we do not remove the `*x -> y` edge until `bb0[3]`.
  118     /// This ensures that the edge appears in the graph at the end of `bb0[2]`
  119     /// (rather than never at all).
  120     ///
  121     /// Additional caveat: we do not remove dead places that are function
  122     /// arguments. At least for now this interferes with the implementation in
  123     /// the Prusti purified encoding for accessing the final value of a
  124     /// reference-typed function argument in its postcondition.
  125     pub(crate) fn pack_old_and_dead_borrow_leaves(
  126         &mut self,
  127         for_place: Option<Place<'tcx>>,
  128     ) -> Result<(), PcgError> {
  129         let debug_iteration_limit = 10000;
  130         let mut iteration = 0;
  131         let ctxt = self.ctxt;
  132         let block = self.location().block;
  133         self.restore_capability_to_leaf_places(
  134             for_place,
  135             |place| !ctxt.body_analysis().is_read_only_in_loop(block, place),
  136             ctxt,
  137         )?;
  138         let mut removed_edges = Vec::new();
  139         loop {
  140             iteration += 1;
  141             let edges_to_remove = self.identify_leaf_edges_to_remove(for_place);
  142             if edges_to_remove.is_empty() {
  143                 break self.finish_weakening_places_with_dead_lifetime_projections(&removed_edges);
  144             }
  145             for action in edges_to_remove.removal_actions() {
  146                 match action {
  147                     RemovalAction::RemoveDerefEdgesTo(place, edges, reason) => {
  148                         self.remove_deref_edges_to(place, edges, &reason)?;
  149                     }
  150                     RemovalAction::RemoveOtherEdge(borrow_pcg_edge, reason) => {
  151                         self.remove_edge_and_perform_associated_state_updates(
  152                             &borrow_pcg_edge,
  153                             &reason,
  154                         )?;
  155                         removed_edges.push(borrow_pcg_edge);
  156                     }
  157                 }
  158             }
  159             if iteration % 10 == 0 {
  160                 tracing::debug!(
  161                     "Packing old and dead borrow leaves: iteration {}",
  162                     iteration
  163                 );
  164             }
  165             assert!(
  166                 iteration <= debug_iteration_limit,
  167                 "Packing old and dead borrow leaves took more than {debug_iteration_limit} iterations"
  168             );
  169         }
  170     }

  172     /// Cuts the delegations that keep a loan of `place` alive.
  173     ///
  174     /// A delegation stays alive for as long as its raw pointer is used, which
  175     /// also keeps alive the borrows leading to it. The borrow checker doesn't
  176     /// see this: for `let r = &mut x; let p = r as *mut i32; x = 1; *p = 2;` it
  177     /// accepts the access to `x`, since `r` is dead. Such an access invalidates
  178     /// the raw pointers derived from the loan, so the delegations are removed
  179     /// and the raw pointers lose their capability. A delegation of `place`
  180     /// itself is unaffected, since `place` is not lent out to the raw pointer.
  181     pub(crate) fn cut_delegations_through_loans_of(
  182         &mut self,
  183         place: Place<'tcx>,
  184         obtain_cap: CapabilityKind,
  185     ) -> Result<(), PcgError> {
  186         let has_capability = match self.pcg.capability_of(place, self.ctxt) {
  187             None => false,
  188             Some(CapabilityKind::Read) => obtain_cap.is_read(),
  189             Some(_) => true,
  190         };
  191         if has_capability {
  192             return Ok(());
  193         }
  194         let delegations = self.delegations_through_loans_of(place);
  195         if delegations.is_empty() {
  196             return Ok(());
  197         }
  198         let context = format!(
  199             "Delegation is cut because access to {} is needed",
  200             place.display_string(self.ctxt)
  201         );
  202         for edge in delegations {
  203             let BorrowPcgEdgeKind::Delegation(delegation) = edge.kind() else {
  204                 unreachable!()
  205             };
  206             let rawptr = delegation
  207                 .rawptr_place
  208                 .as_current_place()
  209                 .and_then(Place::parent_place);
  210             self.record_and_apply_action(
  211                 BorrowPcgAction::remove_edge(edge.clone(), context.clone()).into(),
  212             )?;
  213             if let Some(rawptr) = rawptr
  214                 && self.pcg.capability_of(rawptr, self.ctxt) == Some(CapabilityKind::Exclusive)
  215             {
  216                 self.record_and_apply_action(
  217                     BorrowPcgAction::weaken(
  218                         rawptr,
  219                         CapabilityKind::Exclusive,
  220                         Some(CapabilityKind::Write),
  221                         context.clone(),
  222                     )
  223                     .into(),
  224                 )?;
  225             }
  226         }
  227         self.pack_old_and_dead_borrow_leaves(None)
  228     }

  230     /// The delegation edges that (transitively) block a loan of `place`, or of
  231     /// a place overlapping with it.
  232     fn delegations_through_loans_of(&self, place: Place<'tcx>) -> Vec<BorrowPcgEdge<'tcx>> {
  233         let ctxt = self.ctxt.bc_ctxt();
  234         let graph = self.pcg.borrows_graph();
  235         let mut worklist = graph
  236             .edges()
  237             .filter(|edge| {
  238                 let BorrowPcgEdgeKind::Borrow(borrow) = edge.kind else {
  239                     return false;
  240                 };
  241                 borrow
  242                     .blocked_place
  243                     .as_current_place()
  244                     .is_some_and(|blocked| {
  245                         blocked.is_prefix_of(place) || place.is_prefix_of(blocked)
  246                     })
  247             })
  248             .map(BorrowPcgEdgeLike::to_owned_edge)
  249             .collect::<Vec<_>>();
  250         let mut seen = HashSet::default();
  251         let mut delegations = Vec::new();
  252         while let Some(edge) = worklist.pop() {
  253             if !seen.insert(edge.clone()) {
  254                 continue;
  255             }
  256             if matches!(edge.kind(), BorrowPcgEdgeKind::Delegation(_)) {
  257                 delegations.push(edge);
  258                 continue;
  259             }
  260             for node in edge.blocked_by_nodes(ctxt) {
  261                 worklist.extend(
  262                     graph
  263                         .edges_blocking(node.into(), ctxt)
  264                         .map(BorrowPcgEdgeLike::to_owned_edge),
  265                 );
  266             }
  267         }
  268         delegations
  269     }

  271     /// Packs the expansions at or below `place` that can be packed without
  272     /// losing information.
  273     ///
  274     /// Under [`PackStrategy::Lazy`] this is the only point at which such
  275     /// expansions are packed: `place` is about to be required at a capability
  276     /// it cannot hold while it is expanded.
  277     pub(crate) fn pack_expansions_for_obtain_of(
  278         &mut self,
  279         place: Place<'tcx>,
  280     ) -> Result<(), PcgError> {
  281         loop {
  282             let to_pack = self.packable_expansions_at_or_below(place);
  283             if to_pack.is_empty() {
  284                 return Ok(());
  285             }
  286             for edge in to_pack {
  287                 self.remove_edge_and_perform_associated_state_updates(
  288                     &edge,
  289                     &format!(
  290                         "Pack expansions to obtain {}",
  291                         place.display_string(self.ctxt)
  292                     ),
  293                 )?;
  294             }
  295         }
  296     }

  298     fn packable_expansions_at_or_below(&self, place: Place<'tcx>) -> Vec<BorrowPcgEdge<'tcx>> {
  299         let fg = self.pcg.borrow.graph.frozen_graph();
  300         fg.leaf_edges(self.ctxt)
  301             .into_iter()
  302             .map(BorrowPcgEdgeLike::to_owned_edge)
  303             .filter(|edge| {
  304                 let BorrowPcgEdgeKind::BorrowPcgExpansion(expansion) = edge.kind() else {
  305                     return false;
  306                 };
  307                 expansion
  308                     .base()
  309                     .as_current_place()
  310                     .is_some_and(|base| place.is_prefix_of(base))
  311                     && expansion.is_packable(self.pcg.place_capabilities, self.ctxt)
  312             })
  313             .collect()
  314     }

  316     /// Identifies the set of edges that should be removed from the graph
  317     ///
  318     /// If `ancestor_place` is defined, we only consider a subset of the edges.
  319     ///
  320     /// The subset of edges is defined as follows:
  321     ///
  322     /// A node *can be killed* if it has a related current place that is a
  323     /// postfix of `ancestor_place`.
  324     ///
  325     /// Then we consider edges where either:
  326     ///
  327     /// 1. The edge is an expansion edge, and the base can be killed, or
  328     /// 2. The edge is some other kind of edge, and all of the blocking nodes can be killed
  329     fn identify_leaf_edges_to_remove<'slf>(
  330         &'slf self,
  331         ancestor_place: Option<Place<'tcx>>,
  332     ) -> EdgesToRemove<'tcx> {
  333         enum ShouldKillNode {
  334             Yes { reason: Cow<'static, str> },
  335             No,
  336         }
  337 
  338         let ctxt = self.ctxt;
  339         let location = self.location();
  340 
  341         let ancestor_predicate_allows_killing = |p: LocalNode<'tcx>| {
  342             let place = match p {
  343                 PcgNode::Place(p) => p,
  344                 PcgNode::LifetimeProjection(rp) => rp.base(),
  345             };
  346             if let Some(ancestor_place) = ancestor_place
  347                 && (!place.is_current() || !ancestor_place.place().is_prefix_of(place.place()))
  348             {
  349                 return false;
  350             }
  351             true
  352         };
  353 
  354         let should_kill_node = |p: LocalNode<'tcx>, fg: &FrozenGraphRef<'slf, 'tcx>| {
  355             let place = match p {
  356                 PcgNode::Place(p) => p,
  357                 PcgNode::LifetimeProjection(rp) => rp.base(),
  358             };
  359             if !ancestor_predicate_allows_killing(p) {
  360                 return ShouldKillNode::No;
  361             }
  362 
  363             let has_live_rawptr_delegation = |node: PcgNode<'tcx>| {
  364                 self.pcg
  365                     .borrows_graph()
  366                     .edges_blocking(node, ctxt)
  367                     .into_iter()
  368                     .any(|e| {
  369                         matches!(
  370                             e.kind,
  371                             BorrowPcgEdgeKind::Delegation(delegation)
  372                                 if !ctxt.bc().is_dead(
  373                                     delegation.rawptr_place.place().into(),
  374                                     location,
  375                                 )
  376                         )
  377                     })
  378             };
  379             if has_live_rawptr_delegation(PcgNode::from(p))
  380                 || (matches!(p, PcgNode::LifetimeProjection(_))
  381                     && has_live_rawptr_delegation(PcgNode::from(place)))
  382             {
  383                 return ShouldKillNode::No;
  384             }
  385 
  386             if place.is_old() {
  387                 return ShouldKillNode::Yes {
  388                     reason: "Place is old".into(),
  389                 };
  390             }
  391 
  392             if p.is_place()
  393                 && !place.place().projection.is_empty()
  394                 && !fg.has_edge_blocking(place.into(), ctxt)
  395                 && !place.place().contains_unsafe_deref(ctxt)
  396             {
  397                 return ShouldKillNode::Yes {
  398                     reason:
  399                         "Node is a place with a non-empty projection and is not blocked by an edge"
  400                             .into(),
  401                 };
  402             }
  403             if ctxt.bc().is_dead(p.into(), location) {
  404                 return ShouldKillNode::Yes {
  405                     reason: "Borrow-checker reports node as dead".into(),
  406                 };
  407             }
  408 
  409             ShouldKillNode::No
  410         };
  411 
  412         enum ShouldPackEdge {
  413             Yes { reason: Cow<'static, str> },
  414             No,
  415         }
  416 
  417         let mut edges_to_remove = EdgesToRemove::new();
  418         let fg = self.pcg.borrow.graph.frozen_graph();
  419         let location = self.location();
  420         let should_pack_edge = |edge: &BorrowPcgEdgeKind<'tcx>| {
  421             if let BorrowPcgEdgeKind::BorrowPcgExpansion(expansion) = edge {
  422                 if !ancestor_predicate_allows_killing(expansion.base()) {
  423                     return ShouldPackEdge::No;
  424                 }
  425                 if expansion.expansion().iter().all(|node| {
  426                     node.is_old() || self.ctxt.bc().is_dead(node.place().into(), location)
  427                 }) {
  428                     ShouldPackEdge::Yes {
  429                         reason: "Expansion is old or dead".into(),
  430                     }
  431                 } else if self.ctxt.settings().pack_strategy == PackStrategy::Eager
  432                     && expansion.is_packable(self.pcg.place_capabilities, self.ctxt)
  433                 {
  434                     ShouldPackEdge::Yes {
  435                         reason: format!(
  436                             "Expansion {} is packable",
  437                             expansion.display_string(self.ctxt)
  438                         )
  439                         .into(),
  440                     }
  441                 } else {
  442                     ShouldPackEdge::No
  443                 }
  444             } else {
  445                 let mut why_killed_reasons = Vec::new();
  446                 for node in edge.blocked_by_nodes(self.ctxt) {
  447                     if let ShouldKillNode::Yes { reason } = should_kill_node(node, &fg) {
  448                         why_killed_reasons.push(format!(
  449                             "{}: {}",
  450                             node.display_string(self.ctxt),
  451                             reason
  452                         ));
  453                     } else {
  454                         tracing::debug!(
  455                             "Node {} will not be killed",
  456                             node.display_string(self.ctxt)
  457                         );
  458                         return ShouldPackEdge::No;
  459                     }
  460                 }
  461                 ShouldPackEdge::Yes {
  462                     reason: format!(
  463                         "Edge is blocked by nodes that should be killed: {}",
  464                         why_killed_reasons.join(", ")
  465                     )
  466                     .into(),
  467                 }
  468             }
  469         };
  470         let leaf_edges = fg.leaf_edges(self.ctxt);
  471         for edge in leaf_edges.into_iter().map(BorrowPcgEdgeLike::to_owned_edge) {
  472             tracing::debug!(
  473                 "Checking leaf edge: {}",
  474                 edge.value.display_string(self.ctxt)
  475             );
  476             if let BorrowPcgEdgeKind::Delegation(delegation) = edge.value {
  477                 if let ShouldKillNode::Yes { reason } =
  478                     should_kill_node(delegation.rawptr_place.into(), &fg)
  479                 {
  480                     edges_to_remove.push(&edge, reason);
  481                     let edges = self
  482                         .pcg
  483                         .borrows_graph()
  484                         .edges_blocking(delegation.rawptr_place.into(), self.ctxt.bc_ctxt());
  485                     for edge_to_remove in edges {
  486                         edges_to_remove.push(
  487                             &edge_to_remove.to_owned_edge(),
  488                             "Delegation edge is removed".into(),
  489                         );
  490                     }
  491                 }
  492             } else if let ShouldPackEdge::Yes { reason } = should_pack_edge(edge.kind()) {
  493                 edges_to_remove.push(&edge, reason);
  494             }
  495         }
  496         edges_to_remove
  497     }
