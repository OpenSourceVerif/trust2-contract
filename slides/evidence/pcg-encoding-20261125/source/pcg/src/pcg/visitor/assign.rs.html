   32 impl<'a, 'tcx: 'a, Ctxt: DataflowCtxt<'a, 'tcx>> PcgVisitor<'_, 'a, 'tcx, Ctxt> {
   33     /// The label that should be used when referencing (after `PostOperands`), the
   34     /// value at the place before the move.
   35     pub(crate) fn pre_operand_move_label(&self) -> SnapshotLocation {
   36         SnapshotLocation::Before(AnalysisLocation::new(
   37             self.location(),
   38             EvalStmtPhase::PostOperands,
   39         ))
   40     }

   42     /// The maybe-labelled place to use to reference the value of an operand after
   43     /// the `PostOperands` phase. If the operand was copied, the place is returned
   44     /// as-is. If the operand was moved, the place is returned with the label of
   45     /// the value before the move.
   46     pub(crate) fn maybe_labelled_operand(
   47         &self,
   48         operand: &Operand<'tcx>,
   49     ) -> PlaceOrConst<'tcx, MaybeLabelledPlace<'tcx>> {
   50         match operand {
   51             Operand::Copy(place) => PlaceOrConst::Place((*place).into()),
   52             Operand::Move(place) => PlaceOrConst::Place(MaybeLabelledPlace::new(
   53                 (*place).into(),
   54                 Some(self.pre_operand_move_label()),
   55             )),
   56             Operand::Constant(const_) => PlaceOrConst::Const(const_.const_),
   57             #[allow(unreachable_patterns)]
   58             _ => todo!(),
   59         }
   60     }

   62     pub(crate) fn assign_post_main(
   63         &mut self,
   64         target: Place<'tcx>,
   65         rvalue: &Rvalue<'tcx>,
   66     ) -> Result<(), PcgError> {
   67         let ctxt = self.ctxt;
   68 
   69         // If `target` is a reference, then the dereferenced place technically
   70         // still retains its capabilities. However, because we currently only
   71         // keep capabilities for non-labelled places, we remove all the capabilities
   72         // to everything postfix of `target`.
   73         //
   74         // We should change this logic once we start keeping capabilities for
   75         // labelled places.
   76         if target.is_ref(ctxt) {
   77             self.pcg
   78                 .place_capabilities
   79                 .remove_all_postfixes(target, ctxt);
   80         }
   81 
   82         self.pcg
   83             .place_capabilities
   84             .insert(target, CapabilityKind::Exclusive, self.ctxt);
   85         match rvalue {
   86             Rvalue::Aggregate(
   87                 box (mir::AggregateKind::Adt(..)
   88                 | mir::AggregateKind::Tuple
   89                 | mir::AggregateKind::Array(..)
   90                 | mir::AggregateKind::Closure(..)),
   91                 fields,
   92             ) => {
   93                 let target: Place<'tcx> = (*target).into();
   94                 for (field_idx, field) in fields.iter().enumerate() {
   95                     let operand_place: Place<'tcx> = if let Some(place) = field.place() {
   96                         place.into()
   97                     } else {
   98                         continue;
   99                     };
  100                     for (source_rp_idx, source_proj) in operand_place
  101                         .lifetime_projections(self.ctxt)
  102                         .iter()
  103                         .enumerate()
  104                     {
  105                         let maybe_labelled = self.maybe_labelled_operand(field);
  106                         let source_proj = source_proj.with_base(maybe_labelled);
  107                         self.connect_outliving_projections(source_proj, target, |_| {
  108                             BorrowFlowEdgeKind::Aggregate {
  109                                 field_idx,
  110                                 target_rp_index: source_rp_idx,
  111                             }
  112                         })?;
  113                     }
  114                 }
  115             }
  116             Rvalue::Use(operand, ..) => {
  117                 let p = operand.place();
  118                 if let Some(p) = p {
  119                     let p: Place = p.into();
  120                     if p.ty(self.ctxt).ty.is_raw_ptr() {
  121                         let p = p.with_inherent_region(self.ctxt).project_deref(self.ctxt);
  122                         let node = PcgNode::from(p);
  123                         let edges = self
  124                             .pcg
  125                             .borrows_graph()
  126                             .edges_blocked_by(node, self.ctxt.bc_ctxt());
  127                         let alias_edges = edges
  128                             .into_iter()
  129                             .filter_map(|e| match e.kind {
  130                                 BorrowPcgEdgeKind::Delegation(de) => Some(de),
  131                                 _ => None,
  132                             })
  133                             .collect::<Vec<_>>();
  134                         if !alias_edges.is_empty() {
  135                             self.record_and_apply_action(
  136                                 BorrowPcgAction::add_edge(
  137                                     BorrowPcgEdge::new(
  138                                         DelegationEdge {
  139                                             rawptr_place: target.project_deref(ctxt).into(),
  140                                             aliased_place: alias_edges[0].aliased_place,
  141                                         }
  142                                         .into(),
  143                                         self.pcg.borrow.validity_conditions.clone(),
  144                                     ),
  145                                     "assign_post_main",
  146                                 )
  147                                 .into(),
  148                             )?;
  149                             return Ok(());
  150                         }
  151                     }
  152                 }
  153                 self.assignment_projections(operand, target, None)?;
  154             }
  155             Rvalue::Cast(kind, operand, ty) => {
  156                 if let CastKind::PtrToPtr = kind {
  157                     let p = operand.place().unwrap();
  158                     let p: Place = p.into();
  159                     let p = p.with_inherent_region(self.ctxt).project_deref(self.ctxt);
  160                     let edges = self.pcg.borrows_graph().edges_blocked_by(p.into(), ctxt);
  161                     let alies_edges = edges
  162                         .into_iter()
  163                         .filter_map(|e| match e.kind {
  164                             BorrowPcgEdgeKind::Delegation(de) => Some(de),
  165                             _ => None,
  166                         })
  167                         .collect::<Vec<_>>();
  168                     if !alies_edges.is_empty() {
  169                         assert_eq!(alies_edges.len(), 1);
  170                         self.record_and_apply_action(
  171                             BorrowPcgAction::add_edge(
  172                                 BorrowPcgEdge::new(
  173                                     DelegationEdge {
  174                                         rawptr_place: target.project_deref(ctxt).into(),
  175                                         aliased_place: alies_edges[0].aliased_place,
  176                                     }
  177                                     .into(),
  178                                     self.pcg.borrow.validity_conditions.clone(),
  179                                 ),
  180                                 "assign_post_main",
  181                             )
  182                             .into(),
  183                         )?;
  184                     }
  185                 } else {
  186                     self.assignment_projections(operand, target, Some(CastData::new(*kind, *ty)))?;
  187                 }
  188             }
  189             Rvalue::Ref(borrow_region, kind, blocked_place) => {
  190                 let blocked_place: Place<'tcx> = (*blocked_place).into();
  191                 let blocked_place = blocked_place.with_inherent_region(self.ctxt);
  192                 if !target.ty(self.ctxt).ty.is_ref() {
  193                     return Err(PcgError::unsupported(
  194                         PcgUnsupportedError::AssignBorrowToNonReferenceType,
  195                     ));
  196                 }
  197                 if matches!(kind, mir::BorrowKind::Fake(_)) {
  198                     return Ok(());
  199                 }
  200                 self.pcg.borrow.add_borrow(
  201                     blocked_place,
  202                     target,
  203                     *kind,
  204                     self.location(),
  205                     *borrow_region,
  206                     &mut self.pcg.place_capabilities,
  207                     self.ctxt,
  208                 );
  209                 self.label_lifetime_projections_for_borrow(blocked_place, target, *kind)?;
  210             }
  211             Rvalue::RawPtr(kind, p) if !kind.is_fake() => {
  212                 let p: Place<'tcx> = (*p).into();
  213                 let p = p.with_inherent_region(self.ctxt);
  214                 self.record_and_apply_action(
  215                     BorrowPcgAction::add_edge(
  216                         BorrowPcgEdge::new(
  217                             DelegationEdge {
  218                                 rawptr_place: target.project_deref(ctxt).into(),
  219                                 aliased_place: p.into(),
  220                             }
  221                             .into(),
  222                             self.pcg.borrow.validity_conditions.clone(),
  223                         ),
  224                         "assign_post_main",
  225                     )
  226                     .into(),
  227                 )?;
  228             }
  229             _ => {}
  230         }
  231         Ok(())
  232     }

  234     fn assignment_projections(
  235         &mut self,
  236         operand: &Operand<'tcx>,
  237         target_place: Place<'tcx>,
  238         cast_data: Option<CastData<'tcx>>,
  239     ) -> Result<(), PcgError> {
  240         let (source_projections, operand_type) = match operand {
  241             Operand::Move(place) | Operand::Copy(place) => {
  242                 let operand_type = if matches!(operand, Operand::Move(_)) {
  243                     OperandType::Move
  244                 } else {
  245                     OperandType::Copy
  246                 };
  247                 let place_label = matches!(operand_type, OperandType::Move)
  248                     .then(|| self.pre_operand_move_label());
  249                 let place: Place<'tcx> = (*place).into();
  250                 let place = place.with_inherent_region(self.ctxt);
  251                 (
  252                     PlaceOrConst::Place(MaybeLabelledPlace::new(place, place_label))
  253                         .generalized_lifetime_projections(self.ctxt),
  254                     operand_type,
  255                 )
  256             }
  257             Operand::Constant(const_) => (
  258                 PlaceOrConst::Const(const_.const_).generalized_lifetime_projections(self.ctxt),
  259                 OperandType::Const,
  260             ),
  261             #[allow(unreachable_patterns)]
  262             _ => return Ok(()),
  263         };
  264         for source_proj in source_projections {
  265             self.connect_outliving_projections(source_proj, target_place, |_| {
  266                 BorrowFlowEdgeKind::Assignment(AssignmentData::new(operand_type, cast_data))
  267             })?;
  268         }
  269         Ok(())
  270     }

  272     fn label_lifetime_projections_for_borrow(
  273         &mut self,
  274         blocked_place: Place<'tcx>,
  275         target: Place<'tcx>,
  276         kind: mir::BorrowKind,
  277     ) -> Result<(), PcgError> {
  278         let ctxt = self.ctxt;
  279         for source_proj in blocked_place.lifetime_projections(self.ctxt) {
  280             let mut obtainer = self.place_obtainer();
  281             let source_proj = if kind.mutability().is_mut() {
  282                 let label = obtainer.prev_snapshot_location();
  283                 obtainer.apply_action(
  284                     BorrowPcgAction::label_lifetime_projection(
  285                         LabelNodePredicate::postfix_lifetime_projection(source_proj.into()),
  286                         Some(label.into()),
  287                         "Label region projections of newly borrowed place",
  288                     )
  289                     .into(),
  290                 )?;
  291                 source_proj.with_label(Some(label.into()), self.ctxt)
  292             } else {
  293                 source_proj.with_label(
  294                     obtainer
  295                         .label_for_shared_expansion_of_rp(source_proj, obtainer.ctxt)
  296                         .map(Into::into),
  297                     self.ctxt,
  298                 )
  299             };
  300             let source_region: GeneralizedLifetime<'tcx> = source_proj.region(self.ctxt.ctxt());
  301             let mut nested_ref_mut_targets = vec![];
  302             for target_proj in target.lifetime_projections(self.ctxt) {
  303                 let target_region: GeneralizedLifetime<'tcx> = target_proj.region(self.ctxt.ctxt());
  304                 if self.ctxt.bc().outlives_generalized(
  305                     source_region,
  306                     target_region,
  307                     self.location(),
  308                 ) {
  309                     let regions_equal = self.ctxt.bc().same_region_generalized(
  310                         source_region,
  311                         target_region,
  312                         self.location(),
  313                     );
  314                     self.record_and_apply_action(
  315                         BorrowPcgAction::add_edge(
  316                             BorrowPcgEdge::new(
  317                                 BorrowFlowEdge::new(
  318                                     source_proj.into(),
  319                                     target_proj.into(),
  320                                     BorrowFlowEdgeKind::BorrowOutlives { regions_equal },
  321                                 )
  322                                 .into(),
  323                                 self.pcg.borrow.validity_conditions.clone(),
  324                             ),
  325                             "assign_post_main",
  326                         )
  327                         .into(),
  328                     )?;
  329                     if regions_equal && kind.mutability().is_mut() {
  330                         nested_ref_mut_targets.push(target_proj.into());
  331                     }
  332                 }
  333             }
  334             self.place_obtainer().add_and_update_placeholder_edges(
  335                 source_proj.into(),
  336                 &nested_ref_mut_targets,
  337                 "assign",
  338                 ctxt,
  339             )?;
  340         }
  341         Ok(())
  342     }
