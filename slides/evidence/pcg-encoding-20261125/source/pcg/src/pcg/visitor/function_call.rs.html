   34 impl<'a, 'tcx: 'a, Ctxt: DataflowCtxt<'a, 'tcx>> PcgVisitor<'_, 'a, 'tcx, Ctxt> {
   35     pub(crate) fn settings(&self) -> &'a PcgSettings {
   36         self.ctxt.settings()
   37     }
   38 
   39     fn node_for_input(
   40         &self,
   41         call: &FunctionCall<'_, 'tcx>,
   42         input: LifetimeProjection<'tcx, ArgIdx>,
   43     ) -> FunctionCallAbstractionInput<'tcx> {
   44         let operand = call.inputs[*input.base];
   45         let operand = self.maybe_labelled_operand(operand);
   46         FunctionCallAbstractionInput(
   47             LifetimeProjection::from_index(operand, input.region_idx)
   48                 .with_label(Some(self.prev_snapshot_location().into()), self.ctxt),
   49         )
   50     }
   51 
   52     fn node_for_output(
   53         &self,
   54         call: &FunctionCall<'_, 'tcx>,
   55         output: LifetimeProjection<'tcx, ArgIdxOrResult>,
   56     ) -> Option<FunctionCallAbstractionOutput<'tcx>> {
   57         match output.base {
   58             ArgIdxOrResult::Argument(arg_idx) => {
   59                 let operand = call.inputs[*arg_idx];
   60                 // Constants are never targets of borrow flow (they have no place)
   61                 if let PlaceOrConst::Place(place) = self.maybe_labelled_operand(operand) {
   62                     Some(
   63                         LifetimeProjection::from_index(place, output.region_idx)
   64                             .with_label(
   65                                 Some(SnapshotLocation::After(self.location().block).into()),
   66                                 self.ctxt,
   67                             )
   68                             .into(),
   69                     )
   70                 } else {
   71                     None
   72                 }
   73             }
   74             ArgIdxOrResult::Result => {
   75                 Some(LifetimeProjection::from_index(call.output, output.region_idx).into())
   76             }
   77         }
   78     }

   80     fn create_edges_for_shape(
   81         &mut self,
   82         shape: &FunctionShape,
   83         call: &FunctionCall<'_, 'tcx>,
   84     ) -> Result<(), PcgError> {
   85         let metadata = FunctionCallAbstractionEdgeMetadata {
   86             location: call.location,
   87             defined_fn_call: call.defined,
   88         };
   89         let abstraction_edges: HashSet<AbstractionBlockEdge<'_, _, _>> = shape
   90             .edges()
   91             .filter_map(|AbstractionBlockEdge { input, output, .. }| {
   92                 Some(AbstractionBlockEdge::new_checked(
   93                     self.node_for_input(call, input),
   94                     self.node_for_output(call, output)?,
   95                     self.ctxt,
   96                 ))
   97             })
   98             .collect();
   99         if self.settings().coupling {
  100             let coupled_edges = CoupledEdgesData::new(abstraction_edges.iter().copied());
  101             if !coupled_edges.is_empty() {
  102                 tracing::debug!("Coupled edges: {coupled_edges:?}");
  103             }
  104             for edge in coupled_edges {
  105                 let pcg_coupled_edge = PcgCoupledEdgeKind::function_call(
  106                     FunctionCallCoupledEdgeKind::new(metadata, edge),
  107                 );
  108                 self.record_and_apply_action(
  109                     BorrowPcgAction::add_edge(
  110                         BorrowPcgEdge::new(
  111                             pcg_coupled_edge.into(),
  112                             self.pcg.borrow.validity_conditions.clone(),
  113                         ),
  114                         "Function call",
  115                     )
  116                     .into(),
  117                 )?;
  118             }
  119         } else {
  120             for edge in abstraction_edges {
  121                 self.record_and_apply_action(
  122                     BorrowPcgAction::add_edge(
  123                         BorrowPcgEdge::new(
  124                             AbstractionEdge::FunctionCall(FunctionCallAbstraction::new(
  125                                 metadata, edge,
  126                             ))
  127                             .into(),
  128                             self.pcg.borrow.validity_conditions.clone(),
  129                         ),
  130                         "Function call",
  131                     )
  132                     .into(),
  133                 )?;
  134             }
  135         }
  136         Ok(())
  137     }
  138     #[tracing::instrument(skip(self, func, args, destination))]
  139     pub(super) fn make_function_call_abstraction(
  140         &mut self,
  141         func: &Operand<'tcx>,
  142         call_span: Span,
  143         args: &[&Operand<'tcx>],
  144         destination: utils::Place<'tcx>,
  145         location: Location,
  146     ) -> Result<(), PcgError> {
  147         let call = FunctionCall::new(
  148             location,
  149             args,
  150             destination,
  151             match func.ty(self.ctxt.body(), self.ctxt.tcx()).kind() {
  152                 ty::TyKind::FnDef(def_id, substs) => Some(DefinedFnCall::new(
  153                     FunctionData::new(*def_id),
  154                     fn_def_args(substs),
  155                     self.ctxt.ctxt().def_id(),
  156                     call_span,
  157                 )),
  158                 _ => None,
  159             },
  160         );
  161 
  162         let ctxt = self.ctxt;
  163 
  164         // The versions of the region projections for the function inputs just
  165         // before they were moved out, labelled with their last modification
  166         // time
  167         let arg_region_projections = args
  168             .iter()
  169             .map(|arg| self.maybe_labelled_operand(arg))
  170             .flat_map(|input_place| input_place.generalized_lifetime_projections(self.ctxt))
  171             .collect::<Vec<_>>();
  172 
  173         let pre_rps = arg_region_projections
  174             .iter()
  175             .map(|rp| {
  176                 rp.with_label(
  177                     Some(self.place_obtainer().prev_snapshot_location().into()),
  178                     self.ctxt,
  179                 )
  180             })
  181             .collect::<Vec<_>>();
  182 
  183         let post_rps = arg_region_projections
  184             .iter()
  185             .map(|rp| {
  186                 rp.with_label(
  187                     Some(SnapshotLocation::After(self.location().block).into()),
  188                     self.ctxt,
  189                 )
  190             })
  191             .collect::<Vec<_>>();
  192 
  193         for (rp, post_rp) in arg_region_projections.iter().zip(post_rps.iter()) {
  194             if let (Some(rp), Some(post_rp)) = (
  195                 rp.try_to_local_lifetime_projection(),
  196                 post_rp.try_to_local_lifetime_projection(),
  197             ) {
  198                 self.place_obtainer()
  199                     .redirect_source_of_future_edges(rp, post_rp, ctxt)?;
  200             }
  201         }
  202 
  203         for (rp, pre_rp) in arg_region_projections.iter().zip(pre_rps.iter()) {
  204             if let Some(rp) = rp.try_to_local_lifetime_projection() {
  205                 self.record_and_apply_action(
  206                     BorrowPcgAction::label_lifetime_projection(
  207                         LabelNodePredicate::equals_lifetime_projection(rp),
  208                         pre_rp.label(),
  209                         format!(
  210                             "Function call:Label Pre version of {}",
  211                             rp.display_string(self.ctxt),
  212                         ),
  213                     )
  214                     .into(),
  215                 )?;
  216             }
  217         }
  218         let shape = call.shape(self.ctxt);
  219         self.create_edges_for_shape(&shape, &call)?;
  220 
  221         Ok(())
  222     }
