  214 pub trait PcgNodeLike<'tcx, Ctxt, P>:
  215     Clone + Copy + std::fmt::Debug + Eq + PartialEq + std::hash::Hash
  216 {
  217     fn to_pcg_node(self, ctxt: Ctxt) -> PcgNodeWithPlace<'tcx, P>;
  218 
  219     fn try_to_local_node(self, ctxt: Ctxt) -> Option<LocalNode<'tcx, P>>
  220     where
  221         P: Copy,
  222     {
  223         match self.to_pcg_node(ctxt) {
  224             PcgNode::Place(p) => Some(p.into()),
  225             PcgNode::LifetimeProjection(rp) => rp
  226                 .base
  227                 .as_local_place()
  228                 .map(|local_place| LocalNode::LifetimeProjection(rp.with_base(local_place))),
  229         }
  230     }
  231 }

  248 pub(crate) trait LabelPlaceConditionally<'tcx, Ctxt, P: PcgPlace<'tcx, Ctxt>>:
  249     PcgNodeLike<'tcx, Ctxt, P> + LabelPlace<'tcx, Ctxt, P>
  250 {
  251     fn label_place_conditionally(
  252         &mut self,
  253         replacements: &mut HashSet<NodeReplacement<'tcx, P>>,
  254         predicate: &LabelNodePredicate<'tcx, P>,
  255         labeller: &impl PlaceLabeller<'tcx, Ctxt, P>,
  256         node_context: LabelNodeContext,
  257         ctxt: Ctxt,
  258     ) {
  259         let orig = self.to_pcg_node(ctxt);
  260         if predicate.applies_to(orig, node_context) {
  261             let changed = self.label_place(labeller, ctxt);
  262             if changed {
  263                 replacements.insert(NodeReplacement::new(orig, self.to_pcg_node(ctxt)));
  264             }
  265         }
  266     }
  267 }
