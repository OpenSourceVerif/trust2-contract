    9 /// Macro for generating patterns that match a capability and all "greater" capabilities
   10 /// according to the partial ordering.
   11 ///
   12 /// Usage:
   13 /// ```ignore
   14 /// match cap {
   15 ///     capability_gte!(Read) => { /* matches Read or Exclusive */ }
   16 ///     capability_gte!(Write) => { /* matches Write, ShallowExclusive, or Exclusive */ }
   17 ///     capability_gte!(ShallowExclusive) => { /* matches ShallowExclusive or Exclusive */ }
   18 ///     capability_gte!(Exclusive) => { /* matches only Exclusive */ }
   19 /// }
   20 /// ```
   21 ///
   22 /// Also supports single-letter abbreviations:
   23 /// ```ignore
   24 /// match cap {
   25 ///     capability_gte!(R) => { /* matches Read or Exclusive */ }
   26 ///     capability_gte!(W) => { /* matches Write, ShallowExclusive, or Exclusive */ }
   27 ///     capability_gte!(e) => { /* matches ShallowExclusive or Exclusive */ }
   28 ///     capability_gte!(E) => { /* matches only Exclusive */ }
   29 /// }
   30 /// ```
   31 #[macro_export]
   32 macro_rules! capability_gte {
   33     // Full names
   34     (Read) => {
   35         CapabilityKind::Read | CapabilityKind::Exclusive
   36     };
   37     (Write) => {
   38         CapabilityKind::Write | CapabilityKind::ShallowExclusive | CapabilityKind::Exclusive
   39     };
   40     (ShallowExclusive) => {
   41         CapabilityKind::ShallowExclusive | CapabilityKind::Exclusive
   42     };
   43     (Exclusive) => {
   44         CapabilityKind::Exclusive
   45     };
   46 
   47     // Single-letter abbreviations
   48     (R) => {
   49         CapabilityKind::Read | CapabilityKind::Exclusive
   50     };
   51     (W) => {
   52         CapabilityKind::Write | CapabilityKind::ShallowExclusive | CapabilityKind::Exclusive
   53     };
   54     (e) => {
   55         CapabilityKind::ShallowExclusive | CapabilityKind::Exclusive
   56     };
   57     (E) => {
   58         CapabilityKind::Exclusive
   59     };
   60 }
   61 
   62 #[derive(Copy, Clone, PartialEq, Eq, Hash, Serialize)]
   63 #[cfg_attr(feature = "type-export", derive(ts_rs::TS))]
   64 #[cfg_attr(feature = "type-export", ts(export))]
   65 pub enum CapabilityKind {
   66     /// For borrowed places only: permits reads from the location, but not writes or
   67     /// drops.
   68     Read,
   69 
   70     /// For owned places, this capability is used when the place is moved out
   71     /// of. This capability is used for both owned and borrowed places just before
   72     /// they are overwritten.
   73     Write,
   74 
   75     /// Writes and reads are permitted to this place, and the place is not
   76     /// borrowed.
   77     Exclusive,
   78 
   79     /// [`CapabilityKind::Exclusive`] for everything not through a dereference,
   80     /// [`CapabilityKind::Write`] for everything through a dereference.
   81     ShallowExclusive,
   82 }
   83 impl Debug for CapabilityKind {
   84     fn fmt(&self, f: &mut Formatter<'_>) -> Result {
   85         match self {
   86             CapabilityKind::Read => write!(f, "R"),
   87             CapabilityKind::Write => write!(f, "W"),
   88             CapabilityKind::Exclusive => write!(f, "E"),
   89             CapabilityKind::ShallowExclusive => write!(f, "e"),
   90         }
   91     }
   92 }
   93 
   94 impl PartialOrd for CapabilityKind {
   95     fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
   96         if *self == *other {
   97             return Some(Ordering::Equal);
   98         }
   99         match (*self, *other) {
  100             // Greater relationships
  101             (
  102                 CapabilityKind::Exclusive,
  103                 CapabilityKind::ShallowExclusive | CapabilityKind::Write | CapabilityKind::Read,
  104             )
  105             | (CapabilityKind::ShallowExclusive, CapabilityKind::Write) => Some(Ordering::Greater),
  106 
  107             // Less relationships (inverses of the above)
  108             (
  109                 CapabilityKind::ShallowExclusive | CapabilityKind::Write | CapabilityKind::Read,
  110                 CapabilityKind::Exclusive,
  111             )
  112             | (CapabilityKind::Write, CapabilityKind::ShallowExclusive) => Some(Ordering::Less),
  113 
  114             // All other pairs are incomparable
  115             _ => None,
  116         }
  117     }
  118 }

  120 impl<Ctxt> DisplayWithCtxt<Ctxt> for CapabilityKind {
  121     fn display_output(&self, _ctxt: Ctxt, mode: OutputMode) -> DisplayOutput {
  122         let str = match mode {
  123             OutputMode::Normal => match self {
  124                 CapabilityKind::Read => "Read",
  125                 CapabilityKind::Write => "Write",
  126                 CapabilityKind::Exclusive => "Exclusive",
  127                 CapabilityKind::ShallowExclusive => "ShallowExclusive",
  128             },
  129             OutputMode::Short | OutputMode::Test => match self {
  130                 CapabilityKind::Read => "R",
  131                 CapabilityKind::Write => "W",
  132                 CapabilityKind::Exclusive => "E",
  133                 CapabilityKind::ShallowExclusive => "e",
  134             },
  135         };
  136         DisplayOutput::Text(str.into())
  137     }
  138 }

  140 impl CapabilityKind {
  141     #[must_use]
  142     pub fn is_exclusive(self) -> bool {
  143         matches!(self, CapabilityKind::Exclusive)
  144     }
  145     #[must_use]
  146     pub fn is_read(self) -> bool {
  147         matches!(self, CapabilityKind::Read)
  148     }
  149     #[must_use]
  150     pub fn is_write(self) -> bool {
  151         matches!(self, CapabilityKind::Write)
  152     }
  153     #[must_use]
  154     pub fn is_shallow_exclusive(self) -> bool {
  155         matches!(self, CapabilityKind::ShallowExclusive)
  156     }
  157 
  158     #[must_use]
  159     pub fn minimum(self, other: Self) -> Option<Self> {
  160         self.partial_cmp(&other).map(|ord| match ord {
  161             Ordering::Greater => other,
  162             Ordering::Less | Ordering::Equal => self,
  163         })
  164     }
  165 }

  172     #[test]
  173     fn test_capability_gte_macro() {
  174         // Test that the macro generates correct patterns
  175         let test_read = |cap: CapabilityKind| -> bool { matches!(cap, capability_gte!(R)) };
  176         let test_write = |cap: CapabilityKind| -> bool { matches!(cap, capability_gte!(W)) };
  177         let test_shallow = |cap: CapabilityKind| -> bool { matches!(cap, capability_gte!(e)) };
  178         let test_exclusive = |cap: CapabilityKind| -> bool { matches!(cap, capability_gte!(E)) };
  179 
  180         // Test Read pattern (should match Read and Exclusive)
  181         assert!(test_read(CapabilityKind::Read));
  182         assert!(!test_read(CapabilityKind::Write));
  183         assert!(!test_read(CapabilityKind::ShallowExclusive));
  184         assert!(test_read(CapabilityKind::Exclusive));
  185 
  186         // Test Write pattern (should match Write, ShallowExclusive, and Exclusive)
  187         assert!(!test_write(CapabilityKind::Read));
  188         assert!(test_write(CapabilityKind::Write));
  189         assert!(test_write(CapabilityKind::ShallowExclusive));
  190         assert!(test_write(CapabilityKind::Exclusive));
  191 
  192         // Test ShallowExclusive pattern (should match ShallowExclusive and Exclusive)
  193         assert!(!test_shallow(CapabilityKind::Read));
  194         assert!(!test_shallow(CapabilityKind::Write));
  195         assert!(test_shallow(CapabilityKind::ShallowExclusive));
  196         assert!(test_shallow(CapabilityKind::Exclusive));
  197 
  198         // Test Exclusive pattern (should match only Exclusive)
  199         assert!(!test_exclusive(CapabilityKind::Read));
  200         assert!(!test_exclusive(CapabilityKind::Write));
  201         assert!(!test_exclusive(CapabilityKind::ShallowExclusive));
  202         assert!(test_exclusive(CapabilityKind::Exclusive));
  203 
  204         // Also test with full names
  205         assert!(matches!(CapabilityKind::Read, capability_gte!(Read)));
  206         assert!(matches!(CapabilityKind::Exclusive, capability_gte!(Read)));
  207         assert!(matches!(CapabilityKind::Write, capability_gte!(Write)));
  208         assert!(matches!(
  209             CapabilityKind::ShallowExclusive,
  210             capability_gte!(ShallowExclusive)
  211         ));
  212     }

  214     #[test]
  215     fn test_capability_kind_dag_reachability() {
  216         use petgraph::{algo::has_path_connecting, graph::DiGraph};
  217 
  218         // Create directed graph
  219         let mut graph = DiGraph::new();
  220         let mut node_indices = HashMap::new();
  221 
  222         let caps = [
  223             CapabilityKind::Exclusive,
  224             CapabilityKind::ShallowExclusive,
  225             CapabilityKind::Write,
  226             CapabilityKind::Read,
  227         ];
  228         // Add nodes
  229         for cap in caps {
  230             node_indices.insert(cap, graph.add_node(cap));
  231         }
  232 
  233         // Add edges (a -> b means a is greater than b)
  234         let edges = [
  235             (CapabilityKind::Exclusive, CapabilityKind::ShallowExclusive),
  236             (CapabilityKind::ShallowExclusive, CapabilityKind::Write),
  237             (CapabilityKind::Exclusive, CapabilityKind::Read),
  238         ];
  239 
  240         for (from, to) in edges {
  241             graph.add_edge(node_indices[&from], node_indices[&to], ());
  242         }
  243 
  244         // Test that partial_cmp matches graph reachability
  245         for a in caps {
  246             for b in caps {
  247                 if a == b {
  248                     assert_eq!(a.partial_cmp(&b), Some(Ordering::Equal));
  249                 } else if has_path_connecting(&graph, node_indices[&a], node_indices[&b], None) {
  250                     assert_eq!(a.partial_cmp(&b), Some(Ordering::Greater));
  251                 } else if has_path_connecting(&graph, node_indices[&b], node_indices[&a], None) {
  252                     assert_eq!(a.partial_cmp(&b), Some(Ordering::Less));
  253                 } else {
  254                     assert!(a.partial_cmp(&b).is_none());
  255                 }
  256             }
  257         }
  258     }
