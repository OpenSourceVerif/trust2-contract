   55     #[tracing::instrument(skip(self, obtain_type, ctxt))]
   56     fn expand_to(
   57         &mut self,
   58         place: Place<'tcx>,
   59         obtain_type: ObtainType,
   60         ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx> + HasSettings<'a>,
   61     ) -> Result<(), PcgError> {
   62         for (base, _) in place.iter_projections() {
   63             let base: Place = base.into();
   64             let base = base.with_inherent_region(ctxt);
   65             let expansion = base.expand_one_level(place, ctxt)?;
   66             if self.expand_place_one_level(base, &expansion, None, obtain_type, ctxt)? {
   67                 tracing::debug!(
   68                     "expand region projections for {} one level",
   69                     base.display_string(ctxt.ctxt())
   70                 );
   71                 self.expand_lifetime_projections_one_level(base, &expansion, obtain_type, ctxt)?;
   72             }
   73         }
   74         Ok(())
   75     }

   77     fn label_for_rp(
   78         &self,
   79         rp: LifetimeProjection<'tcx, Place<'tcx>>,
   80         obtain_type: ObtainType,
   81         ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx>,
   82     ) -> LabelForLifetimeProjection {
   83         use LabelForLifetimeProjection::{
   84             ExistingLabelOfTwoPhaseReservation, NewLabelAtCurrentLocation, NoLabel,
   85         };
   86         if obtain_type.should_label_rp(rp.base, ctxt) {
   87             NewLabelAtCurrentLocation(self.prev_snapshot_location())
   88         } else {
   89             match self.label_for_shared_expansion_of_rp(rp, ctxt) {
   90                 Some(label) => ExistingLabelOfTwoPhaseReservation(label),
   91                 None => NoLabel,
   92             }
   93         }
   94     }

   96     /// If the base of `rp` is blocked by a two-phase borrow, we want to use the
   97     /// existing label of its expansion
   98     fn label_for_shared_expansion_of_rp(
   99         &self,
  100         rp: LifetimeProjection<'tcx, Place<'tcx>>,
  101         ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx>,
  102     ) -> Option<SnapshotLocation> {
  103         ctxt.bc()
  104             .borrows_blocking(rp.base, self.location(), ctxt.bc_ctxt())
  105             .first()
  106             .map(|borrow| {
  107                 let borrow_reserve_location = get_reserve_location(borrow);
  108                 SnapshotLocation::after_statement_at(borrow_reserve_location, ctxt)
  109             })
  110     }

  119     fn expand_owned_place_one_level(
  120         &mut self,
  121         base: OwnedPlace<'tcx>,
  122         expansion: &ShallowExpansion<'tcx>,
  123         obtain_type: ObtainType,
  124         ctxt: impl HasCompilerCtxt<'a, 'tcx>,
  125     ) -> Result<bool, PcgError> {
  126         if self.contains_owned_expansion_to(expansion.target_place) {
  127             tracing::debug!(
  128                 "Already contains owned expansion from {}",
  129                 base.place().display_string(ctxt.ctxt())
  130             );
  131             return Ok(false);
  132         }
  133         tracing::debug!(
  134             "New owned expansion from {}",
  135             base.display_string(ctxt.ctxt())
  136         );
  137         if expansion.kind.is_deref_box()
  138             && obtain_type
  139                 .capability(base.place(), ctxt)
  140                 .is_shallow_exclusive()
  141         {
  142             self.apply_action(
  143                 OwnedPcgAction::new(
  144                     RepackOp::DerefShallowInit(expansion.base_place(), expansion.target_place),
  145                     None,
  146                 )
  147                 .into(),
  148             )?;
  149         } else if !base.is_raw_ptr(ctxt) {
  150             self.apply_action(
  151                 OwnedPcgAction::new(
  152                     RepackOp::expand(
  153                         expansion.base_place(),
  154                         expansion.guide(),
  155                         self.capability_for_expand(expansion.base_place(), obtain_type, ctxt),
  156                         ctxt,
  157                     ),
  158                     Some(format!("Expand owned place one level ({obtain_type:?})").into()),
  159                 )
  160                 .into(),
  161             )?;
  162         }
  163         Ok(true)
  164     }

  166     fn expand_place_one_level(
  167         &mut self,
  168         base: Place<'tcx>,
  169         expansion: &ShallowExpansion<'tcx>,
  170         guide: Option<RepackGuide>,
  171         obtain_type: ObtainType,
  172         ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx> + HasSettings<'a>,
  173     ) -> Result<bool, PcgError> {
  174         let place_expansion =
  175             PlaceExpansion::from_places(expansion.expansion(), ctxt).with_guide(guide);
  176         if matches!(expansion.kind, ProjectionKind::DerefRef(_)) {
  177             if self
  178                 .borrows_graph()
  179                 .contains_deref_edge_to(base.project_deref(ctxt))
  180             {
  181                 return Ok(false);
  182             }
  183             let blocked_lifetime_projection_label =
  184                 if base.is_mut_ref(ctxt) && obtain_type.should_label_rp(base, ctxt) {
  185                     Some(self.prev_snapshot_location())
  186                 } else {
  187                     self.label_for_shared_expansion_of_rp(
  188                         base.base_lifetime_projection(ctxt).unwrap(),
  189                         ctxt,
  190                     )
  191                 };
  192             let deref = DerefEdge::new(base, blocked_lifetime_projection_label, ctxt);
  193             self.render_debug_graph(None, "expand_place_one_level: before apply action");
  194             let action = BorrowPcgAction::add_edge(
  195                 BorrowPcgEdge::new(deref.into(), self.path_conditions()),
  196                 "expand_place_one_level: add deref edge",
  197             );
  198             self.apply_action(action.into())?;
  199             self.render_debug_graph(None, "expand_place_one_level: after apply action");
  200             self.update_capabilities_for_deref(
  201                 base,
  202                 obtain_type.capability(base, ctxt),
  203                 ctxt.bc_ctxt(),
  204             );
  205             self.render_debug_graph(
  206                 None,
  207                 "expand_place_one_level: after update_capabilities_for_deref",
  208             );
  209             if deref.blocked_lifetime_projection.label().is_some() {
  210                 self.apply_action(
  211                     BorrowPcgAction::label_lifetime_projection(
  212                         LabelNodePredicate::equals_lifetime_projection(
  213                             deref.blocked_lifetime_projection.with_label(None, ctxt),
  214                         ),
  215                         deref.blocked_lifetime_projection.label(),
  216                         "block deref",
  217                     )
  218                     .into(),
  219                 )?;
  220             }
  221             Ok(true)
  222         } else if matches!(expansion.kind, ProjectionKind::DerefRawPtr(_)) {
  223             let deref = RawPtrDerefEdge::new(base, ctxt);
  224             let action = BorrowPcgAction::add_edge(
  225                 BorrowPcgEdge::new(deref.into(), self.path_conditions()),
  226                 "expand_place_one_level: add rawptr deref edge",
  227             );
  228             self.apply_action(action.into())?;
  229             self.render_debug_graph(None, "expand_place_one_level: after apply action");
  230             let _ = self.update_capabilities_for_deref(
  231                 base,
  232                 obtain_type.capability(base, ctxt),
  233                 ctxt.bc_ctxt(),
  234             );
  235             Ok(true)
  236         } else if let Some(owned_base) = base.as_owned_place(ctxt) {
  237             self.expand_owned_place_one_level(owned_base, expansion, obtain_type, ctxt)
  238         } else {
  239             self.add_borrow_pcg_expansion(base, place_expansion, obtain_type, ctxt)
  240         }
  241     }

  249     fn add_borrow_pcg_expansion(
  250         &mut self,
  251         base: Place<'tcx>,
  252         place_expansion: PlaceExpansion<'tcx>,
  253         obtain_type: ObtainType,
  254         ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx> + HasSettings<'a>,
  255     ) -> Result<bool, PcgError>
  256     where
  257         'tcx: 'a,
  258     {
  259         let expanded_place = ExpandedPlace {
  260             place: base,
  261             expansion: place_expansion,
  262         };
  263         if self
  264             .borrows_graph()
  265             .contains_borrow_pcg_expansion(&expanded_place, ctxt.bc_ctxt())?
  266         {
  267             return Ok(false);
  268         }
  269         tracing::debug!(
  270             "Create expansion from {}",
  271             base.display_string(ctxt.bc_ctxt())
  272         );
  273         let block_type = expanded_place.expansion.block_type(base, obtain_type, ctxt);
  274         tracing::debug!(
  275             "Block type for {} is {:?}",
  276             base.display_string(ctxt.bc_ctxt()),
  277             block_type
  278         );
  279         if base.is_raw_ptr(ctxt) {
  280             return Ok(true); // We have no aliasing information, PCG should not expand anything
  281         }
  282         let expansion: BorrowPcgExpansion<'tcx> =
  283             BorrowPcgExpansion::new_place_expansion(base, &expanded_place.expansion, ctxt)?;
  284 
  285         self.render_debug_graph(
  286             None,
  287             &format!(
  288                 "add_borrow_pcg_expansion: before update_capabilities_for_borrow_expansion {}",
  289                 expansion.display_string(ctxt.bc_ctxt())
  290             ),
  291         );
  292         self.update_capabilities_for_borrow_expansion(&expansion, block_type, ctxt.bc_ctxt());
  293         self.render_debug_graph(
  294             None,
  295             "add_borrow_pcg_expansion: after update_capabilities_for_borrow_expansion",
  296         );
  297         let action = BorrowPcgAction::add_edge(
  298             BorrowPcgEdge::new(
  299                 BorrowPcgEdgeKind::BorrowPcgExpansion(expansion),
  300                 self.path_conditions(),
  301             ),
  302             "add_borrow_pcg_expansion",
  303         );
  304         self.apply_action(action.into())?;
  305         Ok(true)
  306     }

  308     #[tracing::instrument(skip(self, base, expansion, ctxt))]
  309     fn expand_lifetime_projections_one_level(
  310         &mut self,
  311         base: Place<'tcx>,
  312         expansion: &ShallowExpansion<'tcx>,
  313         obtain_type: ObtainType,
  314         ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx> + HasSettings<'a>,
  315     ) -> Result<(), PcgError> {
  316         for base_rp in base.lifetime_projections(ctxt) {
  317             if let Some(place_expansion) =
  318                 expansion.place_expansion_for_region(base_rp.region(ctxt.ctxt()), ctxt)
  319             {
  320                 tracing::debug!("Expand {}", base_rp.display_string(ctxt.bc_ctxt()));
  321                 let mut expansion = BorrowPcgExpansion::new_lifetime_projection_expansion(
  322                     base_rp,
  323                     &place_expansion,
  324                     ctxt,
  325                 )?;
  326                 let expansion_label = self.label_for_rp(base_rp, obtain_type, ctxt);
  327                 if let Some(label) = expansion_label.label() {
  328                     expansion.label_lifetime_projections(
  329                         &LabelNodePredicate::equals_lifetime_projection(base_rp.into()),
  330                         Some(label.into()),
  331                         ctxt.bc_ctxt(),
  332                     );
  333                 }
  334                 self.apply_action(
  335                     BorrowPcgAction::add_edge(
  336                         BorrowPcgEdge::new(
  337                             BorrowPcgEdgeKind::BorrowPcgExpansion(expansion.clone()),
  338                             self.path_conditions(),
  339                         ),
  340                         "expand_region_projections_one_level",
  341                     )
  342                     .into(),
  343                 )?;
  344                 if let LabelForLifetimeProjection::NewLabelAtCurrentLocation(label) =
  345                     expansion_label
  346                 {
  347                     self.apply_action(
  348                         BorrowPcgAction::label_lifetime_projection(
  349                             LabelNodePredicate::equals_lifetime_projection(base_rp.into()),
  350                             Some(label.into()),
  351                             "expand_region_projections_one_level: create new RP label",
  352                         )
  353                         .into(),
  354                     )?;
  355 
  356                     // Don't add placeholder edges for owned expansions, unless it's a deref
  357                     if base.as_borrowed_place(ctxt).is_some() || base.is_mut_ref(ctxt) {
  358                         let old_rp_base = base_rp.with_label(Some(label.into()), ctxt);
  359                         let expansion_rps = expansion
  360                             .expansion()
  361                             .iter()
  362                             .map(|node| {
  363                                 node.to_pcg_node(ctxt.bc_ctxt())
  364                                     .expect_lifetime_projection()
  365                             })
  366                             .collect::<Vec<_>>();
  367                         self.add_and_update_placeholder_edges(
  368                             old_rp_base.into(),
  369                             &expansion_rps,
  370                             "obtain",
  371                             ctxt,
  372                         )?;
  373                     }
  374                 }
  375             }
  376         }
  377         Ok(())
  378     }

  380     /// Performs bookkeeping for future nodes in the case where capability from
  381     /// (generally labelled) `origin_rp` is temporarily transferred to
  382     /// (unlabelled) `expansion_rps`, but will eventually be transferred back to
  383     /// `origin_rp`.
  384     ///
  385     /// This happens in the case of borrows and expansions of borrowed places,
  386     /// in which case `origin_rp` is labelled.
  387     /// We also use this for constructing loop abstractions where `origin_rp`
  388     /// is unlabelled (the labels are added in a later stage).
  389     ///
  390     /// The logic is as follows:
  391     ///
  392     /// Adds a node `future_rp` which is the future node for `origin_rp`.
  393     ///
  394     /// 1. Add a Future edge from `origin_rp` to `future_rp`
  395     /// 2. Add Future edges from each `expansion_rp` to `future_rp`
  396     /// 3. All Future edges with source `origin_rp` (except for the one
  397     ///    created in step 1) are modified to now have source `future_rp`
  398     ///
  399     fn add_and_update_placeholder_edges(
  400         &mut self,
  401         origin_rp: LocalLifetimeProjection<'tcx>,
  402         expansion_rps: &[LifetimeProjection<'tcx>],
  403         context: &str,
  404         ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx>,
  405     ) -> Result<(), PcgError> {
  406         if expansion_rps.is_empty() {
  407             return Ok(());
  408         }
  409         let future_rp = origin_rp.with_placeholder_label(ctxt);
  410 
  411         // Add edge {origin|r'a at l} -> {origin|r'a at FUTURE}
  412         self.apply_action(
  413             BorrowPcgAction::add_edge(
  414                 BorrowPcgEdge::new(
  415                     BorrowFlowEdge::new(
  416                         origin_rp.into(),
  417                         future_rp,
  418                         BorrowFlowEdgeKind::Future(FutureEdgeKind::ToFutureSelf),
  419                     )
  420                     .into(),
  421                     self.path_conditions(),
  422                 ),
  423                 format!("{context}: placeholder bookkeeping"),
  424             )
  425             .into(),
  426         )?;
  427 
  428         // For each field F add edge {origin.F|'a} -> {origin|r'a at FUTURE}
  429         for expansion_rp in expansion_rps {
  430             self.apply_action(
  431                 BorrowPcgAction::add_edge(
  432                     BorrowPcgEdge::new(
  433                         BorrowFlowEdge::new(
  434                             *expansion_rp,
  435                             future_rp,
  436                             BorrowFlowEdgeKind::Future(FutureEdgeKind::FromExpansion),
  437                         )
  438                         .into(),
  439                         self.path_conditions(),
  440                     ),
  441                     format!("{context}: placeholder bookkeeping"),
  442                 )
  443                 .into(),
  444             )?;
  445         }
  446         self.redirect_source_of_future_edges(origin_rp, future_rp, ctxt)
  447     }
  448 
  449     fn redirect_source_of_future_edges(
  450         &mut self,
  451         old_source: LocalLifetimeProjection<'tcx>,
  452         new_source: LocalLifetimeProjection<'tcx>,
  453         ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx>,
  454     ) -> Result<(), PcgError> {
  455         let to_replace = self
  456             .borrows_graph()
  457             .edges_blocking(old_source.into(), ctxt.bc_ctxt())
  458             .filter_map(|edge| {
  459                 if let BorrowPcgEdgeKind::BorrowFlow(BorrowFlowEdge {
  460                     kind: kind @ BorrowFlowEdgeKind::Future(_),
  461                     target: short,
  462                     ..
  463                 }) = edge.kind
  464                     && *short != new_source
  465                 {
  466                     return Some((
  467                         edge.to_owned_edge(),
  468                         BorrowPcgEdge::new(
  469                             BorrowFlowEdge::new(new_source.into(), *short, *kind).into(),
  470                             edge.conditions.clone(),
  471                         ),
  472                     ));
  473                 }
  474                 None
  475             })
  476             .collect::<Vec<_>>();
  477         for (to_remove, to_insert) in to_replace {
  478             self.apply_action(
  479                 BorrowPcgAction::remove_edge(to_remove, "placeholder bookkeeping").into(),
  480             )?;
  481             self.apply_action(
  482                 BorrowPcgAction::add_edge(to_insert, "placeholder bookkeeping").into(),
  483             )?;
  484         }
  485         Ok(())
  486     }
