   14 pub(crate) fn ty_pure<'vir>(
   15     task_key: &TyData<'vir, RustTyDatas>,
   16     data: &RustMutRef<'vir>,
   17     deps: &mut TaskEncoderDependencies<'vir, TyPureEnc>,
   18     builder: &mut AdtBuilder<'vir>,
   19 ) -> Result<TyPureMutRef<'vir>, EncodeFullError<'vir, TyPureEnc>> {
   20     let ty = data.metadata.decompose(task_key.params);
   21     let metadata = deps.require_ref::<TyUsePureEnc>(ty)?.snapshot.downcast_ty();
   22 
   23     let ty = data.referent.decompose(task_key.params);
   24     let referent = deps.require_ref::<TyUsePureEnc>(ty)?.snapshot.downcast_ty();
   25 
   26     let (field_snaps_to_snap, field_access) =
   27         builder.constructor("", (vir::TYPE_REF, metadata, referent), None);
   28 
   29     Ok(TyPureMutRefData {
   30         prim_to_snap: field_snaps_to_snap,
   31         deref_access: field_access[0].downcast_ty(),
   32         metadata_access: field_access[1].downcast_ty(),
   33         value_access: field_access[2].downcast_ty(),
   34     })
   35 }

   37 pub(crate) fn ty_impure<'vir>(
   38     task_key: &TyData<'vir, (RustTyDatas, PureTyDatas)>,
   39     data: &(&RustMutRef<'vir>, &TyPureMutRef<'vir>),
   40     deps: &mut TaskEncoderDependencies<'vir, TyImpureEnc>,
   41     builder: &mut PredicateBuilder<'vir>,
   42 ) -> Result<TyImpureMutRef<'vir>, EncodeFullError<'vir, TyImpureEnc>> {
   43     let snap_type = builder.csnap_type();
   44     let ref_param = builder.vcx.mk_local_decl("r", vir::TYPE_REF);
   45     let ref_ex = builder.vcx.mk_local_ex(ref_param);
   46 
   47     let metadata_decomp = data.0.metadata.decompose(task_key.0.params);
   48     deps.require_dep::<TyUseImpureEnc>(metadata_decomp)?;
   49     let inner_type = data.0.referent.decompose(task_key.0.params);
   50     deps.require_dep::<TyUseImpureEnc>(inner_type)?;
   51 
   52     let metadata_type = deps
   53         .require_ref::<TyUsePureEnc>(metadata_decomp)?
   54         .snapshot
   55         .downcast_ty();
   56     let metadata_param = builder.vcx.mk_local_decl("metadata", metadata_type);
   57     let metadata_ex = builder.vcx.mk_local_ex(metadata_param);
   58     let arbitrary_value = builder.inner.function(
   59         "arbitrary_value",
   60         (
   61             vir::TYPE_REF,
   62             metadata_type,
   63             builder.params.ty_args(),
   64             builder.params.const_args(),
   65         ),
   66         snap_type,
   67         (
   68             ref_param,
   69             metadata_param,
   70             builder.params.ty_decls(),
   71             builder.params.const_decls(),
   72         ),
   73         &[],
   74         &[
   75             vir::expr! {
   76                 ([data.1.deref_access](result: [snap_type])) == ([ref_ex])
   77             },
   78             vir::expr! {
   79                 ([data.1.metadata_access](result: [snap_type])) == ([metadata_ex])
   80             },
   81         ],
   82         None,
   83     );
   84 
   85     let ref_self_decl = builder.ref_self_decl();
   86     let ref_self = builder.vcx.mk_local_ex(ref_self_decl);
   87     let vcx = builder.vcx;
   88 
   89     // fields: the referent's address and the pointer metadata. Unlike a
   90     // shared reference, no permission to the referent is held, so its value
   91     // stays unconstrained -- `arbitrary_value` supplies it, keyed on the
   92     // address so that it is at least stable across reborrows.
   93     let addr_field = builder.field("addr", vir::TYPE_REF);
   94     let metadata_field = builder.field("metadata", metadata_type);
   95 
   96     let metadata_ty = deps.require_dep::<TyExprEnc>(metadata_decomp)?;
   97     let typ = deps.require_dep::<ParamTypEnc>(())?.typ;
   98     let metadata_typ = |metadata| vcx.mk_eq_expr(typ(metadata), metadata_ty);
   99 
  100     // main predicate
  101     builder.mk_predicate(
  102         "",
  103         Some(vcx.mk_conj(&[
  104             vir::expr! { acc((ref_self).[addr_field]) },
  105             vir::expr! { acc((ref_self).[metadata_field]) },
  106             metadata_typ(vir::expr! { [metadata_field](ref_self) }),
  107         ])),
  108     );
  109 
  110     // Ref-to-snap
  111     builder.mk_snap_function(
  112         Some(arbitrary_value(
  113             vir::expr! { [addr_field](ref_self) },
  114             vir::expr! { [metadata_field](ref_self) },
  115             builder.params.ty_exprs(),
  116             builder.params.const_exprs(),
  117         )),
  118         &[metadata_typ(data.1.metadata_access.call()(
  119             vcx.mk_result(snap_type),
  120         ))],
  121     );
  122 
  123     Ok(TyImpureMutRefData {
  124         pure: *data.1,
  125         arbitrary_value,
  126     })
  127 }
