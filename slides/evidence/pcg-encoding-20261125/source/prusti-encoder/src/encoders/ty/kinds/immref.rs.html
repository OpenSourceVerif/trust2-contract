   15 pub(crate) fn ty_pure<'vir>(
   16     task_key: &TyData<'vir, RustTyDatas>,
   17     data: &RustImmRef<'vir>,
   18     deps: &mut TaskEncoderDependencies<'vir, TyPureEnc>,
   19     builder: &mut AdtBuilder<'vir>,
   20 ) -> Result<TyPureImmRef<'vir>, EncodeFullError<'vir, TyPureEnc>> {
   21     let ty = data.metadata.decompose(task_key.params);
   22     let metadata = deps.require_ref::<TyUsePureEnc>(ty)?.snapshot.downcast_ty();
   23 
   24     let ty = data.referent.decompose(task_key.params);
   25     let referent = deps.require_ref::<TyUsePureEnc>(ty)?.snapshot.downcast_ty();
   26 
   27     let (field_snaps_to_snap, field_access) =
   28         builder.constructor("", (vir::TYPE_REF, metadata, referent), None);
   29 
   30     Ok(TyPureImmRefData {
   31         prim_to_snap: field_snaps_to_snap,
   32         deref_access: field_access[0].downcast_ty(),
   33         metadata_access: field_access[1].downcast_ty(),
   34         value_access: field_access[2].downcast_ty(),
   35     })
   36 }

   38 pub(crate) fn ty_impure<'vir>(
   39     task_key: &TyData<'vir, (RustTyDatas, PureTyDatas)>,
   40     data: &(&RustImmRef<'vir>, &TyPureImmRef<'vir>),
   41     deps: &mut TaskEncoderDependencies<'vir, TyImpureEnc>,
   42     builder: &mut PredicateBuilder<'vir>,
   43 ) -> Result<TyImpureImmRef<'vir>, EncodeFullError<'vir, TyImpureEnc>> {
   44     let metadata_type = data.0.metadata.decompose(task_key.0.params);
   45     deps.require_dep::<TyUseImpureEnc>(metadata_type)?;
   46     let inner_type = data.0.referent.decompose(task_key.0.params);
   47     deps.require_dep::<TyUseImpureEnc>(inner_type)?;
   48 
   49     let ref_self_decl = builder.ref_self_decl();
   50     let ref_self = builder.vcx.mk_local_ex(ref_self_decl);
   51 
   52     let metadata_snap = deps
   53         .require_ref::<TyUsePureEnc>(metadata_type)?
   54         .snapshot
   55         .downcast_ty();
   56 
   57     // fields: the referent's address and the pointer metadata. The referent's
   58     // value is not stored; it is read out of the `p_Param` held below, which
   59     // is what ties it to the type this predicate is instantiated at.
   60     let addr_field = builder.field("addr", vir::TYPE_REF);
   61     let metadata_field = builder.field("metadata", metadata_snap);
   62 
   63     let vcx = builder.vcx;
   64     let read = deps.require_dep::<ReadPermEnc>(())?.read;
   65     let param_ty = deps.require_ref::<TyImpureEnc>(RustTyDecomposition::param())?;
   66     let referent_ty = deps.require_dep::<TyExprEnc>(inner_type)?;
   67     let metadata_ty = deps.require_dep::<TyExprEnc>(metadata_type)?;
   68     let addr = vir::expr! { [addr_field](ref_self) };
   69     let referent_pred = vcx.mk_predicate_app_expr((param_ty.ref_to_pred)(
   70         addr,
   71         vcx.alloc_slice(&[referent_ty]),
   72         &[],
   73     )(Some(read())));
   74     // Unlike the referent, the metadata is stored rather than held as a
   75     // `p_Param`, so its type has to be stated for the variant bridge.
   76     let typ = deps.require_dep::<ParamTypEnc>(())?.typ;
   77     let metadata_typ = |metadata| vcx.mk_eq_expr(typ(metadata), metadata_ty);
   78 
   79     // main predicate
   80     builder.mk_predicate(
   81         "",
   82         Some(vcx.mk_conj(&[
   83             vir::expr! { acc((ref_self).[addr_field]) },
   84             vir::expr! { acc((ref_self).[metadata_field]) },
   85             metadata_typ(vir::expr! { [metadata_field](ref_self) }),
   86             referent_pred,
   87         ])),
   88     );
   89 
   90     // Ref-to-snap: the referent's value comes from the `p_Param` above, whose
   91     // snapshot carries its type, so the type is known by construction.
   92     builder.mk_snap_function(
   93         Some(data.1.prim_to_snap.call()(
   94             addr,
   95             vir::expr! { [metadata_field](ref_self) },
   96             (param_ty.ref_to_snap)(addr, vcx.alloc_slice(&[referent_ty]), &[]).downcast_ty(),
   97         )),
   98         &[
   99             metadata_typ(data.1.metadata_access.call()(
  100                 vcx.mk_result(builder.csnap_type()),
  101             )),
  102             vcx.mk_eq_expr(
  103                 typ(data.1.value_access.call()(
  104                     vcx.mk_result(builder.csnap_type()),
  105                 )),
  106                 referent_ty,
  107             ),
  108         ],
  109     );
  110 
  111     Ok(TyImpureImmRefData {})
  112 }
