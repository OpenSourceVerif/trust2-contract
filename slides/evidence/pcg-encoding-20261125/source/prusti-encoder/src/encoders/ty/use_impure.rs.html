  144     fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
  145         *task

  148     fn do_encode_full<'vir>(
  149         task_key: &Self::TaskKey<'vir>,
  150         deps: &mut task_encoder::TaskEncoderDependencies<'vir, Self>,
  151     ) -> EncodeFullResult<'vir, Self> {
  152         deps.emit_output_ref(*task_key, ())?;
  153 
  154         let ty_impure = deps.require_dep::<TyImpureEnc>(task_key.ty)?;
  155         let inhabited = deps.require_ref::<TyUseInhabitedEnc>(*task_key)?;
  156         let mut walker = TyUseImpureWalker::new(deps, task_key.args, inhabited)?;
  157         let ty_use_impure = walker.encode_ty(task_key.ty.zip(ty_impure))?;
  158         Ok(((), ty_use_impure.alloc()))
  159     }
  160 
  161     fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
  162         TyImpureEnc::emit_outputs(program)
  163     }

  173 impl<'a, 'vir> TyUseImpureWalker<'a, 'vir> {
  174     fn new(
  175         deps: &'a mut TaskEncoderDependencies<'vir, TyUseImpureEnc>,
  176         args: GArgs<'vir>,
  177         inhabited: TyUseInhabitedRef<'vir>,
  178     ) -> EncResult<'vir, Self> {
  179         let args_t = deps.require_dep::<GArgsTyEnc>(args)?;
  180         Ok(Self {
  181             deps,
  182             args_t,
  183             args,
  184             inhabited,
  185         })

  188     fn encode_ty(
  189         &mut self,
  190         ty: TyData<'vir, (RustTyDatas, ImpureTyDatas)>,
  191     ) -> EncResult<'vir, TyData<'vir, UseImpureTyDatas>> {
  192         let specifics = match &ty.specifics {
  193             TySpecifics::Param(..) => TySpecifics::mk_param(()),
  194             TySpecifics::Opaque(..) => TySpecifics::mk_opaque(()),
  195             TySpecifics::Primitive(..) => TySpecifics::mk_primitive(()),
  196             TySpecifics::ImmRef(data) => {
  197                 let referent_caster = self.encode_normalized(data.0.referent, ty.0.params)?;
  198                 let metadata_caster = self.encode_normalized_pure(data.0.metadata, ty.0.params)?;
  199                 TySpecifics::mk_immref(TyUseImpureImmRef {
  200                     referent_caster,
  201                     metadata_caster,
  202                     args: self.args_t,
  203                     impure: *data.1,
  204                 })
  205             }
  206             TySpecifics::MutRef(data) => {
  207                 let referent_caster = self.encode_normalized(data.0.referent, ty.0.params)?;
  208                 let metadata_caster = self.encode_normalized_pure(data.0.metadata, ty.0.params)?;
  209                 TySpecifics::mk_mutref(TyUseImpureMutRef {
  210                     referent_caster,
  211                     metadata_caster,
  212                     args: self.args_t,
  213                     impure: *data.1,
  214                     ref_to_snap: ty.1.ref_to_snap,
  215                 })
  216             }
  217             TySpecifics::Raw(data) => {
  218                 let metadata_caster = self.encode_normalized_pure(data.0.metadata, ty.0.params)?;
  219                 TySpecifics::mk_raw(TyUseImpureRaw {
  220                     metadata_caster,
  221                     args: self.args_t,
  222                     impure: *data.1,
  223                 })
  224             }
  225             TySpecifics::ArrayLike(data) => {
  226                 TySpecifics::ArrayLike(self.encode_array(data, ty.1.ref_to_pred, ty.0.params)?)
  227             }
  228             TySpecifics::StructLike(data) => {
  229                 let box_data = match ty.0.special {
  230                     RustTySpecial::Box => Some(TyUseImpureBoxData {
  231                         metadata_caster: self
  232                             .encode_normalized_pure(ty.box_metadata_ty(), ty.0.params)?,
  233                         unique: self.deps.require_dep::<TyUseImpureEnc>(
  234                             ty.box_unique_ty().decompose_normalize(self.args),
  235                         )?,
  236                         alloc: self.deps.require_dep::<TyUseImpureEnc>(
  237                             ty.box_alloc_ty().decompose_normalize(self.args),
  238                         )?,
  239                     }),
  240                     RustTySpecial::None => None,
  241                 };
  242                 TySpecifics::StructLike(self.encode_structlike(
  243                     data,
  244                     ty.1.ref_to_pred,
  245                     ty.0.params,
  246                     box_data,
  247                 )?)
  248             }
  249             TySpecifics::EnumLike(data) => {
  250                 TySpecifics::EnumLike(self.encode_enumlike(data, ty.0.params)?)
  251             }
  252             TySpecifics::Builtin(..) => TySpecifics::mk_builtin(()),
  253         };
  254         let data = TyUseImpureData {
  255             args: self.args_t,
  256             impure: *ty.1,
  257             inhabited: self.inhabited,
  258         };
  259         Ok(TyData::new(data, specifics))
  260     }
  261 
  262     fn encode_normalized(
  263         &mut self,
  264         inner: LazyRustTy<'vir>,
  265         params: GParams<'vir>,
  266     ) -> EncResult<'vir, FieldCaster<'vir>> {
  267         let normalized = inner.decompose_compare_normalize(params, self.args);
  268         self.deps.require_dep::<GArgsCastEnc<Impure>>(normalized)
  269     }
  270 
  271     fn encode_normalized_pure(
  272         &mut self,
  273         inner: LazyRustTy<'vir>,
  274         params: GParams<'vir>,
  275     ) -> EncResult<'vir, GArgCaster<'vir, crate::encoders::Pure>> {
  276         let normalized = inner.decompose_compare_normalize(params, self.args);
  277         self.deps
  278             .require_dep::<GArgsCastEnc<crate::encoders::Pure>>(normalized)
  279     }

  281     fn encode_array(
  282         &mut self,
  283         data: &ArrayData<'vir, (RustTyDatas, ImpureTyDatas)>,
  284         _ref_to_pred: PredicateIdn<'vir, (vir::Ref, vir::ManyTyVal, vir::ManyCSnap)>,
  285         params: GParams<'vir>,
  286     ) -> EncResult<'vir, ArrayData<'vir, UseImpureTyDatas>> {
  287         let caster = self.encode_normalized(*data.0, params)?;
  288         let slice = data.slice;
  289         let data = TyUseImpureArrayData {
  290             args: self.args_t,
  291             impure: *data.data.1,
  292             element_caster: caster,
  293         };
  294         Ok(ArrayData::new(data, slice))
  295     }
  296 
  297     fn encode_structlike(
  298         &mut self,
  299         data: &StructData<'vir, (RustTyDatas, ImpureTyDatas)>,
  300         ref_to_pred: PredicateIdn<'vir, (vir::Ref, vir::ManyTyVal, vir::ManyCSnap)>,
  301         params: GParams<'vir>,
  302         box_data: Option<TyUseImpureBoxData<'vir>>,
  303     ) -> EncResult<'vir, StructData<'vir, UseImpureTyDatas>> {
  304         let fields = data
  305             .fields
  306             .iter()
  307             .map(|field| {
  308                 let caster = self.encode_normalized(field.0.ty(), params)?;
  309                 Ok(TyUseImpureField {
  310                     caster,
  311                     args: self.args_t,
  312                     impure: *field.1,
  313                 })
  314             })
  315             .collect::<EncResult<'vir, Vec<_>>>()?;
  316         let data = TyUseImpureStructData {
  317             args: self.args_t,
  318             ref_to_pred,
  319             impure: *data.1,
  320             box_data,
  321         };
  322         Ok(StructData::new(data, fields))
  323     }
  324 
  325     fn encode_enumlike(
  326         &mut self,
  327         data: &EnumData<'vir, (RustTyDatas, ImpureTyDatas)>,
  328         params: GParams<'vir>,
  329     ) -> EncResult<'vir, EnumData<'vir, UseImpureTyDatas>> {
  330         let discr_ty = self
  331             .deps
  332             .require_dep::<TyPureEnc>(RustTyDecomposition::from_prim_ty(data.0.discr).ty)?;
  333         let discr_prim = *discr_ty.expect_primitive();
  334         let variant_inhabited = data
  335             .variants
  336             .iter()
  337             .map(|variant| {
  338                 let fields = variant
  339                     .inner
  340                     .fields
  341                     .iter()
  342                     .map(|field| {
  343                         let field = field.0.ty().decompose_normalize(self.args);
  344                         Ok(self
  345                             .deps
  346                             .require_ref::<TyUseInhabitedEnc>(field)?
  347                             .inhabited())
  348                     })
  349                     .collect::<EncResult<'vir, Vec<_>>>()?;
  350                 let inhabited = vir::with_vcx(|vcx| vcx.mk_conj(&fields));
  351                 let discr = discr_prim
  352                     .prim_to_snap(discr_prim.expr_from_bits(data.0.discr, variant.0.discr_val));
  353                 Ok((discr, inhabited))
  354             })
  355             .collect::<EncResult<'vir, Vec<_>>>()?;
  356         let variant_inhabited = vir::with_vcx(|vcx| vcx.alloc_slice(&variant_inhabited));
  357 
  358         let variants = data
  359             .variants
  360             .iter()
  361             .map(|variant| {
  362                 let structlike =
  363                     self.encode_structlike(&variant.inner, variant.1.predicate, params, None)?;
  364                 Ok(VariantData::new((), structlike))
  365             })
  366             .collect::<EncResult<'vir, Vec<_>>>()?;
  367         let data = TyUseImpureEnumData {
  368             args: self.args_t,
  369             impure: *data.1,
  370             variant_inhabited,
  371         };
  372         Ok(EnumData::new(data, variants))
  373     }
  374 }

  376 impl<'vir> TyUseImpureData<'vir> {
  377     /// Generates a call to `method_assign`, which asserts that the snapshot of
  378     /// `self_ref` is `self_new_snap`. Appropriate type arguments are used.
  379     pub fn apply_method_assign<'tcx>(
  380         &self,
  381         vcx: &'vir vir::VirCtxt<'tcx>,
  382         self_ref: vir::ExprRef<'vir>,
  383         self_new_snap: vir::ExprSnap<'vir>,
  384     ) -> vir::Stmt<'vir> {
  385         vcx.alloc(vir::StmtData::new(vcx.alloc((self.impure.method_assign)(
  386             self_ref,
  387             self.args.get_ty(),
  388             self.args.get_const(),
  389             self_new_snap,
  390         ))))
  391     }
  392 
  393     /// Constructs the Viper predicate application expression.
  394     pub fn ref_to_pred<'tcx>(
  395         &self,
  396         vcx: &'vir vir::VirCtxt<'tcx>,
  397         self_ref: vir::ExprRef<'vir>,
  398         perm: Option<vir::ExprPerm<'vir>>,
  399     ) -> vir::ExprBool<'vir> {
  400         let inhabited = self.inhabited.inhabited();
  401         let predicate = vcx.mk_predicate_app_expr(self.ref_to_pred_app(self_ref, perm));
  402         vcx.mk_conj(&[inhabited, predicate])
  403     }
  404 
  405     /// Constructs the Viper predicate application.
  406     pub fn ref_to_pred_app(
  407         &self,
  408         self_ref: vir::ExprRef<'vir>,
  409         perm: Option<vir::ExprPerm<'vir>>,
  410     ) -> vir::PredicateApp<'vir> {
  411         (self.impure.ref_to_pred)(self_ref, self.args.get_ty(), self.args.get_const())(perm)
  412     }

  414     /// Calls the predicate (heap) dependent snapshot construction function.
  415     pub fn ref_to_snap<Curr, Next>(
  416         &self,
  417         self_ref: vir::ExprGenRef<'vir, Curr, Next>,
  418     ) -> vir::ExprGenSnap<'vir, Curr, Next> {
  419         self.impure.ref_to_snap.call()(self_ref, self.args.get_ty(), self.args.get_const())
  420     }

  422     pub fn snapshot(&self) -> vir::TypeSnap<'vir> {
  423         self.impure.ref_to_snap.result()

  429         self.data.inhabited.inhabited()

  432     /// The predicates that `fold` consumes besides those of the places the
  433     /// predicate is expanded into (its fields, variant, element, or the value
  434     /// of a `Box`): the discriminant of an enum, the rest of an array whose
  435     /// element is folded back, and the other fields of a `Box`.
  436     pub fn fold_frame(
  437         &self,
  438         variant: Option<abi::VariantIdx>,
  439         self_ref: vir::ExprRef<'vir>,
  440         index: Option<vir::ExprCSnap<'vir>>,
  441     ) -> Vec<vir::ExprBool<'vir>> {
  442         if variant.is_some() {
  443             return Vec::new();
  444         }
  445         vir::with_vcx(|vcx| match &self.specifics {
  446             TySpecifics::EnumLike(data) => {
  447                 let discr = data.discr(self_ref);
  448                 vec![data.discr_ty().ref_to_pred(vcx, discr, None)]
  449             }
  450             TySpecifics::ArrayLike(array) => {
  451                 let index = index.expect("cannot fold array type without index");
  452                 vec![array.ref_to_index_frame_pred(self_ref, index)]
  453             }
  454             TySpecifics::StructLike(data) => match data.data.box_data {
  455                 Some(box_data) => vec![
  456                     box_data
  457                         .unique
  458                         .ref_to_pred(vcx, data.fields[0].field_ref(self_ref), None),
  459                     box_data
  460                         .alloc
  461                         .ref_to_pred(vcx, data.fields[1].field_ref(self_ref), None),
  462                 ],
  463                 None => Vec::new(),
  464             },
  465             _ => Vec::new(),
  466         })
  467     }
  468 
  469     /// Fold the predicate (including generic casts).
  470     pub fn fold(
  471         &self,
  472         variant: Option<abi::VariantIdx>,
  473         self_ref: vir::ExprRef<'vir>,
  474         index: Option<vir::ExprCSnap<'vir>>,
  475         perm: Option<vir::ExprPerm<'vir>>,
  476         label: Option<vir::OldLabel<'vir>>,
  477     ) -> Vec<vir::Stmt<'vir>> {
  478         if let Some(variant) = variant {
  479             return self
  480                 .expect_variant(variant)
  481                 .inner
  482                 .fold(self_ref, perm)
  483                 .collect();
  484         };
  485         match &self.specifics {
  486             TySpecifics::Param(_) | TySpecifics::Primitive(_) | TySpecifics::Builtin(_) => {
  487                 unreachable!()
  488             }
  489             TySpecifics::Opaque(_) => panic!("cannot fold opaque type"),
  490             TySpecifics::ArrayLike(array) => {
  491                 let index = index.expect("cannot fold array type without index");
  492                 array
  493                     .element_caster
  494                     .cast_to_callee_ctx(array.ref_to_index_ref(self_ref, index))
  495                     .into_iter()
  496                     .chain([vir::with_vcx(|vcx| {
  497                         vcx.alloc(vir::StmtData::new(vcx.alloc(
  498                             (array.data.impure.method_fold)(
  499                                 index,
  500                                 self_ref,
  501                                 self.args.get_ty(),
  502                                 self.args.get_const(),
  503                             ),
  504                         )))
  505                     })])
  506                     .collect()
  507             }
  508             TySpecifics::ImmRef(..) | TySpecifics::Raw(..) => Vec::new(),
  509             TySpecifics::MutRef(data) => data.fold(self_ref, label).into_iter().collect(),
  510             TySpecifics::StructLike(data) => data.fold(self_ref, perm).collect(),
  511             TySpecifics::EnumLike(..) => {
  512                 let pred_app = self.ref_to_pred_app(self_ref, perm);
  513                 vec![vir::with_vcx(|vcx| vcx.mk_fold_stmt(pred_app))]
  514             }
  515         }
  516     }
  517 
  518     /// Unfold the predicate (including generic casts).
  519     pub fn unfold(
  520         &self,
  521         variant: Option<abi::VariantIdx>,
  522         self_ref: vir::ExprRef<'vir>,
  523         index: Option<vir::ExprCSnap<'vir>>,
  524         perm: Option<vir::ExprPerm<'vir>>,
  525         old: Option<vir::OldLabel<'vir>>,
  526     ) -> Vec<vir::Stmt<'vir>> {
  527         if let Some(variant) = variant {
  528             return self
  529                 .expect_variant(variant)
  530                 .inner
  531                 .unfold(self_ref, perm)
  532                 .collect();
  533         };
  534         match &self.specifics {
  535             TySpecifics::Param(_) | TySpecifics::Primitive(_) | TySpecifics::Builtin(_) => {
  536                 unreachable!()
  537             }
  538             TySpecifics::Opaque(_) => panic!("cannot unfold opaque type"),
  539             TySpecifics::ArrayLike(array) => {
  540                 let index = index.expect("cannot unfold array type without index");
  541                 [vir::with_vcx(|vcx| {
  542                     vcx.alloc(vir::StmtData::new(vcx.alloc(
  543                         (array.data.impure.method_unfold)(
  544                             index,
  545                             self_ref,
  546                             self.args.get_ty(),
  547                             self.args.get_const(),
  548                         ),
  549                     )))
  550                 })]
  551                 .into_iter()
  552                 .chain(
  553                     array
  554                         .element_caster
  555                         .cast_to_caller_ctx(array.ref_to_index_ref(self_ref, index)),
  556                 )
  557                 .collect()
  558             }
  559             TySpecifics::ImmRef(..) | TySpecifics::Raw(..) => Vec::new(),
  560             TySpecifics::MutRef(data) => data.unfold(self_ref, old).into_iter().collect(),
  561             TySpecifics::StructLike(data) => data.unfold(self_ref, perm).collect(),
  562             TySpecifics::EnumLike(data) => {
  563                 let pred_app = self.ref_to_pred_app(self_ref, perm);
  564                 vir::with_vcx(|vcx| {
  565                     let discr = data
  566                         .discr_ty()
  567                         .ref_to_snap(data.discr(self_ref))
  568                         .downcast_ty::<vir::CSnap>();
  569                     std::iter::once(vcx.mk_unfold_stmt(pred_app))
  570                         .chain(data.variant_inhabited.iter().map(
  571                             |(variant_discr, variant_inhabited)| {
  572                                 let is_variant = vcx.mk_eq_expr(discr, *variant_discr);
  573                                 let implication = vcx
  574                                     .mk_bin_op_expr(
  575                                         vir::BinOpKind::Implies,
  576                                         is_variant,
  577                                         *variant_inhabited,
  578                                     )
  579                                     .downcast_ty();
  580                                 vcx.mk_inhale_stmt(implication)
  581                             },
  582                         ))
  583                         .collect()
  584                 })
  585             }
  586         }
  587     }
  588 }

  617 impl<'vir> TyUseImpureStruct<'vir> {
  618     /// The pointer metadata of a `Box`, read out of the (folded) `Unique`
  619     /// field's predicate.
  620     #[track_caller]
  621     pub fn box_metadata(&self, self_ref: vir::ExprRef<'vir>) -> vir::ExprSnap<'vir> {
  622         let box_data = self.data.impure.box_data.expect("expected box");
  623         let metadata = box_data.metadata.call()(
  624             self_ref,
  625             self.data.args.get_ty(),
  626             self.data.args.get_const(),
  627         );
  628         self.data
  629             .box_data
  630             .expect("expected box")
  631             .metadata_caster
  632             .cast_to_caller_ctx(metadata.upcast_ty())
  633     }
  634 
  635     /// The (Ref) address of the boxed `T` (the box's last field), read out of
  636     /// the (folded) `Unique` field's predicate.
  637     #[track_caller]
  638     pub fn box_address(&self, self_ref: vir::ExprRef<'vir>) -> vir::ExprRef<'vir> {
  639         assert!(self.data.impure.box_data.is_some(), "expected box");
  640         self.fields.last().unwrap().field_ref(self_ref)
  641     }
  642 
  643     fn ref_to_pred_app(
  644         &self,
  645         self_ref: vir::ExprRef<'vir>,
  646         perm: Option<vir::ExprPerm<'vir>>,
  647     ) -> vir::PredicateApp<'vir> {
  648         (self.ref_to_pred)(self_ref, self.args.get_ty(), self.args.get_const())(perm)
  649     }
  650 
  651     /// Fold the predicate (including generic casts).
  652     fn fold(
  653         &self,
  654         self_ref: vir::ExprRef<'vir>,
  655         perm: Option<vir::ExprPerm<'vir>>,
  656     ) -> impl Iterator<Item = vir::Stmt<'vir>> + '_ {
  657         let pred_app = self.ref_to_pred_app(self_ref, perm);
  658         let fold = vir::with_vcx(|vcx| vcx.mk_fold_stmt(pred_app));
  659         self.cast_to_callee_ctx(self_ref).chain([fold])
  660     }
  661 
  662     /// Unfold the predicate (including generic casts).
  663     fn unfold(
  664         &self,
  665         self_ref: vir::ExprRef<'vir>,
  666         perm: Option<vir::ExprPerm<'vir>>,
  667     ) -> impl Iterator<Item = vir::Stmt<'vir>> + '_ {
  668         let pred_app = self.ref_to_pred_app(self_ref, perm);
  669         let unfold = vir::with_vcx(|vcx| vcx.mk_unfold_stmt(pred_app));
  670         [unfold]
  671             .into_iter()
  672             .chain(self.cast_to_caller_ctx(self_ref))
  673     }
  674 
  675     fn cast_to_caller_ctx(
  676         &self,
  677         self_ref: vir::ExprRef<'vir>,
  678     ) -> impl Iterator<Item = vir::Stmt<'vir>> {
  679         self.fields
  680             .iter()
  681             .filter_map(|f| f.cast_to_caller_ctx(self_ref))
  682     }
  683 
  684     fn cast_to_callee_ctx(
  685         &self,
  686         self_ref: vir::ExprRef<'vir>,
  687     ) -> impl Iterator<Item = vir::Stmt<'vir>> {
  688         self.fields
  689             .iter()
  690             .filter_map(|f| f.cast_to_callee_ctx(self_ref))
  691     }
  692 }

  743         (self.impure.arbitrary_value)(
  744             self_ref,
  745             metadata.downcast_ty(),
  746             self.args.get_ty(),
  747             self.args.get_const(),
  748         )
  749     }
