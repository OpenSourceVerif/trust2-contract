    8 /// The inputs of a coupled edge with the resources each is given back as.
    9 type Resources<'a> = Vec<(PcgNode<'a>, Vec<vir::ExprBool<'a>>)>;
   10 
   11 impl<'vir, 'enc, E: TaskEncoder> ImpureEncVisitor<'vir, 'enc, E> {
   12     pub(crate) fn pcs_handle_wand(
   13         &mut self,
   14         borrows_state: &BorrowsState<'_, 'vir>,
   15         package: bool,
   16         edge: &PcgCoupledEdgeKind<'vir>,
   17         label: Option<&'vir str>,
   18         edge_to_loop: bool,
   19         footprint: Option<&mut Footprint<'_, 'vir>>,
   20     ) -> EncodeResult<'vir, (), E> {
   21         // TODO: there is something in the pcs which emits spurious? opaque edge creation, skip these
   22         if package && !edge_to_loop {
   23             return Ok(());
   24         }
   25         let mut old_outer = WandOldOuter::Label(label);
   26         let (wand, rhs) = self.coupled_wand(edge, &mut old_outer)?;
   27         if !package {
   28             if let Some(footprint) = footprint {
   29                 footprint.parts.push(self.vcx.mk_wand_expr(wand));
   30             }
   31             self.stmt(self.vcx.mk_apply_stmt(wand));
   32             return Ok(());
   33         }
   34         let WandOldOuter::Label(label) = &mut old_outer else {
   35             unreachable!()
   36         };
   37         let label = *label.get_or_insert_with(|| self.new_label("outer_package"));
   38         let proof_block = self.pcs_package_script(borrows_state, rhs, label)?;
   39         let proof_block = self.vcx.alloc_slice(&proof_block);
   40         self.stmt(self.vcx.mk_package_stmt(wand, proof_block));
   41         Ok(())
   42     }
   43 
   44     /// The wand of a coupled edge, and the resources each input is given
   45     /// back as on its right-hand side.
   46     pub(super) fn coupled_wand(
   47         &mut self,
   48         edge: &PcgCoupledEdgeKind<'vir>,
   49         old_outer: &mut WandOldOuter<'vir>,
   50     ) -> EncodeResult<'vir, (vir::Wand<'vir>, Resources<'vir>), E> {
   51         let inputs = edge.inputs(self.pcg_ctxt());
   52         let outputs = edge.outputs();
   53         let mut rhs = Vec::new();
   54         for i in inputs {
   55             let mut resources = Vec::new();
   56             self.encode_pcg_node(&*i, &mut resources, old_outer)?;
   57             rhs.push((*i, resources));
   58         }
   59         let wand_rhs = rhs
   60             .iter()
   61             .flat_map(|(_, resources)| resources.iter().copied())
   62             .collect::<Vec<_>>();
   63         let mut wand_lhs = Vec::new();
   64         for i in outputs {
   65             let i = i.expect_lifetime_projection();
   66             let exprs = self.encode_lifetime_projection(i, old_outer)?;
   67             wand_lhs.extend(exprs);
   68         }
   69         let wand = self.vcx.mk_wand(
   70             self.vcx.mk_conj(self.vcx.alloc_slice(&wand_lhs)),
   71             self.vcx.mk_conj(self.vcx.alloc_slice(&wand_rhs)),
   72         );
   73         Ok((wand, rhs))
   74     }
   75 }
