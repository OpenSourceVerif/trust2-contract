   42 impl<'vir, E: TaskEncoder> ImpureEncVisitor<'vir, '_, E> {
   43     pub fn package_wands(
   44         &mut self,
   45         final_borrow_state: &BorrowsState<'_, 'vir>,
   46     ) -> EncodeResult<'vir, Vec<vir::Stmt<'vir>>, E> {
   47         let mut wand_packages = Vec::new();
   48         let label = self.new_label("package_post");
   49         let result = self.local_defs[mir::RETURN_PLACE].impure_snap;
   50         let result = self.vcx.mk_local_labelled_old_expr(result, label);
   51         let args = self
   52             .local_defs
   53             .args()
   54             .map(|a| self.vcx.mk_old_expr(a.impure_snap));
   55         let args = PledgeExpr::pledge_args(result, args);
   56 
   57         for wand_data in self.wands.viper_wands() {
   58             let Some(wand) = self
   59                 .wands
   60                 .mk_wand(&wand_data, args, None, None, self.vcx, self.deps)
   61             else {
   62                 continue;
   63             };
   64             let rhs = wand_data
   65                 .rhs
   66                 .iter()
   67                 .map(|g| {
   68                     let resources = self.wands.encode_predicates_for_function_shape_node(
   69                         self.vcx,
   70                         self.deps,
   71                         *g,
   72                         None,
   73                         |i| args[i],
   74                     );
   75                     (
   76                         mir::Place::from(g.mir_local()),
   77                         resources.into_iter().collect(),
   78                     )
   79                 })
   80                 .collect();
   81             let mut package_script = self.pcs_package_script(final_borrow_state, rhs, label)?;
   82 
   83             for EncodedPledge {
   84                 expiry_postcondition,
   85                 ..
   86             } in &wand_data.pledges
   87             {
   88                 let span = expiry_postcondition.span();
   89                 self.vcx.with_span(span, |vcx| {
   90                     vcx.handle_error("exhale.failed:assertion.false", move |_| {
   91                         Some(vec![PrustiError::verification(
   92                             "pledge in the postcondition might not hold.",
   93                             span.into(),
   94                         )])
   95                     });
   96                     package_script.push(vcx.mk_exhale_stmt(expiry_postcondition.expr(args)));
   97                 });
   98             }
   99             wand_packages.push(
  100                 self.vcx
  101                     .mk_package_stmt(wand, self.vcx.alloc_slice(&package_script)),
  102             );
  103         }
  104         Ok(wand_packages)
  105     }
  106 }

  110 /// Not tied to a caller or callee context. `indirect_pres`, `indirect_posts`,
  111 /// `wand_posts`, and `package_wands` are identity-substituted and intended for
  112 /// use in the callee's own contract; `call_wands` is for caller use and
  113 /// re-substitutes via a [`WandCallContext`].
  114 #[derive(Clone)]
  115 pub struct WandEncOutput<'vir> {
  116     /// Information about the corresponding function.
  117     function_data: FunctionData<'vir>,
  118 
  119     /// The lifetime projections of all arguments to the function.
  120     inputs: Vec<FunctionShapeInput<Generalized>>,
  121 
  122     /// The lifetime projections of all function outputs (according to the
  123     /// corresponding [`FunctionShape`]). This *includes* lifetime projections
  124     /// of nested lifetimes in the function arguments.
  125     outputs: Vec<FunctionShapeOutput<Generalized>>,
  126 
  127     /// Encoded VIR expressions for the magic wands.
  128     wands: Vec<WandData<'vir>>,
  129 }

  131 /// Substitution context for instantiating a wand at a call site. When `None`,
  132 /// the wand is encoded using the callee's identity substitution (appropriate
  133 /// when emitting wands inside the function being defined). When `Some`, the
  134 /// wand is re-encoded with the call-site substitutions and the caller's
  135 /// generic parameters, so that placeholders like `Self` or other callee
  136 /// generics are replaced by concrete types from the caller's perspective.
  137 pub type WandCallContext<'vir> = Option<GArgs<'vir>>;

  139 impl<'vir> WandEncOutput<'vir> {
  140     pub(crate) fn fn_sig(
  141         &self,
  142         vcx: &'vir vir::VirCtxt<'vir>,
  143         call_ctx: WandCallContext<'vir>,
  144     ) -> ty::FnSig<'vir> {
  145         match call_ctx {
  146             // TODO: change pcg's `fn_sig` to take `&[GenericArg]` instead of `GenericArgsRef`
  147             Some(ctx) => self
  148                 .function_data
  149                 .fn_sig(vcx.tcx(), vcx.tcx().mk_args(ctx.args())),
  150             None => self.function_data.identity_fn_sig(vcx.tcx()),
  151         }
  152     }
  153 
  154     pub(crate) fn g_params(
  155         &self,
  156         vcx: &'vir vir::VirCtxt<'vir>,
  157         call_ctx: WandCallContext<'vir>,
  158     ) -> GParams<'vir> {
  159         match call_ctx {
  160             Some(ctx) => ctx.context(),
  161             None => GParams::new(
  162                 self.function_data.identity_substs(vcx.tcx()),
  163                 self.function_data.param_env(vcx.tcx()),
  164             ),
  165         }
  166     }

  168     /// The (unreified) predicates associated with the given node, or `None` if
  169     /// there are no resources associated with it.
  170     #[allow(clippy::type_complexity)]
  171     fn predicates_for_function_shape_node(
  172         &self,
  173         vcx: &'vir vir::VirCtxt<'vir>,
  174         deps: &mut TaskEncoderDependencies<'vir, impl TaskEncoder>,
  175         g: FunctionShapeNode<Generalized>,
  176         call_ctx: WandCallContext<'vir>,
  177     ) -> Option<Vec<vir::ExprGenBool<'vir, vir::ExprSnap<'vir>, vir::ExprKind<'vir>>>> {
  178         let arg_ty = g.ty(self.fn_sig(vcx, call_ctx));
  179         let decomp = RustTyDecomposition::from_ty(arg_ty, self.g_params(vcx, call_ctx));
  180         let region_proj =
  181             projection_for_generalized_idx(arg_ty, g.region_idx(), decomp, vcx.tcx())?;
  182         let predicates = deps
  183             .require_dep::<IndirectPredicatesEnc>(region_proj)
  184             .unwrap()
  185             .predicate_applications;
  186         (!predicates.is_empty()).then_some(predicates)
  187     }
  188 
  189     fn encode_predicates_for_function_shape_node(
  190         &self,
  191         vcx: &'vir vir::VirCtxt<'vir>,
  192         deps: &mut TaskEncoderDependencies<'vir, impl TaskEncoder>,
  193         g: impl Into<FunctionShapeNode<Generalized>>,
  194         call_ctx: WandCallContext<'vir>,
  195         mut snap: impl FnMut(mir::Local) -> vir::ExprSnap<'vir>,
  196     ) -> Option<vir::ExprBool<'vir>> {
  197         use vir::Reify;
  198         let g = g.into();
  199         let predicates = self.predicates_for_function_shape_node(vcx, deps, g, call_ctx)?;
  200 
  201         let local = g.mir_local();
  202         let local_snap = snap(local);
  203         Some(
  204             vcx.mk_conj(
  205                 &predicates
  206                     .iter()
  207                     .map(|p| p.reify(vcx, local_snap))
  208                     .collect::<Vec<_>>(),
  209             ),
  210         )
  211     }

  213     pub fn indirect_pres<'a, E: TaskEncoder>(
  214         &'a self,
  215         vcx: &'vir vir::VirCtxt<'vir>,
  216         local_defs: &'a MirLocalDefEncOutput<'vir>,
  217         deps: &'a mut TaskEncoderDependencies<'vir, E>,
  218     ) -> impl Iterator<Item = vir::ExprBool<'vir>> + 'a {
  219         self.inputs().filter_map(|g| {
  220             self.encode_predicates_for_function_shape_node(vcx, deps, g, None, |i| {
  221                 local_defs[i].impure_snap
  222             })
  223         })
  224     }

  226     pub fn indirect_posts<'a, E: TaskEncoder>(
  227         &'a self,
  228         vcx: &'vir vir::VirCtxt<'vir>,
  229         local_defs: &'a MirLocalDefEncOutput<'vir>,
  230         deps: &'a mut TaskEncoderDependencies<'vir, E>,
  231     ) -> impl Iterator<Item = vir::ExprBool<'vir>> + 'a {
  232         // The encoded predicates for the input lifetime projections that are
  233         // not blocked by any of the result lifetime projections. These will be
  234         // encoded as part of the postcondition of the function (in contrast,
  235         // the predicates for the blocked inputs will appear on the right-hand
  236         // side of a magic wand in the postcondition).
  237         let unblocked_input_posts = self
  238             .inputs()
  239             .filter(|i| !self.blocked_inputs().contains(i))
  240             .filter_map(|lp| {
  241                 self.encode_predicates_for_function_shape_node(vcx, deps, lp, None, |i| {
  242                     vcx.mk_old_expr(local_defs[i].impure_snap)
  243                 })
  244             })
  245             .collect::<Vec<_>>()
  246             .into_iter();
  247 
  248         let output_posts = self.outputs().filter_map(|g| {
  249             self.encode_predicates_for_function_shape_node(vcx, deps, g, None, |i| {
  250                 local_defs[i].impure_snap
  251             })
  252         });
  253         unblocked_input_posts.chain(output_posts)
  254     }

  256     pub fn wand_posts<'a, E: TaskEncoder>(
  257         &'a self,
  258         vcx: &'vir vir::VirCtxt<'vir>,
  259         local_defs: &'a MirLocalDefEncOutput<'vir>,
  260         deps: &'a mut TaskEncoderDependencies<'vir, E>,
  261     ) -> impl Iterator<Item = vir::ExprBool<'vir>> + 'a {
  262         let wand_result =
  263             vcx.mk_local_decl("wand_result", local_defs[mir::RETURN_PLACE].local_snap.ty());
  264         let wand_result_expr = vcx.mk_local_ex(wand_result);
  265         let args = local_defs
  266             .args()
  267             .map(|arg| vcx.mk_old_expr(arg.impure_snap));
  268         let args = PledgeExpr::pledge_args(wand_result_expr, args);
  269 
  270         // TODO: wands for late-bound regions
  271         self.viper_wands().into_iter().filter_map(move |wand_data| {
  272             let wand = self.mk_wand(&wand_data, args, None, None, vcx, deps)?;
  273             Some(vcx.mk_let_expr(
  274                 wand_result,
  275                 local_defs[mir::RETURN_PLACE].impure_snap,
  276                 vcx.mk_wand_expr(wand),
  277             ))
  278         })
  279     }

  281     pub fn call_wands<E: TaskEncoder>(
  282         &self,
  283         arguments: &[vir::ExprSnap<'vir>],
  284         label_pre: &'vir str,
  285         label_post: &'vir str,
  286         call_ctx: GArgs<'vir>,
  287         visitor: &mut ImpureEncVisitor<'vir, '_, E>,
  288     ) -> Vec<vir::Wand<'vir>> {
  289         let result = visitor
  290             .vcx
  291             .mk_local_labelled_old_expr(arguments[mir::RETURN_PLACE.as_usize()], label_post);
  292         let args = (1..arguments.len()).map(|l| {
  293             visitor
  294                 .vcx
  295                 .mk_local_labelled_old_expr(arguments[l], label_pre)
  296         });
  297         let args = PledgeExpr::pledge_args(result, args);
  298         let mut wands = Vec::new();
  299         for wand_data in self.viper_wands() {
  300             let Some(wand) = self.mk_wand(
  301                 &wand_data,
  302                 args,
  303                 Some(label_pre),
  304                 Some(call_ctx),
  305                 visitor.vcx,
  306                 visitor.deps,
  307             ) else {
  308                 continue;
  309             };
  310             wands.push(wand);
  311         }
  312         wands
  313     }

  315     fn mk_wand<E: TaskEncoder>(
  316         &self,
  317         wand_data: &WandData<'vir>,
  318         pledge_args: PledgeArgs<'vir>,
  319         pledge_old_label: Option<&'vir str>,
  320         call_ctx: WandCallContext<'vir>,
  321         vcx: &'vir vir::VirCtxt<'vir>,
  322         deps: &mut TaskEncoderDependencies<'vir, E>,
  323     ) -> Option<vir::Wand<'vir>> {
  324         debug_assert!(!wand_data.lhs.is_empty());
  325         let generics = self.generics_subst(call_ctx, vcx, deps);
  326         let pledge_expr = |pledge: &PledgeExpr<'vir>| match pledge_old_label {
  327             Some(label) => {
  328                 vcx.with_local_subst(generics, || pledge.expr_at_label(pledge_args, label))
  329             }
  330             None => pledge.expr(pledge_args),
  331         };
  332         let rhs = wand_data.rhs.iter().filter_map(|g| {
  333             self.encode_predicates_for_function_shape_node(vcx, deps, *g, call_ctx, |i| {
  334                 pledge_args[i]
  335             })
  336         });
  337         let rhs = rhs
  338             .chain(
  339                 wand_data
  340                     .pledges
  341                     .iter()
  342                     .map(|pledge| pledge_expr(&pledge.expiry_postcondition)),
  343             )
  344             .collect::<Vec<_>>();
  345         if rhs.is_empty() {
  346             // We skip emitting the wand when there is nothing on the RHS, i.e.,
  347             // nothing would be unblocked by applying this wand, nor are there
  348             // any pledge postconditions.
  349             return None;
  350         }
  351         let rhs = vcx.mk_conj(&rhs);
  352         let lhs = wand_data.lhs.iter().filter_map(|g| {
  353             self.encode_predicates_for_function_shape_node(vcx, deps, *g, call_ctx, |i| {
  354                 pledge_args[i]
  355             })
  356         });
  357         let lhs = lhs
  358             .chain(
  359                 wand_data
  360                     .pledges
  361                     .iter()
  362                     .filter_map(|pledge| pledge.expiry_obligation.as_ref().map(pledge_expr)),
  363             )
  364             .collect::<Vec<_>>();
  365         let lhs = vcx.mk_conj(&lhs);
  366         Some(vcx.mk_wand(lhs, rhs))
  367     }

  369     /// The pledges are encoded once, at the callee's identity substitution,
  370     /// so they refer to the callee's generic parameters. At a call site, these
  371     /// are replaced by the call's generic arguments, as Viper does for the
  372     /// callee's postcondition (from which the caller holds the wand).
  373     fn generics_subst<E: TaskEncoder>(
  374         &self,
  375         call_ctx: WandCallContext<'vir>,
  376         vcx: &'vir vir::VirCtxt<'vir>,
  377         deps: &mut TaskEncoderDependencies<'vir, E>,
  378     ) -> &'vir FxHashMap<&'vir str, vir::ExprDyn<'vir>> {
  379         let Some(call_args) = call_ctx else {
  380             return vcx.alloc(FxHashMap::default());
  381         };
  382         let params = deps
  383             .require_dep::<GenericParamsEnc>(self.g_params(vcx, None))
  384             .unwrap();
  385         let args = deps.require_dep::<GArgsTyEnc>(call_args).unwrap();
  386         let tys = params
  387             .ty_decls()
  388             .iter()
  389             .zip(args.get_ty::<(), !>())
  390             .map(|(decl, arg)| (decl.name, arg.as_dyn()));
  391         let consts = params
  392             .const_decls()
  393             .iter()
  394             .zip(args.get_const::<(), !>())
  395             .map(|(decl, arg)| (decl.name, arg.as_dyn()));
  396         vcx.alloc(tys.chain(consts).collect())
  397     }
  398 }

  447     type OutputFullDependency<'vir> = WandEncOutput<'vir>;
  448 
  449     const ENCODER_NAME: &'static str = "wand encoder";
  450 
  451     fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
  452         task.clone()
  453     }
  454 
  455     fn do_encode_full<'vir>(
  456         task_key: &Self::TaskKey<'vir>,
  457         deps: &mut TaskEncoderDependencies<'vir, Self>,
  458     ) -> EncodeFullResult<'vir, Self> {
  459         deps.emit_output_ref(task_key.clone(), ())?;
  460         vir::with_vcx(|vcx| {
  461             let def_id = task_key.def_id();
  462 
  463             let shape = task_key.function_shape(vcx).map_err(|e| {
  464                 EncodeFullError::unsupported(format!("function shape: {e:?}"), Vec::new())
  465             })?;
  466 
  467             let coupled_edges = shape.coupled_edges();
  468             let edges: FxHashSet<_> = shape.edges().map(|e| (e.input(), e.output())).collect();
  469 
  470             let (inputs, outputs) = shape.take_inputs_and_outputs();
  471             let spec = deps.require_dep::<MirSpecEnc>((def_id, def_id, MirSpecEncMode::Impure))?;
  472             if coupled_edges.is_empty() {
  473                 assert!(spec.pledges.is_empty());
  474                 return Ok((
  475                     (),
  476                     WandEncOutput {
  477                         function_data: task_key.data,
  478                         inputs,
  479                         outputs,
  480                         wands: vec![],
  481                     },
  482                 ));
  483             }
  484             let pledges = spec.pledges;
  485             let wands: Vec<WandData<'vir>> = coupled_edges
  486                 .into_iter()
  487                 .filter_map(|hyper_edge| {
  488                     let (sources, mut targets) = hyper_edge.into_tuple();
  489                     // We don't want to emit an identity wand, like P --* P. This can happen when
  490                     // PCG returns self-edges, like for fn(x: &'a mut &'b i32) where 'b is in
  491                     // invariant position and we therefore have an edge x|'b -> x|'b.
  492                     // Currently, these edges also prevent us from emitting indirect postconditions.
  493                     // TODO: we might want to emit these identity wands in the future to attach functional
  494                     // specifications to them. We still need to emit the resources on the wand's LHS.
  495                     let mut sources_as_nodes = sources
  496                         .iter()
  497                         .map(|&s| s.to_function_shape_node())
  498                         .collect::<Vec<_>>();
  499                     sources_as_nodes.sort();
  500                     targets.sort();
  501                     if sources_as_nodes == targets {
  502                         return None;
  503                     }
  504                     Some(WandData::new(targets, sources, pledges.clone()))
  505                 })
  506                 .collect();
  507             let mut output: WandEncOutput<'vir> = WandEncOutput {
  508                 function_data: task_key.data,
  509                 inputs,
  510                 outputs,
  511                 wands: Vec::new(),
  512             };
  513             output.wands = output
  514                 .select_wands(wands, !pledges.is_empty(), &edges, vcx, deps)
  515                 .map_err(|message| EncodeFullError::unsupported(message, Vec::new()))?;
  516             Ok(((), output))
  517         })
  518     }
  519 }

  521 impl<'vir> WandEncOutput<'vir> {
  522     /// Selects the wands to emit among those of the coupled edges, rejecting
  523     /// the shapes that cannot be encoded precisely. Pledges that cannot be
  524     /// attached to one wand are reported and left out.
  525     fn select_wands<E: TaskEncoder>(
  526         &self,
  527         wands: Vec<WandData<'vir>>,
  528         has_pledges: bool,
  529         edges: &FxHashSet<(WandRhsKey, WandLhsKey)>,
  530         vcx: &'vir vir::VirCtxt<'vir>,
  531         deps: &mut TaskEncoderDependencies<'vir, E>,
  532     ) -> Result<Vec<WandData<'vir>>, String> {
  533         let mut has_resources = |g: FunctionShapeNode<Generalized>| {
  534             self.predicates_for_function_shape_node(vcx, deps, g, None)
  535                 .is_some()
  536         };
  537         // A wand is only needed if it gives back a resource, or to carry the
  538         // pledges if none does (e.g. for shared references).
  539         let (mut wands, resourceless): (Vec<_>, Vec<_>) = wands
  540             .into_iter()
  541             .partition(|wand_data| wand_data.rhs.iter().any(|g| has_resources((*g).into())));
  542         // A wand gives back its sources only once all of its targets have
  543         // expired. That matches the signature only if each source with
  544         // resources flows into each target with resources.
  545         for wand_data in &wands {
  546             let sources = wand_data
  547                 .rhs
  548                 .iter()
  549                 .filter(|g| has_resources((**g).into()))
  550                 .collect::<Vec<_>>();
  551             let targets = wand_data
  552                 .lhs
  553                 .iter()
  554                 .filter(|g| has_resources(**g))
  555                 .collect::<Vec<_>>();
  556             let precise = sources.iter().all(|source| {
  557                 let mut targets = targets.iter();
  558                 targets.all(|target| edges.contains(&(**source, **target)))
  559             });
  560             if !precise {
  561                 return Err(
  562                     "borrows in the result that expire separately but depend on a common \
  563                      argument are not supported"
  564                         .to_string(),
  565                 );
  566             }
  567         }
  568         if has_pledges {
  569             if wands.is_empty() {
  570                 wands.extend(resourceless.into_iter().take(1));
  571             } else if wands.len() > 1 {
  572                 // It is unclear which expiry the pledges refer to. They are
  573                 // left out, each with its expiry obligation: then neither the
  574                 // function nor its callers rely on any of them, so both are
  575                 // still encoded. The encoding goes on, so the proof that the
  576                 // error was reported is not needed.
  577                 let span = wands[0].pledges[0].expiry_postcondition.span();
  578                 let _ = vcx.emit_early_error(
  579                     PrustiError::unsupported(
  580                         "pledges on a function whose result contains borrows that expire \
  581                          separately are not supported",
  582                         span.into(),
  583                     )
  584                     .add_note("the pledges of the function are skipped", None),
  585                 );
  586                 for wand in &mut wands {
  587                     wand.pledges.clear();
  588                 }
  589             }
  590         }
  591         Ok(wands)
  592     }
  593 
  594     pub fn viper_wands(&self) -> Vec<WandData<'vir>> {
  595         self.wands.clone()
  596     }
  597 
  598     /// All lifetime projections in the arguments that are blocked by any of the
  599     /// lifetime projections in the function's result.
  600     pub fn blocked_inputs(&self) -> FxHashSet<FunctionShapeInput<Generalized>> {
  601         self.wands
  602             .iter()
  603             .flat_map(|wand| wand.rhs.iter().copied())
  604             .collect()
  605     }
  606 
  607     pub fn inputs(&self) -> impl Iterator<Item = FunctionShapeInput<Generalized>> + '_ {
  608         self.inputs.iter().copied()
  609     }
  610 
  611     pub fn outputs(&self) -> impl Iterator<Item = FunctionShapeOutput<Generalized>> + '_ {
  612         self.outputs.iter().copied()
  613     }
