  222 struct EncodedRvalue<'vir> {
  223     /// A snapshot of the rvalue. This snapshot is guaranteed to be well-formed
  224     /// in the state *before* the Rvalue has been assigned to a place. For
  225     /// example, in the statement `let rx = &mut x`, the snapshot encoding of
  226     /// `&mut x` relies on capabilities to `x` as they are *before* the
  227     /// assignment occurs. This is because the encoding of the assignment itself
  228     /// requires a snapshot of `&mut x`, and the snapshot constructor for
  229     /// &mut expects a snapshot of the borrowed place which can only be created
  230     /// when the capabilities are in the pre-assign state.
  231     expr: vir::ExprSnap<'vir>,
  232 
  233     /// Additional statements necessary to obtain the predicate for the assigned
  234     /// place of this Rvalue *after* the assignment. Such folds are necessary if
  235     /// make_generic / make_concrete type casts are necessary to move the capabilities
  236     /// from the state before the assignment to the predicate of the assigned place.
  237     ///
  238     /// For example, for the statement `let rx: &mut u32 = &mut x`, this should
  239     /// be called with the place corresponding to `rx`, and it will fold the
  240     /// type predicate for `rx` (in this case, this will make the `u32`
  241     /// permission held conceptually in *rx generic by calling
  242     /// `make_generic_u32(*rx)`
  243     ///
  244     /// Note that after executing these statements, the snapshot in `expr` is
  245     /// no-longer well-formed.
  246     post_assign_folds: Option<Box<dyn FnOnce(vir::ExprRef<'vir>) -> Vec<vir::Stmt<'vir>> + 'vir>>,
  247 }
  248 
  249 impl<'vir> EncodedRvalue<'vir> {
  250     fn post_fold_stmts(self, lhs_place: vir::ExprRef<'vir>) -> Vec<vir::Stmt<'vir>> {
  251         match self.post_assign_folds {
  252             Some(f) => f(lhs_place),
  253             None => Vec::new(),
  254         }
  255     }
  256 }

  301     fn encode_rvalue(
  302         &mut self,
  303         rvalue: &mir::Rvalue<'vir>,
  304         span: Span,
  305         operand_snaps: &OperandSnaps<'vir>,
  306     ) -> EncodeResult<'vir, EncodedRvalue<'vir>, E> {
  307         let rvalue_ty = rvalue.ty(self.local_decls, self.vcx.tcx());
  308         match rvalue {
  309             mir::Rvalue::Use(op, _) => Ok(self.encode_operand_snap(op, operand_snaps)?.into()),
  310             mir::Rvalue::Cast(cast_kind, operand, ty) => {
  311                 assert_eq!(*ty, rvalue_ty);
  312                 let (stmt, cast) =
  313                     self.encode_cast_snap(rvalue_ty, *cast_kind, operand, operand_snaps)?;
  314                 self.stmts(stmt);
  315                 Ok(cast.into())
  316             }
  317             mir::Rvalue::BinaryOp(op, box (l, r)) => Ok(self
  318                 .encode_binop_snap(rvalue_ty, *op, l, r, operand_snaps, span)?
  319                 .into()),
  320 
  321             mir::Rvalue::UnaryOp(unop, operand) => Ok(self
  322                 .encode_unary_op_snap(rvalue_ty, *unop, operand, operand_snaps)?
  323                 .into()),
  324 
  325             mir::Rvalue::Aggregate(box mir::AggregateKind::Array(..), elements) => Ok(self
  326                 .encode_array_snap(rvalue_ty, elements, operand_snaps)?
  327                 .into()),
  328 
  329             mir::Rvalue::Aggregate(
  330                 box kind @ (mir::AggregateKind::Adt(..)
  331                 | mir::AggregateKind::Tuple
  332                 | mir::AggregateKind::Closure(..)),
  333                 fields,
  334             ) => Ok(self
  335                 .encode_aggregate_snap(rvalue_ty, kind, fields, operand_snaps)?
  336                 .into()),
  337 
  338             mir::Rvalue::Repeat(operand, len) => Ok(self
  339                 .encode_repeat_snap(rvalue_ty, operand, *len, operand_snaps)?
  340                 .into()),
  341 
  342             mir::Rvalue::Discriminant(place) => {
  343                 let e_rvalue_ty = self.ty_use_pure(rvalue_ty);
  344                 let place_ty = place.ty(self.local_decls, self.vcx.tcx());
  345                 let ty = self.ty_use_impure(place_ty.ty);
  346                 if ty.specifics.is_opaque() {
  347                     return Err(unsupported_opaque(place_ty.ty, span));
  348                 }
  349                 let place_expr = self.encode_place(Place::from(*place))?.expr;
  350 
  351                 Ok(match ty
  352                     .get_enumlike()
  353                     .filter(|_| place_ty.variant_index.is_none())
  354                 {
  355                     Some(el) => {
  356                         if let Some(snap) = place_expr.snap {
  357                             let ty = self.ty_use_pure(place_ty.ty).expect_enumlike();
  358                             ty.snap_to_discr_snap(snap.downcast_ty())
  359                         } else {
  360                             let place_expr = place_expr.expect_predicate();
  361                             self.vcx.mk_unfolding_expr(
  362                                 ty.ref_to_pred_app(place_expr, Some(self.vcx.mk_wildcard())),
  363                                 el.discr_ty()
  364                                     .ref_to_snap(el.discr(place_expr))
  365                                     .downcast_ty(),
  366                             )
  367                         }
  368                     }
  369                     None => {
  370                         // mir::Rvalue::Discriminant documents "Returns zero for types without discriminant"
  371                         let zero = self.vcx.mk_uint::<0>();
  372                         e_rvalue_ty
  373                             .expect_primitive()
  374                             .prim_to_snap(zero.upcast_ty())
  375                     }
  376                 }
  377                 .upcast_ty()
  378                 .into())
  379             }
  380 
  381             mir::Rvalue::Ref(_reg, _kind, place) => Ok(match rvalue_ty.kind() {
  382                 TyKind::Ref(.., ty::Mutability::Not) => {
  383                     let (address, snap, _, _) = self.encode_place_with_snap((*place).into())?;
  384                     let metadata = address.expr.metadata;
  385                     let metadata =
  386                         metadata.unwrap_or_else(|| self.expect_thin_ptr_metadata(rvalue_ty));
  387                     let inner = self.ty_use_pure(rvalue_ty).expect_immref();
  388                     inner
  389                         .prim_to_snap(address.expr.address, metadata, snap)
  390                         .upcast_ty()
  391                         .into()
  392                 }
  393                 TyKind::Ref(.., ty::Mutability::Mut) => {
  394                     let p_rvalue_ty = self.ty_use_impure(rvalue_ty);
  395                     let place_expr = self.encode_place(Place::from(*place))?;
  396 
  397                     let metadata = place_expr.expr.metadata;
  398                     let metadata =
  399                         metadata.unwrap_or_else(|| self.expect_thin_ptr_metadata(rvalue_ty));
  400                     let inner = p_rvalue_ty.expect_mutref();
  401                     let place_ref = place_expr.expr.expect_predicate();
  402                     EncodedRvalue {
  403                         expr: inner.prim_to_snap_assign(place_ref, metadata).upcast_ty(),
  404                         post_assign_folds: Some(Box::new(move |lhs_place| {
  405                             p_rvalue_ty.fold(None, lhs_place, None, None, None)
  406                         })),
  407                     }
  408                 }
  409                 _ => unreachable!(),
  410             }),
  411 
  412             mir::Rvalue::RawPtr(_, place) => {
  413                 let place_expr = self.encode_place(Place::from(*place))?;
  414                 let metadata = place_expr.expr.metadata;
  415                 let metadata = metadata.unwrap_or_else(|| self.expect_thin_ptr_metadata(rvalue_ty));
  416                 let raw = self.ty_use_pure(rvalue_ty).expect_raw();
  417                 Ok(raw
  418                     .prim_to_snap(place_expr.expr.address, metadata)
  419                     .upcast_ty()
  420                     .into())
  421             }
  422 
  423             _ => Err(EncodeFullError::unsupported(
  424                 format!("unsupported rvalue {rvalue:?}"),
  425                 vec![span],
  426             )),
  427         }
  428     }

  445     /// Emit the repacks of the terminator edge to `target` ([Self::pcs_succ]).
  446     /// The succ is found by block: the target is usually the terminator's
  447     /// only non-unwind successor, but for a ghost switch the (second)
  448     /// `ghost_erased` successor is skipped.
  449     fn pcs_succ_to(&mut self, target: mir::BasicBlock) -> EncodeResult<'vir, (), E> {
  450         let current_fpcs = self.current_fpcs.take().unwrap();
  451         let cfpcs = current_fpcs.statements.last().unwrap();
  452         let succ = current_fpcs
  453             .terminator
  454             .succs
  455             .iter()
  456             .find(|succ| succ.block() == target)
  457             .unwrap();
  458         let res = self.pcs_succ(cfpcs, succ);
  459         self.current_fpcs = Some(current_fpcs);
  460         res
  461     }

  507                     .unwrap();

  521                 else {
  522                     todo!("from-end `ConstantIndex` (into a slice)")

  524                 Some(self.usize_const_snap(offset))

  526             _ => None,

  545         };

  547         // We don't want to unfold because for immutable refs we only use snapshot read/writes
  548         if place.is_shared_ref(self.pcg_ctxt()) || place.projects_shared_ref(self.pcg_ctxt()) {
  549             return Ok(None);
  550         }
  551 
  552         let ref_p = self.encode_place(place)?;
  553 
  554         let place_ty = ref_p.ty;
  555         let ref_p = self
  556             .vcx
  557             .maybe_apply_label(ref_p.expr.expect_predicate(), label);
  558         let data = self.ty_use_impure(place_ty.ty);
  559         if data.specifics.is_opaque() {
  560             return Err(unsupported_opaque(place_ty.ty, self.current_span()));
  561         }
  562 
  563         let index = expansion
  564             .and_then(|expansion| expansion.guide())
  565             .and_then(|guide| self.expansion_index(guide));
  566 
  567         let (stmts, frame) = match fold_or_unfold {
  568             FoldOrUnfold::Unfold => (
  569                 data.unfold(place_ty.variant_index, ref_p, index, None, label),
  570                 Vec::new(),
  571             ),
  572             FoldOrUnfold::Fold => (
  573                 data.fold(place_ty.variant_index, ref_p, index, None, label),
  574                 data.fold_frame(place_ty.variant_index, ref_p, index),
  575             ),
  576         };
  577         self.stmts(stmts);
  578         Ok(Some(frame))
  579     }

  581     fn branch_choices_cond(&mut self, choices: &BranchChoices) -> vir::ExprBool<'vir> {
  582         let successors = choices.successors(self.body);
  583         let from = choices.from();
  584         let conj = successors
  585             .iter()
  586             .map(|to| self.from_to_vars.get_or_create(self.vcx, from, *to).expr)
  587             .collect::<Vec<_>>();
  588         // Control flow must continue from `choices.from()` to any one of the `successors`
  589         self.vcx.mk_disj(self.vcx.alloc_slice(&conj))
  590     }

  592     /// The conjuncts of the condition under which `edge` is valid.
  593     fn edge_condition(&mut self, edge: &BorrowPcgEdge<'vir>) -> Vec<vir::ExprBool<'vir>> {
  594         let conditions = edge.conditions();
  595 
  596         // For each block `b` where the edge is only valid if control flow
  597         // continues from `b` to a specified subset of its successors, `cond`
  598         // contains the corresponding VIR expression.
  599         conditions
  600             .all_branch_choices()
  601             .map(|choices| self.branch_choices_cond(choices))
  602             .collect()
  603     }

  605     /// Handles the edge under its condition, also for the parts of the
  606     /// footprint it adds. A `footprint` is given in a package script.
  607     fn pcs_handle_edge(
  608         &mut self,
  609         borrows_state: &BorrowsState<'_, 'vir>,
  610         edge: &BorrowPcgEdge<'vir>,
  611         edge_action: EdgeAction,
  612         edge_to_loop: bool,
  613         to_skip: &mut Vec<mir::BasicBlock>,
  614         mut footprint: Option<&mut Footprint<'_, 'vir>>,
  615     ) -> EncodeResult<'vir, (), E> {
  616         let cond_conjuncts = self.edge_condition(edge);
  617         // For each block `b` where the edge validity depends on the successor taken from `b`,
  618         // every successor must be valid.
  619         let cond = self.vcx.mk_conj(self.vcx.alloc_slice(&cond_conjuncts));
  620         let parts = footprint
  621             .as_ref()
  622             .map_or(0, |footprint| footprint.parts.len());
  623         let stmts = self.block(|self_| {
  624             self_.pcs_handle_edge_conditionless(
  625                 borrows_state,
  626                 edge,
  627                 edge_action,
  628                 edge_to_loop,
  629                 to_skip,
  630                 footprint.as_deref_mut(),
  631             )
  632         })?;
  633         if let Some(footprint) = footprint
  634             && !cond_conjuncts.is_empty()
  635             && footprint.parts.len() > parts
  636         {
  637             let added = self.vcx.mk_conj(&footprint.parts.split_off(parts));
  638             let guarded = self
  639                 .vcx
  640                 .mk_bin_op_expr(vir::BinOpKind::Implies, cond, added);
  641             footprint.parts.push(guarded.downcast_ty());
  642         }
  643         if stmts.is_empty()
  644             || stmts
  645                 .iter()
  646                 .all(|stmt| matches!(stmt.kind, vir::StmtKindData::Comment(_)))
  647             || cond_conjuncts.is_empty()
  648         {
  649             self.stmts(stmts);
  650             return Ok(());
  651         }
  652         let stmts = self.vcx.alloc_slice(&stmts);
  653         self.stmt(self.vcx.mk_if_stmt(cond, stmts, &[]));
  654         Ok(())
  655     }

  657     fn pcs_handle_edge_conditionless(
  658         &mut self,
  659         borrows_state: &BorrowsState<'_, 'vir>,
  660         edge: &BorrowPcgEdge<'vir>,
  661         edge_action: EdgeAction,
  662         edge_to_loop: bool,
  663         to_skip: &mut Vec<mir::BasicBlock>,
  664         mut footprint: Option<&mut Footprint<'_, 'vir>>,
  665     ) -> EncodeResult<'vir, (), E> {
  666         // In a package script, current places are read in the state before the
  667         // package.
  668         let label = footprint.as_ref().map(|footprint| footprint.label);
  669         match edge.kind() {
  670             BorrowPcgEdgeKind::Borrow(borrow) if borrow.is_mut() && edge_action.is_remove() => {
  671                 // For a borrow e.g. let x = &mut y; the capability to `y` is
  672                 // folded into the Rvalue `&mut y` that is stored in `x`. This
  673                 // reverses that effect
  674                 self.unfold(borrow.assigned_ref(), label)?;
  675             }
  676             BorrowPcgEdgeKind::BorrowFlow(borrow_flow)
  677                 if let BorrowFlowEdgeKind::Assignment(assignment_data) = borrow_flow.kind()
  678                     && let Some(mir::CastKind::PointerCoercion(
  679                         ty::adjustment::PointerCoercion::Unsize,
  680                         _,
  681                     )) = assignment_data.cast_kind()
  682                     && edge_action.is_remove() =>
  683             {
  684                 let kind = assignment_data.cast_kind().unwrap();
  685 
  686                 // For an unsize operation `let slice = &mut array;` the PCG
  687                 // will keep track of the connection between the two places;
  688                 // during the unsize operation we call a method to transter
  689                 // permissions from one to the other, when the slice expires,
  690                 // we need to undo the unsize operation.
  691                 let long = borrow_flow.long();
  692                 let short = borrow_flow.short();
  693                 let PlaceOrConst::Place(src) = long.base() else {
  694                     unreachable!();
  695                 };
  696                 let src = src.as_local_place().unwrap();
  697                 let dst = short.base();
  698 
  699                 let ctxt = CompilerCtxt::new(self.body, self.vcx.tcx(), ());
  700                 let src_ty = src.ty(ctxt).ty;
  701                 let dst_ty = dst.ty(ctxt).ty;
  702                 // An undo is only needed when processing coercions `&mut T -> &mut U` when T: Unsize<U>:
  703                 // The mutable borrow ends and we need to return the permission to the original
  704                 // `&mut T`. For every other CoerceUnsized destination (`&T`, raw pointers,
  705                 // `Box<T>`, `Rc<T>`, `Arc<T>`, ...), there is no mutable borrow expiring and
  706                 // nothing to undo. (For shared references the slice cannot have modified the
  707                 // array it unsized from, so there is likewise nothing to undo.)
  708                 if !matches!(dst_ty.kind(), ty::TyKind::Ref(_, _, ty::Mutability::Mut)) {
  709                     return Ok(());
  710                 }
  711                 // Since we only undo unsize coercions for &mut T -> &mut U when T: Unsize<U>,
  712                 // we only consider the edge with idx=0 on both src and dst. This
  713                 // corresponds to the lifetime projections of the mutable references.
  714                 // The single undo accounts for the entire unsize coercion.
  715                 // All other BorrowFlow(Unsize) edges in the PCG are bookkeeping, for example:
  716                 //   - same-index (e.g. `&mut [&mut T; N] -> &mut [&mut T]`
  717                 //     emits a redundant idx=1 -> idx=1 edge
  718                 //   - cross-index (e.g. `&mut T -> &mut dyn Trait + 'b`
  719                 //     emits an idx=0 -> idx=1 edge).
  720                 if long.region_idx().index() != 0 || short.region_idx().index() != 0 {
  721                     return Ok(());
  722                 }
  723 
  724                 let rvalue_ty = RustTyDecomposition::from_ty(dst_ty, self.context());
  725                 let operand_ty = RustTyDecomposition::from_ty(src_ty, self.context());
  726                 let cast_output =
  727                     self.deps()
  728                         .require_dep::<MirBuiltinUseCastEnc>(MirBuiltinUseCastTask::new(
  729                             rvalue_ty, kind, operand_ty,
  730                         ))?;
  731                 let src_place = src.place();
  732                 let src_label = if let MaybeLabelledPlace::Labelled(snap) = src {
  733                     Some(self.get_location_label(snap.at()))
  734                 } else {
  735                     label.map(vir::OldLabel::Label)
  736                 };
  737                 let src_snap = self.encode_place_with_snap(src_place)?.1.downcast_ty();
  738                 let src_enc = self.vcx.maybe_apply_label(src_snap, src_label);
  739                 let undo = cast_output.undo(src_enc);
  740                 self.stmts(undo);
  741                 return Ok(());
  742             }
  743             BorrowPcgEdgeKind::Deref(deref) => {
  744                 let frame = self.fold_or_unfold(
  745                     deref.blocked_place(),
  746                     FoldOrUnfold::for_action(edge_action),
  747                     None,
  748                     label,
  749                 )?;
  750                 if let Some(footprint) = footprint
  751                     && let Some(frame) = frame
  752                 {
  753                     self.fold_footprint(footprint, edge, frame)?;
  754                 }
  755             }
  756             // Ignore expansions of lifetime projections for now
  757             BorrowPcgEdgeKind::BorrowPcgExpansion(expansion)
  758                 if let PcgNode::Place(base) = expansion.base() =>
  759             {
  760                 let frame = self.fold_or_unfold(
  761                     base,
  762                     FoldOrUnfold::for_action(edge_action),
  763                     Some(expansion),
  764                     label,
  765                 )?;
  766                 if let Some(footprint) = footprint
  767                     && let Some(frame) = frame
  768                 {
  769                     self.fold_footprint(footprint, edge, frame)?;
  770                 }
  771             }
  772             BorrowPcgEdgeKind::Coupled(PcgCoupledEdgeKind(FunctionCallOrLoop::FunctionCall(
  773                 call_edge,
  774             ))) => {
  775                 if edge_action.is_add() {
  776                     // The wand will be introduced by the method call itself.
  777                     return Ok(());
  778                 }
  779                 let call = call_edge.metadata();
  780                 // We may be encoding multiple edges as a single wand, skip
  781                 // further edge removals. This is a hack to get around the fact
  782                 // that Viper doesn't support hyperwands.
  783                 if to_skip.contains(&call.location().block) {
  784                     return Ok(());
  785                 }
  786                 to_skip.push(call.location().block);
  787                 // TODO: this applies *all* the wands for the referenced
  788                 //   function call; instead we should figure out which
  789                 //   wand it is based on the edge info.
  790                 // TODO: closures
  791                 let wands = self.deps.require_dep::<WandEnc>(WandEncTask {
  792                     data: call.function_data().unwrap(),
  793                 })?;
  794                 let bb = &self.body[call.location().block];
  795                 let terminator = bb.terminator.as_ref().unwrap();
  796                 match &terminator.kind {
  797                     mir::TerminatorKind::Call {
  798                         func,
  799                         args,
  800                         destination,
  801                         ..
  802                     } => {
  803                         // Calls not encoded as impure method calls create no
  804                         // wands (e.g., a Viper function application).
  805                         if self.wandless_calls.contains(&call.location().block) {
  806                             return Ok(());
  807                         }
  808                         let (_, caller_substs) = self.get_call_data(func);
  809 
  810                         let (_, dest_snap, _, _) =
  811                             self.encode_place_with_snap((*destination).into())?;
  812                         let wand_args =
  813                             std::iter::once(Ok(dest_snap))
  814                                 .chain(args.iter().map(|operand| {
  815                                     self.encode_operand_snap_immediate(&operand.node)
  816                                 }))
  817                                 .collect::<EncodeResult<'vir, Vec<_>, E>>()?;
  818                         let (label_pre, label_post) = self.call_labels[&call.location().block];
  819                         let call_ctx = self.gargs(caller_substs);
  820                         let wands =
  821                             wands.call_wands(&wand_args, label_pre, label_post, call_ctx, self);
  822                         for wand in wands {
  823                             if let Some(footprint) = &mut footprint {
  824                                 footprint.parts.push(self.vcx.mk_wand_expr(wand));
  825                             }
  826                             self.stmt(self.vcx.mk_apply_stmt(wand));
  827                         }
  828                     }
  829                     _ => unreachable!(),
  830                 }
  831             }
  832             BorrowPcgEdgeKind::Abstraction(at @ AbstractionEdge::Loop(_)) => {
  833                 self.pcs_handle_wand(
  834                     borrows_state,
  835                     edge_action.is_add(),
  836                     &at.clone().into_singleton_coupled_edge(),
  837                     label,
  838                     edge_to_loop,
  839                     footprint,
  840                 )?;
  841             }
  842             other => comment!(self, "(ignoring) {edge_action:?} {other:?}"),
  843         }
  844         comment!(
  845             self,
  846             "(PCG) handled edge {edge_action:?}: {}",
  847             edge.to_short_string(self.pcg_ctxt())
  848         );
  849         Ok(())
  850     }

  872     /// The proof script of a package giving back the nodes of `rhs`, each with
  873     /// the resources it is given back as: unblocking the nodes, after asserting
  874     /// the footprint of the wand. Statements in the script only see resources
  875     /// in the package state, and a resource of the state outside is only moved
  876     /// there when consumed. Asserting the footprint first moves it there before
  877     /// any statement, which may need it earlier (e.g. to read the address of a
  878     /// `Box` that is folded).
  879     pub(crate) fn pcs_package_script<N: Into<BlockedNode<'vir>>>(
  880         &mut self,
  881         borrows_state: &BorrowsState<'_, 'vir>,
  882         rhs: Vec<(N, Vec<vir::ExprBool<'vir>>)>,
  883         label: &'vir str,
  884     ) -> EncodeResult<'vir, Vec<vir::Stmt<'vir>>, E> {
  885         let mut parts = Vec::new();
  886         let mut unblock = Vec::new();
  887         for (node, resources) in rhs {
  888             let node = node.into();
  889             let ug = UnblockGraph::for_node(node, borrows_state, self.pcg_ctxt());
  890             let actions = ug.actions(self.pcg_ctxt()).unwrap();
  891             let mut footprint = Footprint {
  892                 actions: &actions,
  893                 label,
  894                 parts: Vec::new(),
  895             };
  896             // Where nothing unblocks the node, it is given back as it is.
  897             if !resources.is_empty() {
  898                 let resources = self.vcx.mk_conj(&resources);
  899                 let part = self.unless_unblocked(&footprint, node, None, resources);
  900                 footprint.parts.extend(part);
  901             }
  902             unblock.extend(
  903                 self.block(|visitor| visitor.pcs_unblock_actions(borrows_state, &mut footprint))?,
  904             );
  905             parts.extend(footprint.parts);
  906         }
  907         let mut script = Vec::new();
  908         if !parts.is_empty() {
  909             script.push(self.vcx.mk_assert_stmt(self.vcx.mk_conj(&parts)));
  910         }
  911         script.extend(unblock);
  912         Ok(script)
  913     }

  915     /// Adds what folding along `edge` takes from the state outside the
  916     /// package: the `frame` of the fold (see `fold_or_unfold`), and the
  917     /// children of the edge, unless (or except under the conditions where) the
  918     /// footprint's actions unblock them.
  919     fn fold_footprint(
  920         &mut self,
  921         footprint: &mut Footprint<'_, 'vir>,
  922         edge: &BorrowPcgEdge<'vir>,
  923         frame: Vec<vir::ExprBool<'vir>>,
  924     ) -> EncodeResult<'vir, (), E> {
  925         footprint.parts.extend(frame);
  926         for child in edge.blocked_by_nodes(self.pcg_ctxt()) {
  927             if !matches!(child, PcgNode::Place(_)) {
  928                 continue;
  929             }
  930             let mut resources = Vec::new();
  931             let mut old_outer = WandOldOuter::Label(Some(footprint.label));
  932             self.encode_pcg_node(&child, &mut resources, &mut old_outer)?;
  933             let resources = self.vcx.mk_conj(&resources);
  934             let part = self.unless_unblocked(footprint, child.into(), Some(edge), resources);
  935             footprint.parts.extend(part);
  936         }
  937         Ok(())
  938     }

  940     /// `resource` under the conditions where none of the footprint's actions
  941     /// unblock `node`, `None` if one always does (or does whenever `edge` is
  942     /// valid).
  943     fn unless_unblocked(
  944         &mut self,
  945         footprint: &Footprint<'_, 'vir>,
  946         node: BlockedNode<'vir>,
  947         edge: Option<&BorrowPcgEdge<'vir>>,
  948         resource: vir::ExprBool<'vir>,
  949     ) -> Option<vir::ExprBool<'vir>> {
  950         let ctxt = self.pcg_ctxt();
  951         let mut unblocked_under = Vec::new();
  952         for action in footprint.actions {
  953             let unblocking = action.edge();
  954             if !unblocking.blocks_node(node, ctxt) {
  955                 continue;
  956             }
  957             let cond_conjuncts = self.edge_condition(unblocking);
  958             if cond_conjuncts.is_empty()
  959                 || edge.is_some_and(|edge| unblocking.conditions() == edge.conditions())
  960             {
  961                 return None;
  962             }
  963             unblocked_under.push(self.vcx.mk_conj(&cond_conjuncts));
  964         }
  965         if unblocked_under.is_empty() {
  966             return Some(resource);
  967         }
  968         let unblocked = self.vcx.mk_disj(&unblocked_under);
  969         let outside = self
  970             .vcx
  971             .mk_unary_op_expr(vir::UnOpKind::Not, unblocked.upcast_ty())
  972             .downcast_ty();
  973         Some(
  974             self.vcx
  975                 .mk_bin_op_expr(vir::BinOpKind::Implies, outside, resource)
  976                 .downcast_ty(),
  977         )
  978     }

 1012         edge_to_loop: bool,
 1013     ) -> EncodeResult<'vir, (), E> {
 1014         let mut to_skip = Vec::new();
 1015         match action.kind() {
 1016             //Restore(RestoreCapability<'tcx>),

 1027             ),
 1028             BorrowPcgActionKind::AddEdge { edge } => self.pcs_handle_edge(

 1030                 edge,
 1031                 EdgeAction::Add,
 1032                 edge_to_loop,
 1033                 &mut to_skip,

 1035             ),
 1036             BorrowPcgActionKind::Weaken(weaken)
 1037                 if matches!(
 1038                     weaken.from_cap(),

 1055     fn pcg_repack(&mut self, repack_op: &RepackOp<'vir>) -> EncodeResult<'vir, (), E> {
 1056         comment!(self, "[PCG] {repack_op:?}");
 1057 
 1058         fn should_ignore(repack_op: &RepackOp<'_>) -> bool {
 1059             match repack_op {
 1060                 RepackOp::RegainLoanedCapability(..) => true,
 1061                 RepackOp::Weaken(weaken) => {
 1062                     weaken.from_cap().is_exclusive()
 1063                         && matches!(
 1064                             weaken.to_cap(),
 1065                             CapabilityKind::Read | CapabilityKind::ShallowExclusive
 1066                         )
 1067                 }
 1068                 RepackOp::StorageDead(..) => true,
 1069                 _ => false,
 1070             }
 1071         }
 1072 
 1073         match repack_op {
 1074             RepackOp::Expand(_) | RepackOp::Collapse(_) => {
 1075                 let (place, capability_kind) = match repack_op {
 1076                     RepackOp::Expand(expand) => (expand.from(), expand.capability()),
 1077                     RepackOp::Collapse(collapse) => (collapse.to(), collapse.capability()),
 1078                     _ => unreachable!(),
 1079                 };
 1080                 if matches!(capability_kind, CapabilityKind::Write) {
 1081                     // Collapsing an already exhaled place is a no-op
 1082                     // TODO: unless it's through a Ref I imagine?
 1083                     //assert!(matches!(repack_op, RepackOp::Collapse(..)));
 1084                     if !matches!(repack_op, RepackOp::Collapse(..)) {
 1085                         comment!(self, "expected RepackOp::Collapse but got {repack_op:?}");
 1086                     }
 1087                     return Ok(());
 1088                 }
 1089                 let place_enc = self.encode_place(place)?;
 1090                 let place_ty = place_enc.ty;
 1091                 let place_enc = place_enc.expr.expect_predicate();
 1092                 let data = self.ty_use_impure(place_ty.ty);
 1093                 if data.specifics.is_opaque() {
 1094                     return Err(unsupported_opaque(place_ty.ty, self.current_span()));
 1095                 }
 1096                 // TODO: guide should be implemented on `RepackOp`
 1097                 let guide = match repack_op {
 1098                     pcg::free_pcs::RepackOp::Expand(expand) => expand.guide(),
 1099                     pcg::free_pcs::RepackOp::Collapse(collapse) => collapse.guide(),
 1100                     _ => None,
 1101                 };
 1102                 let index = guide.and_then(|guide| self.expansion_index(guide));
 1103                 if matches!(repack_op, pcg::free_pcs::RepackOp::Expand(..)) {
 1104                     self.stmts(data.unfold(place_ty.variant_index, place_enc, index, None, None));
 1105                 } else if matches!(repack_op, pcg::free_pcs::RepackOp::Collapse(..)) {
 1106                     self.stmts(data.fold(place_ty.variant_index, place_enc, index, None, None));
 1107                 } else {
 1108                     unreachable!()
 1109                 }
 1110                 Ok(())
 1111             }
 1112             RepackOp::Weaken(weaken)
 1113                 if matches!(
 1114                     weaken.from_cap(),
 1115                     CapabilityKind::Exclusive | CapabilityKind::ShallowExclusive
 1116                 ) && weaken.to_cap().is_write() =>
 1117             {
 1118                 self.pcg_weaken(weaken.place(), weaken.is_for_storage_dead())
 1119             }
 1120             other => {
 1121                 if should_ignore(other) {
 1122                     self.stmt(self.vcx.mk_comment_stmt(vir::vir_format!(
 1123                         self.vcx,
 1124                         "ignored repack op: {other:?}"
 1125                     )));
 1126                     Ok(())
 1127                 } else {
 1128                     Err(EncodeFullError::internal(
 1129                         format!("unexpected PCG operation {other:?}"),
 1130                         vec![self.current_span()],
 1131                     ))
 1132                 }
 1133             }
 1134         }
 1135     }

 1138         &mut self,
 1139         place: Place<'vir>,
 1140         for_storage_dead: bool,
 1141     ) -> EncodeResult<'vir, (), E> {
 1142         let place_ty = place.ty(self.pcg_ctxt());

 1144

 1148         // is resolved.

 1150             comment!(
 1151                 self,
 1152                 "Weaken to Write for {:?} (skipped exhale: StorageDead)",

 1158         let place_ty_out = self.ty_use_impure(place_ty.ty);
 1159 
 1160         let place_enc = self.encode_place(place)?;
 1161         comment!(self, "exhale due to weaken to Write");

 1163             self.vcx,
 1164             place_enc.expr.expect_predicate(),
 1165             None,
 1166         )));

 1212     fn encode_operand(
 1213         &mut self,
 1214         operand: &mir::Operand<'vir>,
 1215     ) -> EncodeResult<'vir, vir::ExprRef<'vir>, E> {
 1216         let ty = operand.ty(self.local_decls, self.vcx.tcx());
 1217         let (encode_place_result, ty_out) = match operand {
 1218             &mir::Operand::Move(source) => {
 1219                 return Ok(self
 1220                     .encode_place(Place::from(source))?
 1221                     .expr
 1222                     .expect_predicate());

 1226                 (self.encode_operand_snap(operand, &None)?, ty_out)

 1259     /// Read `operand`'s snapshot into a fresh temporary at the current
 1260     /// program point; a `Move` operand additionally consumes its place,
 1261     /// exhaling the place's predicate.
 1262     fn capture_operand_snap(
 1263         &mut self,
 1264         operand: &mir::Operand<'vir>,
 1265     ) -> EncodeResult<'vir, vir::ExprSnap<'vir>, E> {
 1266         match operand {
 1267             &mir::Operand::Copy(place) | &mir::Operand::Move(place) => {
 1268                 let (result, snap_val, _, ty_out) =
 1269                     self.encode_place_with_snap(Place::from(place))?;
 1270                 let tmp = self.new_tmp(ty_out.snapshot());
 1271                 self.stmt(self.vcx.mk_pure_assign_stmt(tmp, snap_val));
 1272                 if matches!(operand, mir::Operand::Move(_)) {

 1274                         self.vcx,
 1275                         result.expr.expect_predicate(),

 1277                     )));
 1278                 }
 1279                 Ok(tmp)
 1280             }
 1281             mir::Operand::Constant(box constant) => {
 1282                 Ok(self.encode_constant_snap(constant)?.upcast_ty())
 1283             }
 1284             mir::Operand::RuntimeChecks(_) => todo!(),
 1285         }

 1288     /// Read `statement`'s operands into an [OperandSnaps] map; see the call
 1289     /// site in [Self::visit_statement].
 1290     fn capture_operand_snaps(
 1291         &mut self,
 1292         statement: &mir::Statement<'vir>,
 1293         location: mir::Location,
 1294     ) -> EncodeResult<'vir, OperandSnaps<'vir>, E> {
 1295         use mir::visit::Visitor;
 1296         struct OperandCollector<'vir>(Vec<mir::Operand<'vir>>);
 1297         impl<'vir> Visitor<'vir> for OperandCollector<'vir> {
 1298             fn visit_operand(&mut self, operand: &mir::Operand<'vir>, _location: mir::Location) {
 1299                 if operand.place().is_some() {
 1300                     self.0.push(operand.clone());
 1301                 }
 1302             }
 1303         }
 1304         let mut collector = OperandCollector(Vec::new());
 1305         collector.visit_statement(statement, location);
 1306         let mut snaps = FxHashMap::default();
 1307         for operand in &collector.0 {
 1308             let place = operand.place().unwrap();
 1309             if snaps.contains_key(&place) {
 1310                 continue;
 1311             }
 1312             let snap = self.capture_operand_snap(operand)?;
 1313             snaps.insert(place, snap);
 1314         }
 1315         Ok(Some(snaps))
 1316     }

 1364     fn encode_place_element(
 1365         &mut self,
 1366         place: Place<'vir>,
 1367         elem: mir::PlaceElem<'vir>,
 1368         expr: PlaceExpr<'vir>,
 1369     ) -> EncodeResult<'vir, PlaceExpr<'vir>, E> {
 1370         let place_ty = place.ty(self.pcg_ctxt());
 1371         Ok(match elem {
 1372             mir::ProjectionElem::Field(field_idx, _) => {
 1373                 let e_ty = self.ty_use_impure(place_ty.ty);
 1374                 if e_ty.specifics.is_opaque() {
 1375                     return Err(unsupported_opaque(place_ty.ty, self.current_span()));
 1376                 }
 1377                 let field_access = e_ty.expect_variant_opt(place_ty.variant_index);
 1378                 // Only the last field of a struct/tuple can be unsized (a DST's
 1379                 // tail); it then shares the containing value's pointer metadata.
 1380                 let is_last_field = field_idx.index() + 1 == field_access.fields.len();
 1381                 PlaceExpr {
 1382                     address: field_access[field_idx].field_ref(expr.address),
 1383                     metadata: if is_last_field { expr.metadata } else { None },
 1384                     snap: expr.snap.map(|snap| {
 1385                         let e_ty = self.ty_use_pure(place_ty.ty);
 1386                         let field_access = e_ty.expect_variant_opt(place_ty.variant_index);
 1387                         field_access[field_idx].read(snap.downcast_ty())
 1388                     }),
 1389                 }
 1390             }
 1391 
 1392             mir::ProjectionElem::Index(idx) => {
 1393                 let e_ty = self.ty_use_impure(place_ty.ty).expect_array();
 1394                 let idx = self
 1395                     .encode_place_with_snap(mir::Place::from(idx).into())?
 1396                     .1
 1397                     .downcast_ty();
 1398                 PlaceExpr {
 1399                     address: e_ty.ref_to_index_ref(expr.address, idx),
 1400                     metadata: None,
 1401                     snap: expr.snap.map(|snap| {
 1402                         let e_ty = self.ty_use_pure(place_ty.ty).expect_array();
 1403                         e_ty.index(snap.downcast_ty(), idx)
 1404                     }),
 1405                 }
 1406             }
 1407             // `from_end` indexes from the runtime length of a slice; it is
 1408             // always false for an array, the only supported subject.
 1409             mir::ProjectionElem::ConstantIndex {
 1410                 offset,
 1411                 from_end: false,
 1412                 ..
 1413             } => {
 1414                 let e_ty = self.ty_use_impure(place_ty.ty).expect_array();
 1415                 let idx = self.usize_const_snap(offset);
 1416                 PlaceExpr {
 1417                     address: e_ty.ref_to_index_ref(expr.address, idx),
 1418                     metadata: None,
 1419                     snap: expr.snap.map(|snap| {
 1420                         let e_ty = self.ty_use_pure(place_ty.ty).expect_array();
 1421                         e_ty.index(snap.downcast_ty(), idx)
 1422                     }),
 1423                 }
 1424             }
 1425             // TODO: should all variants start at the same `Ref`?
 1426             mir::ProjectionElem::Downcast(..) => PlaceExpr {
 1427                 address: expr.address,
 1428                 metadata: None,
 1429                 snap: expr.snap,
 1430             },
 1431             mir::ProjectionElem::Deref => {
 1432                 assert!(place_ty.variant_index.is_none());
 1433                 let e_ty = self.ty_use_impure(place_ty.ty);
 1434                 match place_ty.ty.kind() {
 1435                     ty::TyKind::Adt(adt, _) if adt.is_box() => match expr.snap {
 1436                         // With a snapshot at hand (e.g. behind a shared
 1437                         // reference, where no permission to the box's fields
 1438                         // is held), everything is read out of it.
 1439                         Some(snap) => {
 1440                             let data = self.ty_use_pure(place_ty.ty).expect_structlike();
 1441                             let snap = snap.downcast_ty();
 1442                             PlaceExpr {
 1443                                 address: data.box_address_access(snap),
 1444                                 metadata: Some(data.box_metadata_access(snap)),
 1445                                 snap: Some(data.box_value_access(snap)),
 1446                             }
 1447                         }
 1448                         // Otherwise the address and metadata are read out of
 1449                         // the (folded) `Unique` field's predicate; the box
 1450                         // itself is already unfolded here.
 1451                         None => {
 1452                             let data = e_ty.expect_structlike();
 1453                             PlaceExpr {
 1454                                 address: data.box_address(expr.address),
 1455                                 metadata: Some(data.box_metadata(expr.address)),
 1456                                 snap: None,
 1457                             }
 1458                         }
 1459                     },
 1460                     ty::TyKind::Ref(_, _, ty::Mutability::Not) => {
 1461                         let snap = expr
 1462                             .snap
 1463                             .unwrap_or_else(|| e_ty.ref_to_snap(expr.address))
 1464                             .downcast_ty();
 1465                         let p_ty = self.ty_use_pure(place_ty.ty).expect_immref();
 1466                         PlaceExpr {
 1467                             address: p_ty.addr_access(snap),
 1468                             metadata: Some(p_ty.metadata_access(snap)),
 1469                             snap: Some(p_ty.value_access(snap)),
 1470                         }
 1471                     }
 1472                     ty::TyKind::Ref(_, _, ty::Mutability::Mut) => {
 1473                         let ref_snap = e_ty.ref_to_snap(expr.address).downcast_ty();
 1474                         let p_ty = self.ty_use_pure(place_ty.ty).expect_mutref();
 1475                         PlaceExpr {
 1476                             address: p_ty.deref_access(ref_snap),
 1477                             metadata: Some(p_ty.metadata_access(ref_snap)),
 1478                             snap: None,
 1479                         }
 1480                     }
 1481                     ty::TyKind::RawPtr(..) => {
 1482                         return Err(EncodeFullError::unsupported(
 1483                             format!("dereference of the raw pointer `{}`", place_ty.ty),
 1484                             vec![self.current_span()],
 1485                         ));
 1486                     }
 1487                     _ => unreachable!(),
 1488                 }
 1489             }
 1490             _ => todo!("Unsupported ProjectionElem {:?}", elem),
 1491         })
 1492     }

 1598     pub fn visit_body(&mut self, body: &mir::Body<'vir>) -> EncodeResult<'vir, (), E> {
 1599         /// A work-queue item, min-ordered by the block's reverse-postorder
 1600         /// index (ties broken by insertion order): every non-back-edge
 1601         /// predecessor of a block is then encoded before the block itself,
 1602         /// in particular both branches before their join. The borrow-expiry
 1603         /// handling relies on this: a call's `call_labels`/`wandless_calls`
 1604         /// entry must exist by the time an expiring borrow references the
 1605         /// call's block. `heads_hit` is payload, not part of the order.
 1606         struct WorkItem {
 1607             rpo_idx: usize,
 1608             seq: usize,
 1609             block: mir::BasicBlock,
 1610             heads_hit: FxHashSet<LoopId>,
 1611         }
 1612         impl PartialEq for WorkItem {
 1613             fn eq(&self, other: &Self) -> bool {
 1614                 (self.rpo_idx, self.seq) == (other.rpo_idx, other.seq)
 1615             }
 1616         }
 1617         impl Eq for WorkItem {}
 1618         impl PartialOrd for WorkItem {
 1619             fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
 1620                 Some(self.cmp(other))
 1621             }
 1622         }
 1623         impl Ord for WorkItem {
 1624             fn cmp(&self, other: &Self) -> std::cmp::Ordering {
 1625                 // Reversed so that the max-heap `BinaryHeap` pops the minimum.
 1626                 (other.rpo_idx, other.seq).cmp(&(self.rpo_idx, self.seq))
 1627             }
 1628         }
 1629 
 1630         let rpo_idx: FxHashMap<mir::BasicBlock, usize> = body
 1631             .basic_blocks
 1632             .reverse_postorder()
 1633             .iter()
 1634             .enumerate()
 1635             .map(|(idx, block)| (*block, idx))
 1636             .collect();
 1637         let mut next_seq = 0..;
 1638         let mut queue = std::collections::BinaryHeap::new();
 1639         let mut visited = FxHashSet::default();
 1640 
 1641         // In Prusti, we want to be able to support body invariants, i.e.,
 1642         // functional specifications for loops, placed at a point further
 1643         // *within* the loop body. When encoding such invariants into Viper, we
 1644         // choose to unroll the first part of the loop (before the invariant is
 1645         // reached), then encode the rest of the loop followed by the repeated
 1646         // first part of the loop as an actual Viper loop with an invariant
 1647         // annotation. This means that the part of the loop before the body
 1648         // invariant is reached is encoded in duplicate, thus we need to ensure
 1649         // basic block labels etc are *not* emitted in duplicate. As a result,
 1650         // we keep track of a "pre-loop" set: the set of loops we have entered
 1651         // (in the CFG) but whose body invariants we have not yet reached.
 1652         //
 1653         // The pre-loop set is a set because of pathological cases like this:
 1654         // ```rust
 1655         // while { // (OUT)
 1656         //     loop { // (IN)
 1657         //         // (maybe break;) ... (A)
 1658         //         body_invariant!(...);
 1659         //         // ...
 1660         //     }
 1661         //     // ...
 1662         // } {
 1663         //     // ...
 1664         //     body_invariant!(...);
 1665         //     // ...
 1666         // }
 1667         // ```
 1668         // Here, the loop guard of the outer `while` loop contains a loop as
 1669         // well. The "(A)" code fragment must be encoded multiple times:
 1670         // * when entering (OUT) and (IN);
 1671         // * when entering (OUT), for an arbitrary iteration of (IN);
 1672         // * for an arbitrary iteration of (OUT); entering (IN); and
 1673         // * for an arbitrary iteration of (OUT) and (IN).
 1674         // The pre-loop sets for the four instances are, respectively:
 1675         // * {OUT, IN},
 1676         // * {OUT},
 1677         // * {IN}, and
 1678         // * {}.
 1679         //
 1680         // Loop heads and body invariants: loops as identified by the PCG loop
 1681         // analysis have a loop head at the very first basic block, i.e., the
 1682         // basic block that backedges jump to. For Prusti, we want to instead
 1683         // consider the body invariant, if declared, to be the loop head, as
 1684         // explained above. We only fall back to the PCG loop head if the body
 1685         // invariant is not declared (i.e., we will only emit a permission
 1686         // invariant, not a functional specification).
 1687         //
 1688         // The overall approach in this method is to walk basic blocks of the
 1689         // CFG using a queue. Each item in the queue is a pair (basic block.
 1690         // loop heads hit), where the latter is a set of loop heads that were
 1691         // already crossed on this path. The pre-loop set is then the set of
 1692         // loops the current block is in minus loop heads hit.
 1693 
 1694         // keep track of which loops we have already entered (on a particular
 1695         // path); a LoopId is in this set as soon as the loop head (or the body
 1696         // invariant for a loop with one) is entered
 1697         let mut start_heads = FxHashSet::default();
 1698         let start_loops = self
 1699             .loop_analysis()
 1700             .loops(mir::START_BLOCK)
 1701             .collect::<FxHashSet<_>>();
 1702         if !start_loops.is_empty() {
 1703             // the start block is either not part of a loop, or else it is in
 1704             // exactly one loop, of which it *may* be the head (if the body
 1705             // invariant is omitted or coincides with the start block)
 1706             assert_eq!(start_loops.len(), 1);
 1707             let start_loop = *start_loops.iter().next().unwrap();
 1708             if self.spec_blocks.loop_head_at[&start_loop] == mir::START_BLOCK {
 1709                 start_heads.insert(start_loop);
 1710             }
 1711         }
 1712         queue.push(WorkItem {
 1713             rpo_idx: rpo_idx[&mir::START_BLOCK],
 1714             seq: next_seq.next().unwrap(),
 1715             block: mir::START_BLOCK,
 1716             heads_hit: start_heads,
 1717         });
 1718 
 1719         while let Some(WorkItem {
 1720             block,
 1721             mut heads_hit,
 1722             ..
 1723         }) = queue.pop()
 1724         {
 1725             let in_loops = self.loop_analysis().loops(block).collect::<FxHashSet<_>>();
 1726 
 1727             heads_hit.retain(|l| in_loops.contains(l));
 1728 
 1729             // is this a loop head?
 1730             if let Some(loop_spec) = self.spec_blocks.loop_specs.get(&block)
 1731                 && !heads_hit.insert(loop_spec.loop_id)
 1732             {
 1733                 // we already walked over this loop head, so the full loop
 1734                 // iteration was already emitted on this path
 1735                 continue;
 1736             }
 1737 
 1738             let pre_loops = in_loops
 1739                 .iter()
 1740                 .copied()
 1741                 .filter(|l| !heads_hit.contains(l))
 1742                 .sorted()
 1743                 .collect::<Vec<_>>();
 1744 
 1745             if !visited.insert((block, pre_loops.clone())) {
 1746                 continue;
 1747             }
 1748 
 1749             self.current_block = Some(block);
 1750             // Allocate label for current block and its successors with the
 1751             // correct pre-loop prefixes.
 1752             self.current_block_pres = Some(pre_loops.iter().map(|l| l.index()).collect());
 1753             self.current_block_succs = Some(
 1754                 body.basic_blocks
 1755                     .successors(block)
 1756                     // Successors can have a different prefix than the current
 1757                     // block; the successor can be:
 1758                     // * part of a loop the current block is not in;
 1759                     // * the loop head of a loop the current block is in; and/or
 1760                     // * not part of a loop the current block is in.
 1761                     .map(|succ| {
 1762                         // TODO: this does a lot of duplicate work
 1763                         //   maybe some two-pass approach would be nicer?
 1764                         let succ_in_loops =
 1765                             self.loop_analysis().loops(succ).collect::<FxHashSet<_>>();
 1766                         let mut succ_heads_hit = heads_hit.clone();
 1767                         succ_heads_hit.retain(|l| succ_in_loops.contains(l));
 1768                         if let Some(loop_spec) = self.spec_blocks.loop_specs.get(&succ) {
 1769                             succ_heads_hit.insert(loop_spec.loop_id);
 1770                         }
 1771                         let succ_pre_loops = succ_in_loops
 1772                             .iter()
 1773                             .copied()
 1774                             .filter(|l| !succ_heads_hit.contains(l))
 1775                             .map(|l| l.index())
 1776                             .sorted();
 1777                         (
 1778                             succ,
 1779                             self.vcx.mk_block_label(succ.as_usize(), succ_pre_loops),
 1780                         )
 1781                     })
 1782                     .collect(),
 1783             );
 1784 
 1785             self.visit_basic_block_data(block, &body[block])?;
 1786             self.current_block = None;
 1787             self.current_block_pres = None;
 1788             self.current_block_succs = None;
 1789 
 1790             for successor in body.basic_blocks.successors(block) {
 1791                 // Specification-only arms are not encoded at all; their
 1792                 // switches are encoded as jumps to the live target.
 1793                 if self.spec_blocks.spec_arms.blocks.contains(&successor) {
 1794                     continue;
 1795                 }
 1796                 queue.push(WorkItem {
 1797                     rpo_idx: rpo_idx[&successor],
 1798                     seq: next_seq.next().unwrap(),
 1799                     block: successor,
 1800                     heads_hit: heads_hit.clone(),
 1801                 });
 1802             }
 1803         }
 1804         Ok(())
 1805     }

 1807     fn encode_spec_block(
 1808         &mut self,
 1809         spec_block: mir::BasicBlock,
 1810         pcg_state: &Pcg<'_, 'vir>,
 1811         place_capabilities: Option<&FxHashMap<Place<'vir>, CapabilityKind>>,
 1812     ) -> EncodeResult<'vir, vir::ExprBool<'vir>, E> {
 1813         let enc_output = self.deps.require_dep::<MirPureEnc>(MirPureEncTask {
 1814             encoding_depth: 0,
 1815             def_id: self.def_id,
 1816             gargs: GParams::from(self.def_id).identity_args(),
 1817             kind: PureKind::SpecBlock(spec_block),
 1818         })?;
 1819         let mut locals: FxHashMap<mir::Local, vir::ExprSnap<'vir>> = FxHashMap::default();
 1820         for &local in &enc_output.inputs {
 1821             let place: Place = local.into();
 1822             let place_ty = RustTyDecomposition::from_ty(place.ty(self.pcg_ctxt()).ty, self.def_id);
 1823             // The places the local is unpacked to; computed once per local.
 1824             let local_state = &pcg_state.owned()[local];
 1825             let leaf_places: Vec<Place<'vir>> = if local_state.is_allocated() {
 1826                 local_state
 1827                     .expansions()
 1828                     .leaf_places(self.pcg_ctxt())
 1829                     .into_iter()
 1830                     .map(Into::into)
 1831                     .collect()
 1832             } else {
 1833                 Vec::new()
 1834             };
 1835             let snap = self.encode_partial_place_snap(
 1836                 place,
 1837                 place_ty,
 1838                 pcg_state,
 1839                 &leaf_places,
 1840                 place_capabilities,
 1841             )?;
 1842             locals.insert(local, snap);
 1843         }
 1844         let expr = enc_output.reify(self.vcx.alloc(locals)).downcast_ty();
 1845         Ok(expr)
 1846     }

 1848     fn encode_partial_place_snap(
 1849         &mut self,
 1850         place: Place<'vir>,
 1851         place_ty: RustTyDecomposition<'vir>,
 1852         pcg_state: &Pcg<'_, 'vir>,
 1853         leaf_places: &[Place<'vir>],
 1854         place_capabilities: Option<&FxHashMap<Place<'vir>, CapabilityKind>>,
 1855     ) -> EncodeResult<'vir, vir::ExprSnap<'vir>, E> {
 1856         let pure_ty = self.deps.require_dep::<TyUsePureEnc>(place_ty)?;
 1857 
 1858         // Check if place is unpacked
 1859         if let Some(struct_data) = pure_ty.get_structlike()
 1860             && leaf_places
 1861                 .iter()
 1862                 .any(|leaf| place.is_strict_prefix_of(*leaf))
 1863         {
 1864             let rust_struct = place_ty.ty.expect_structlike();
 1865             let is_box = place_ty.ty.special == RustTySpecial::Box;
 1866             let field_places = place.expand_field(None, self.pcg_ctxt()).unwrap();
 1867             let mut field_snaps = Vec::with_capacity(rust_struct.fields.len());
 1868             for field_data in rust_struct.fields.iter() {
 1869                 let field_ty = field_data.decompose_normalize(place_ty.args);
 1870                 let field_snap = match field_data.address {
 1871                     // The value of a `Box`, stored at its pointer.
 1872                     RustFieldAddress::Dynamic => self.encode_partial_place_snap(
 1873                         place.project_deref(self.pcg_ctxt()),
 1874                         field_ty,
 1875                         pcg_state,
 1876                         leaf_places,
 1877                         place_capabilities,
 1878                     )?,
 1879                     // The PCG unpacks a `Box` only through its pointer, so
 1880                     // its other fields are held whole once it is unpacked.
 1881                     RustFieldAddress::Constant if is_box => {
 1882                         self.encode_place_with_snap(field_places[field_data.fid.index()])?
 1883                             .1
 1884                     }
 1885                     RustFieldAddress::Constant => self.encode_partial_place_snap(
 1886                         field_places[field_data.fid.index()],
 1887                         field_ty,
 1888                         pcg_state,
 1889                         leaf_places,
 1890                         place_capabilities,
 1891                     )?,
 1892                 };
 1893                 field_snaps.push(field_snap);
 1894             }
 1895 
 1896             return Ok(struct_data.field_snaps_to_snap(field_snaps).upcast_ty());
 1897         }
 1898 
 1899         // Check if we have capability to this place
 1900         let moved_out = if let Some(capabilities) = place_capabilities {
 1901             // for loop invariants we take the calculated capabilities of the invariant
 1902             capabilities.get(&place).is_none_or(|c| c.is_write())
 1903         } else {
 1904             // for other specblocks we take the owned capabilities
 1905             let Some(capability) = pcg_state.computed_owned_capability(place, self.pcg_ctxt())
 1906             else {
 1907                 return Err(EncodeFullError::internal(
 1908                     format!("no capability for `{place:?}` when encoding a specification"),
 1909                     vec![self.current_span()],
 1910                 ));
 1911             };
 1912             capability.is_write()
 1913         };
 1914 
 1915         if moved_out {
 1916             // we do not own this place or it is moved out -> use temporary snap
 1917             let snapshot_ty = pure_ty.snapshot;
 1918             return Ok(self.new_tmp(snapshot_ty));
 1919         }
 1920 
 1921         Ok(self.encode_place_with_snap(place)?.1)
 1922     }

 1924     fn visit_basic_block_data(
 1925         &mut self,
 1926         block: mir::BasicBlock,
 1927         data: &mir::BasicBlockData<'vir>,
 1928     ) -> EncodeResult<'vir, (), E> {
 1929         let current_block_label = self.vcx.mk_block_label(
 1930             block.as_usize(),
 1931             self.current_block_pres.as_ref().unwrap().iter().copied(),
 1932         );
 1933 
 1934         // We are verifying the absence of panics, so cleanup block should never
 1935         // be reached, or even referenced.
 1936         if data.is_cleanup {
 1937             self.encoded_blocks.push(
 1938                 self.vcx.mk_cfg_block(
 1939                     current_block_label,
 1940                     &[],
 1941                     &[],
 1942                     self.vcx
 1943                         .mk_dummy_stmt(vir::vir_format!(self.vcx, "cleanup block")),
 1944                 ),
 1945             );
 1946             return Ok(());
 1947         }
 1948 
 1949         // Specification-only arms are unreachable after their switches are
 1950         // encoded as jumps to the live target; emit nothing. The specs
 1951         // themselves are encoded separately (assertions at the block they
 1952         // are attached to, loop invariants at the loop head, closure specs
 1953         // as the closure's contract).
 1954         if self.spec_blocks.spec_arms.blocks.contains(&block) {
 1955             return Ok(());
 1956         }
 1957 
 1958         self.deps().check_cycle()?;
 1959 
 1960         self.current_stmts = Some(Vec::with_capacity(
 1961             data.statements.len(), // TODO: not exact?
 1962         ));
 1963         self.current_block_label = Some(current_block_label);
 1964         let cfpcs = self.fpcs_analysis.get_all_for_bb(block).unwrap().unwrap();
 1965 
 1966         // Calculate invariant at the body invariant, if specified, or at the
 1967         // loop head by default
 1968         let mut invariant = None;
 1969         if let Some(loop_spec) = self.spec_blocks.loop_specs.get(&block) {
 1970             let loop_place_usages = self
 1971                 .fpcs_analysis
 1972                 .analysis()
 1973                 .loop_place_usages(loop_spec.loop_id)
 1974                 .clone();
 1975             let invariants = loop_spec.invariants.clone();
 1976             let loop_id = loop_spec.loop_id;
 1977             // The permission part of the invariant, and errors encoding it, refer to the
 1978             // loop head.
 1979             let head_span = self
 1980                 .body
 1981                 .source_info(loop_spec.original_head_block.start_location())
 1982                 .span;
 1983             let (permissions, place_capabilities) = self.vcx.with_span(head_span, |_| {
 1984                 self.get_loop_inv(&cfpcs, &loop_place_usages, self.pcg_ctxt())
 1985             })?;
 1986             let pcg_state = &cfpcs.statements[0].states[EvalStmtPhase::PreOperands];
 1987             // The Viper loop havocs the edge flags set in the loop, including
 1988             // those set between the loop's first block and the body invariant,
 1989             // which guards of PCG actions further in the loop still refer to.
 1990             let loop_branch_choices = pcg_state
 1991                 .borrow_pcg()
 1992                 .validity_conditions()
 1993                 .all_branch_choices()
 1994                 .filter(|choices| self.loop_analysis().in_loop(choices.from(), loop_id))
 1995                 .collect::<Vec<_>>();
 1996             let branch_choices = loop_branch_choices
 1997                 .into_iter()
 1998                 .map(|choices| self.branch_choices_cond(choices))
 1999                 .collect::<Vec<_>>();
 2000             let functional = invariants
 2001                 .into_iter()
 2002                 .map(|(spec_block, span)| {
 2003                     let expr =
 2004                         self.encode_spec_block(spec_block, pcg_state, Some(&place_capabilities))?;
 2005                     self.vcx.with_span(span, |vcx| {
 2006                         let error_msg = "loop invariant might not hold after a loop iteration that \
 2007                                          preserves the loop condition.";
 2008                         vcx.handle_error("invariant.not.preserved:assertion.false", move |_| {
 2009                             Some(vec![PrustiError::verification(error_msg, span.into())])
 2010                         });
 2011                         self.vcx.with_span(span, |vcx| {
 2012                             let error_msg =
 2013                                 "loop invariant might not hold in the first loop iteration.";
 2014                             vcx.handle_error(
 2015                                 "invariant.not.established:assertion.false",
 2016                                 move |_| {
 2017                                     Some(vec![PrustiError::verification(error_msg, span.into())])
 2018                                 },
 2019                             );
 2020                             Ok(expr.realloc_span())
 2021                         })
 2022                     })
 2023                 })
 2024                 .collect::<Result<Vec<_>, _>>()?;
 2025             invariant = Some(
 2026                 self.vcx.alloc_slice(
 2027                     &permissions
 2028                         .into_iter()
 2029                         .chain(branch_choices)
 2030                         .chain(functional)
 2031                         .collect::<Vec<_>>(),
 2032                 ),
 2033             );
 2034         }
 2035 
 2036         self.current_fpcs = Some(cfpcs);
 2037 
 2038         assert!(self.current_terminator.is_none());
 2039         for (index, statement) in data.statements.iter().enumerate() {
 2040             let location = mir::Location {
 2041                 block,
 2042                 statement_index: index,
 2043             };
 2044             self.visit_statement(statement, location)?;
 2045         }
 2046         let location = mir::Location {
 2047             block,
 2048             statement_index: data.statements.len(),
 2049         };
 2050         self.visit_terminator(data.terminator(), location)?;
 2051 
 2052         if let Some(specs) = self.spec_blocks.specs_for.get(&block).cloned() {
 2053             for spec in specs {
 2054                 if matches!(spec.kind, SpecBlockKind::LoopInvariant) {
 2055                     continue;
 2056                 }
 2057                 let current_fpcs = self.current_fpcs.take().unwrap();
 2058                 let pcg_state =
 2059                     &current_fpcs.statements.last().unwrap().states[EvalStmtPhase::PostMain];
 2060                 let spec_expr = self.encode_spec_block(spec.block, pcg_state, None);
 2061                 self.current_fpcs = Some(current_fpcs);
 2062                 let spec_expr = spec_expr?;
 2063                 let span = spec.span;
 2064                 match spec.kind {
 2065                     SpecBlockKind::Assert => {
 2066                         self.vcx.with_span(span, |vcx| {
 2067                             let error_msg = "the asserted expression might not hold";
 2068                             vcx.handle_error("exhale.failed:assertion.false", move |_| {
 2069                                 Some(vec![PrustiError::verification(error_msg, span.into())])
 2070                             });
 2071                             self.stmt(self.vcx.mk_exhale_stmt(spec_expr));
 2072                         });
 2073                     }
 2074                     SpecBlockKind::Assume => {
 2075                         self.stmt(self.vcx.mk_inhale_stmt(spec_expr));
 2076                     }
 2077                     SpecBlockKind::Refute => {
 2078                         self.vcx.with_span(span, |vcx| {
 2079                             let error_msg =
 2080                                 "the refuted expression holds in all cases or could not be reached";
 2081                             vcx.handle_error("refute.failed:refutation.true", move |_| {
 2082                                 Some(vec![PrustiError::verification(error_msg, span.into())])
 2083                             });
 2084                             self.stmt(self.vcx.mk_refute_stmt(spec_expr));
 2085                         });
 2086                     }
 2087                     SpecBlockKind::LoopInvariant => {
 2088                         // nothing to do: loop invariants are handled elsewhere
 2089                     }
 2090                 }
 2091             }
 2092         }
 2093 
 2094         let stmts = self.current_stmts.take().unwrap();
 2095         let terminator = self.current_terminator.take().unwrap();
 2096         self.encoded_blocks.push(self.vcx.mk_cfg_block(
 2097             self.current_block_label.take().unwrap(),
 2098             invariant.unwrap_or_default(),
 2099             self.vcx.alloc_slice(&stmts),
 2100             terminator,
 2101         ));
 2102         Ok(())
 2103     }

 2105     fn visit_statement(
 2106         &mut self,
 2107         statement: &mir::Statement<'vir>,
 2108         location: mir::Location,
 2109     ) -> EncodeResult<'vir, (), E> {
 2110         self.vcx.with_span(statement.source_info.span, |_| {
 2111             self.deps().check_cycle()?;
 2112 
 2113             self.new_before_label(location);
 2114 
 2115             comment!(self, "[MIR] {location:?}: {statement:?}");
 2116 
 2117             self.pcg_phase_actions(location, EvalStmtPhase::PreOperands)?;
 2118 
 2119             // Read the statement's operands at their temporal position,
 2120             // between the `PreOperands` and `PostOperands` repacks. An
 2121             // operand's place may alias the destination, whose predicate the
 2122             // `PreMain` weaken exhales before the effect is encoded (`x /= y`
 2123             // lowers to `x = Div(copy x, move y)`).
 2124             let captured = self.capture_operand_snaps(statement, location)?;
 2125 
 2126             self.pcg_phase_actions(location, EvalStmtPhase::PostOperands)?;
 2127             self.pcg_phase_actions(location, EvalStmtPhase::PreMain)?;
 2128 
 2129             // Assignments to the locals only serving specification-only arms
 2130             // (necessarily scaffolding stores) are not encoded, since the
 2131             // locals are not declared.
 2132             if let mir::StatementKind::Assign(box (dest, _)) = &statement.kind
 2133                 && let Some(local) = dest.as_local()
 2134                 && self.spec_blocks.spec_arms.spec_only_locals.contains(&local)
 2135             {
 2136                 return Ok(());
 2137             }
 2138 
 2139             let span = statement.source_info.span;
 2140 
 2141             match &statement.kind {
 2142                 mir::StatementKind::Assign(box (dest, rvalue)) => {
 2143                     // What are we assigning to?
 2144                     let proj_enc = self
 2145                         .encode_place(Place::from(*dest))?
 2146                         .expr
 2147                         .expect_predicate();
 2148 
 2149                     // The snapshot of the value that we are assigning.
 2150                     let rval_enc = self.encode_rvalue(rvalue, span, &captured)?;
 2151 
 2152                     let dest_ty = dest.ty(self.local_decls, self.vcx.tcx());
 2153                     assert!(dest_ty.variant_index.is_none());
 2154                     let dest_ty_out = self.ty_use_impure(dest_ty.ty);
 2155                     let method_assign_app =
 2156                         dest_ty_out.apply_method_assign(self.vcx, proj_enc, rval_enc.expr);
 2157                     self.stmt(method_assign_app);
 2158                     self.stmts(rval_enc.post_fold_stmts(proj_enc));
 2159                 }
 2160 
 2161                 // no-ops
 2162                 mir::StatementKind::StorageLive(..)
 2163                 | mir::StatementKind::StorageDead(..)
 2164                 | mir::StatementKind::FakeRead(_)
 2165                 | mir::StatementKind::PlaceMention(_)
 2166                 | mir::StatementKind::AscribeUserType(..)
 2167                 | mir::StatementKind::Coverage(_)
 2168                 | mir::StatementKind::ConstEvalCounter
 2169                 | mir::StatementKind::Nop
 2170                 | mir::StatementKind::BackwardIncompatibleDropHint { .. } => {}
 2171 
 2172                 mir::StatementKind::Intrinsic(intrinsic_kind) => {
 2173                     return Err(EncodeFullError::unsupported(
 2174                         format!("unsupported intrinsic statement {intrinsic_kind:?}"),
 2175                         vec![span],
 2176                     ));
 2177                 }
 2178                 mir::StatementKind::SetDiscriminant { .. } => unreachable!(
 2179                     "the statement kind {:?} is not allowed in the MIR analysis phase",
 2180                     statement.kind
 2181                 ),
 2182             }
 2183 
 2184             self.pcg_phase_actions(location, EvalStmtPhase::PostMain)?;
 2185             Ok(())
 2186         })
 2187     }

 2199     fn encode_terminator(
 2200         &mut self,
 2201         terminator: &mir::Terminator<'vir>,
 2202         location: mir::Location,
 2203     ) -> EncodeResult<'vir, (), E> {
 2204         self.deps().check_cycle()?;
 2205 
 2206         self.new_before_label(location);
 2207         comment!(self, "[MIR] {location:?}: {:?}", terminator.kind);
 2208         let span = terminator.source_info.span;
 2209 
 2210         // `PostMain` is not emitted here: a terminator's operands are read
 2211         // after these phases' repacks (e.g. a `SwitchInt` discriminant that
 2212         // projects into an aggregate is read in the branch condition, relying
 2213         // on the `PreOperands` unfolds), while the `PostMain` actions re-pack
 2214         // owned places for the CFG join and thus may fold those operands'
 2215         // places away. [Self::finish_terminator] emits them after the operands
 2216         // have been read, then records a snapshot before the edge repacks;
 2217         // every terminator arm with a successor must call it. Terminators
 2218         // without successors have no successor state to re-pack for.
 2219         for phase in [
 2220             EvalStmtPhase::PreOperands,
 2221             EvalStmtPhase::PostOperands,
 2222             EvalStmtPhase::PreMain,
 2223         ] {
 2224             comment!(self, "PCG (T) {phase}");
 2225             self.pcg_phase_actions(location, phase)?;
 2226         }
 2227 
 2228         // A specification-only arm's `if false` switch is encoded as an
 2229         // unconditional jump to the live target (the continuation, or the
 2230         // inline ghost body); the arm itself is not encoded.
 2231         let spec_goto = self
 2232             .spec_blocks
 2233             .spec_arms
 2234             .switches
 2235             .get(&location.block)
 2236             .map(|live_target| mir::TerminatorKind::Goto {
 2237                 target: *live_target,
 2238             });
 2239         let terminator = match spec_goto.as_ref().unwrap_or(&terminator.kind) {
 2240             mir::TerminatorKind::Goto { target }
 2241             | mir::TerminatorKind::FalseUnwind {
 2242                 real_target: target,
 2243                 ..
 2244             }
 2245             | mir::TerminatorKind::FalseEdge {
 2246                 real_target: target,
 2247                 ..
 2248             }
 2249             | mir::TerminatorKind::Drop { target, .. } => {
 2250                 // A `Drop`'s semantics (releasing the dropped place's
 2251                 // permission via a weaken exhale) are carried by the PCG
 2252                 // statements, so only its goto remains to be encoded here.
 2253                 self.goto_single_succ(location, *target)?
 2254             }
 2255             mir::TerminatorKind::SwitchInt { discr, targets } => {
 2256                 let discr_ty_rs = discr.ty(self.local_decls, self.vcx.tcx());
 2257                 let discr_ty = self.ty_use_pure(discr_ty_rs).expect_primitive();
 2258 
 2259                 let discr_ex =
 2260                     discr_ty.snap_to_prim(self.encode_operand_snap(discr, &None)?.downcast_ty());
 2261                 let discr_tmp = self.new_tmp(discr_ex.ty());
 2262                 self.stmt(self.vcx.mk_pure_assign_stmt(discr_tmp, discr_ex));
 2263                 self.finish_terminator(location)?;
 2264 
 2265                 let goto_targets = self.vcx.alloc_slice(
 2266                     &targets
 2267                         .iter()
 2268                         .enumerate()
 2269                         .map(|(idx, (value, target))| {
 2270                             let mut extra_stmts = self.collect_pcs_succ_at(idx, target)?;
 2271                             extra_stmts.push(self.set_from_to_flag(location.block, target));
 2272 
 2273                             Ok(self.vcx.mk_goto_if_target(
 2274                                 discr_ty.expr_from_bits(discr_ty_rs, value).as_dyn(),
 2275                                 self.current_block_succs.as_ref().unwrap()[&target],
 2276                                 self.vcx.alloc_slice(&extra_stmts),
 2277                             ))
 2278                         })
 2279                         .collect::<EncodeResult<'vir, Vec<_>, E>>()?,
 2280                 );
 2281                 let goto_otherwise =
 2282                     self.current_block_succs.as_ref().unwrap()[&targets.otherwise()];
 2283 
 2284                 let otherwise_succ_idx = goto_targets.len();
 2285                 let mut otherwise_stmts =
 2286                     self.collect_pcs_succ_at(otherwise_succ_idx, targets.otherwise())?;
 2287                 otherwise_stmts.push(self.set_from_to_flag(location.block, targets.otherwise()));
 2288 
 2289                 self.vcx.mk_goto_if_stmt(
 2290                     discr_tmp.as_dyn(),
 2291                     goto_targets,
 2292                     goto_otherwise,
 2293                     self.vcx.alloc_slice(&otherwise_stmts),
 2294                 )
 2295             }
 2296             mir::TerminatorKind::Return => {
 2297                 let current_fpcs = self.current_fpcs.take().unwrap();
 2298                 let borrows = current_fpcs.statements.last().unwrap().states
 2299                     [EvalStmtPhase::PostMain]
 2300                     .borrow_pcg();
 2301                 let wand_packages = self.package_wands(borrows)?;
 2302                 self.current_fpcs = Some(current_fpcs);
 2303                 self.stmts(wand_packages);
 2304 
 2305                 self.vcx
 2306                     .mk_goto_stmt(self.vcx.alloc(vir::CfgBlockLabelData::End))
 2307             }
 2308             mir::TerminatorKind::Call {
 2309                 func,
 2310                 args,
 2311                 destination,
 2312                 target,
 2313                 ..
 2314             } => {
 2315                 // emit the current block, create a new label for the terminator
 2316                 // TODO: should we do this for any other terminators?
 2317                 let terminator_label = self.vcx.mk_terminator_label(
 2318                     self.current_block.unwrap().as_usize(),
 2319                     self.current_block_pres.as_ref().unwrap().iter().copied(),
 2320                 );
 2321                 self.encoded_blocks.push(
 2322                     self.vcx.mk_cfg_block(
 2323                         self.current_block_label.replace(terminator_label).unwrap(),
 2324                         &[],
 2325                         self.vcx
 2326                             .alloc_slice(&self.current_stmts.replace(Vec::new()).unwrap()),
 2327                         self.vcx.mk_goto_stmt(terminator_label),
 2328                     ),
 2329                 );
 2330 
 2331                 let (func_def_id, caller_substs) = self.get_call_data(func);
 2332                 // A call whose span comes from a macro expansion (e.g. the
 2333                 // `core::panicking::panic` call inside a failing `assert!`)
 2334                 // was not written by the user, so reporting "precondition
 2335                 // might not hold" at the expanded fragment is confusing: no
 2336                 // visible call exists there. Report at the macro invocation
 2337                 // that produced the call, naming the actually-called function.
 2338                 let verification_error = if span.from_expansion() {
 2339                     let name = self.vcx.tcx().def_path_str(func_def_id);
 2340                     let message = format!(
 2341                         "precondition of `{name}` (called by this macro expansion) might not hold"
 2342                     );
 2343                     PrustiError::verification(message, span.source_callsite().into())
 2344                 } else {
 2345                     PrustiError::verification("precondition might not hold.", span.into())
 2346                 };
 2347 
 2348                 let dest = self
 2349                     .encode_place(Place::from(*destination))?
 2350                     .expr
 2351                     .expect_predicate();
 2352                 self.vcx.with_span(span, |vcx| {
 2353                     let pure = self.pure_call_result(func_def_id, caller_substs, args, span)?;
 2354                     if let Some(pure) = pure {
 2355                         self.wandless_calls.insert(self.current_block.unwrap());
 2356                         let return_ty = destination.ty(self.local_decls, self.vcx.tcx()).ty;
 2357                         let return_ty_use = self.ty_use_impure(return_ty);
 2358                         let assign_stmt = return_ty_use.apply_method_assign(self.vcx, dest, pure);
 2359                         self.stmt(assign_stmt);
 2360                         // The call returned a value of the return type, therefore
 2361                         // it is inhabited.
 2362                         self.stmt(vcx.mk_inhale_stmt(return_ty_use.inhabited()));
 2363                     } else {
 2364                         let func_out = self.deps.require_dep_spanned::<encoders::MethodCallEnc>(
 2365                             CallTaskDescription::new(self.def_id, caller_substs, func_def_id),
 2366                             span,
 2367                         )?;
 2368 
 2369                         let method_in = args
 2370                             .iter()
 2371                             .map(|arg| self.encode_operand(&arg.node))
 2372                             .collect::<Result<Vec<_>, _>>()?;
 2373 
 2374                         let call = func_out.call(method_in, dest);
 2375 
 2376                         let label_pre = self.new_label("pre");
 2377                         vcx.handle_error(
 2378                             "call.precondition:assertion.false",
 2379                             move |reason_span_opt| {
 2380                                 let mut error = verification_error.clone();
 2381                                 if let Some(reason_span) = reason_span_opt {
 2382                                     error.add_note_mut(
 2383                                         "the failing precondition is here",
 2384                                         Some(reason_span.into()),
 2385                                     );
 2386                                 }
 2387                                 Some(vec![error])
 2388                             },
 2389                         );
 2390                         self.stmts(call);
 2391                         let label_post = self.new_label("post");
 2392                         self.call_labels
 2393                             .insert(location.block, (label_pre, label_post));
 2394                     }
 2395                     Ok(())
 2396                 })?;
 2397 
 2398                 match *target {
 2399                     Some(target) => self.goto_single_succ(location, target)?,
 2400                     None => {
 2401                         // TODO: detect panic causes, adjust message accordingly
 2402                         self.vcx.with_span(span, |vcx| {
 2403                             vcx.handle_error("exhale.failed:assertion.false", move |_| {
 2404                                 Some(vec![PrustiError::verification(
 2405                                     "unreachable statement might be reached",
 2406                                     span.into(),
 2407                                 )])
 2408                             });
 2409                             self.stmt(self.vcx.mk_exhale_stmt(self.vcx.mk_bool::<false>()));
 2410                             self.vcx.mk_assume_false_stmt()
 2411                         })
 2412                     }
 2413                 }
 2414             }
 2415             mir::TerminatorKind::Assert {
 2416                 cond,
 2417                 expected,
 2418                 msg,
 2419                 target,
 2420                 ..
 2421             } => {
 2422                 let enc = self
 2423                     .encode_operand_snap(cond, &None)?
 2424                     .downcast_ty::<vir::Bool>();
 2425                 let expected = self
 2426                     .vcx
 2427                     .mk_const_expr(vir::ConstData::Bool(*expected))
 2428                     .downcast_ty();
 2429                 let assert = self.vcx.mk_eq_expr(enc, expected);
 2430                 if let Some(error_msg) = assert_error_message(msg) {
 2431                     self.vcx.with_span(span, |vcx| {
 2432                         vcx.handle_error("exhale.failed:assertion.false", move |_| {
 2433                             Some(vec![PrustiError::verification(error_msg, span.into())])
 2434                         });
 2435                         self.stmt(self.vcx.mk_exhale_stmt(assert));
 2436                     });
 2437                 } else {
 2438                     self.stmt(self.vcx.mk_inhale_stmt(assert));
 2439                 }
 2440                 // The check is the terminator's main effect, emitted before
 2441                 // the deferred `PostMain` re-pack, which may fold the place
 2442                 // the condition projects into.
 2443                 self.goto_single_succ(location, *target)?
 2444             }
 2445             mir::TerminatorKind::Unreachable => self.vcx.with_span(span, |vcx| {
 2446                 vcx.handle_error("exhale.failed:assertion.false", move |_| {
 2447                     Some(vec![PrustiError::verification(
 2448                         "unreachable statement might be reached",
 2449                         span.into(),
 2450                     )])
 2451                 });
 2452                 self.stmt(self.vcx.mk_exhale_stmt(self.vcx.mk_bool::<false>()));
 2453                 self.vcx.mk_assume_false_stmt()
 2454             }),
 2455 
 2456             mir::TerminatorKind::UnwindResume | mir::TerminatorKind::UnwindTerminate(..) => {
 2457                 self.vcx.with_span(span, |vcx| {
 2458                     vcx.handle_error("exhale.failed:assertion.false", move |_| {
 2459                         Some(vec![PrustiError::unsupported(
 2460                             "unwind paths are not supported",
 2461                             span.into(),
 2462                         )])
 2463                     });
 2464                     self.stmt(self.vcx.mk_exhale_stmt(self.vcx.mk_bool::<false>()));
 2465                     self.vcx.mk_assume_false_stmt()
 2466                 })
 2467             }
 2468 
 2469             mir::TerminatorKind::TailCall { .. } => {
 2470                 return Err(EncodeFullError::unsupported(
 2471                     "tail calls are not supported".to_string(),
 2472                     vec![span],
 2473                 ));
 2474             }
 2475             mir::TerminatorKind::Yield { .. } => {
 2476                 return Err(EncodeFullError::unsupported(
 2477                     "yield statements are not supported".to_string(),
 2478                     vec![span],
 2479                 ));
 2480             }
 2481             mir::TerminatorKind::CoroutineDrop => {
 2482                 return Err(EncodeFullError::unsupported(
 2483                     "coroutines are not supported".to_string(),
 2484                     vec![span],
 2485                 ));
 2486             }
 2487             mir::TerminatorKind::InlineAsm { .. } => {
 2488                 return Err(EncodeFullError::unsupported(
 2489                     "inline assembly is not supported".to_string(),
 2490                     vec![span],
 2491                 ));
 2492             }
 2493         };
 2494         assert!(self.current_terminator.replace(terminator).is_none());
 2495         Ok(())
 2496     }

 2498     fn pure_call_result(
 2499         &mut self,
 2500         func_def_id: DefId,
 2501         caller_substs: ty::GenericArgsRef<'vir>,
 2502         args: &[Spanned<mir::Operand<'vir>>],
 2503         span: Span,
 2504     ) -> EncodeResult<'vir, Option<vir::ExprSnap<'vir>>, E> {
 2505         // The bodiless `ptr_metadata` intrinsic is only lowered to
 2506         // `UnOp::PtrMetadata` in optimized MIR; do the lowering here.
 2507         let intrinsic = self.vcx.tcx().intrinsic(func_def_id);
 2508         let intrinsic = intrinsic.and_then(RustcIntrinsic::from_intrinsic);
 2509         Ok(if let Some(intrinsic) = intrinsic {
 2510             Some(self.encode_intrinsic(intrinsic, caller_substs, args, &None)?)
 2511         } else if let Some(builtin) = PrustiBuiltin::new(func_def_id, self.gargs(caller_substs)) {
 2512             // A `prusti_contracts` builtin used in executable code
 2513             // (e.g. `Int::from(2) + Int::from(3)`): encode it with
 2514             // the shared operand-based encoding and assign the
 2515             // resulting ghost snapshot into the destination. The
 2516             // spec-only builtins (those gated behind the `prusti`
 2517             // feature) are rejected, except in ghost code (the inline
 2518             // bodies of `ghost!` blocks), where all non-`Spec` builtins
 2519             // are allowed.
 2520             let in_ghost_code = self
 2521                 .spec_blocks
 2522                 .ghost
 2523                 .code
 2524                 .contains(&self.current_block.unwrap());
 2525             if builtin.is_spec_only()
 2526                 && (!in_ghost_code || matches!(builtin, PrustiBuiltin::Spec(_)))
 2527             {
 2528                 return Err(EncodeFullError::unsupported(
 2529                     format!(
 2530                         "`prusti_contracts` builtin {builtin:?} cannot be used in executable code"
 2531                     ),
 2532                     vec![span],
 2533                 ));
 2534             }
 2535             Some(self.encode_prusti_builtin(
 2536                 builtin,
 2537                 func_def_id,
 2538                 self.gargs(caller_substs),
 2539                 args,
 2540                 span,
 2541                 &None,
 2542             )?)
 2543         } else {
 2544             None
 2545         })
 2546     }
