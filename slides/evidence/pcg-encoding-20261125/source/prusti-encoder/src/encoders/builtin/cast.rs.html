   22 /// Encodes the builtin MIR cast operations (e.g. `IntToInt`, `PointerCoercion`)
   23 /// as Viper functions with the correct semantics. The unsizing coercions and the
   24 /// casts between raw pointers are applied to generic refs (the inner type is a
   25 /// parameter), so their meaning is given by axioms: the general ones of
   26 /// `MetadataCastEnc` and, for unsizing, those for concrete types in
   27 /// `MetadataCastAxiomEnc` (e.g. `&[T; N] -> &[T]`).
   28 /// Also returns a side-effecting `unsize`/`undo` method for coercing the inner
   29 /// referent (i.e. an `p_Param` to `p_Param` with different type).
   30 ///
   31 /// Use `MirBuiltinUseCastEnc` to encode a cast in a specific context (with concrete
   32 /// type arguments).
   33 pub(super) struct MirBuiltinCastEnc;
   34 
   35 #[derive(Debug, Clone, Copy, Hash, PartialEq, Eq)]
   36 pub(super) struct MirBuiltinCastTask<'vir> {
   37     pub(super) result_ty: RustTy<'vir>,
   38     pub(super) kind: mir::CastKind,
   39     pub(super) operand_ty: RustTy<'vir>,
   40 }

   42 /// The types a cast depends on beyond the operand and result types that key
   43 /// its task. Its function and methods take them as type arguments,
   44 /// which `MirBuiltinUseCastEnc` supplies at each use.
   45 #[derive(Debug, Clone, Copy)]
   46 pub(super) enum CastTyArgs {
   47     /// No type arguments: the cast depends only on the operand and result
   48     /// types (e.g. an integer cast).
   49     None,
   50     /// The operand's and the result's referent types `[U, V]`, which give the
   51     /// metadata after a pointer cast (via `metadata_cast`) and the value of an
   52     /// unsizing coercion (e.g. `&[T; N] -> &[T]`).
   53     Referents,
   54 }
   55 
   56 impl CastTyArgs {
   57     /// The type parameters that the cast's function and methods declare.
   58     fn params<'vir>(self, vcx: &'vir vir::VirCtxt<'vir>) -> GParams<'vir> {
   59         match self {
   60             CastTyArgs::None => GParams::empty(),
   61             CastTyArgs::Referents => {
   62                 let param = |idx, name| {
   63                     ty::Ty::new_param(vcx.tcx(), idx, symbol::Symbol::intern(name)).into()
   64                 };
   65                 GParams::empty_env(vcx.tcx().mk_args(&[param(0, "U"), param(1, "V")]))
   66             }
   67         }
   68     }
   69 }

   71 /// The result of encoding a builtin cast.
   72 #[derive(Debug, Clone, Copy)]
   73 pub(super) struct MirBuiltinCastOutput<'vir> {
   74     /// The function from the operand snapshot to the result snapshot.
   75     pub(super) cast: FunctionIdn<'vir, (vir::CSnap, vir::ManyTyVal, vir::ManyCSnap), vir::CSnap>,
   76     /// For a `&mut` unsizing coercion, the side-effecting methods that
   77     /// transfer permission between the original and the coerced reference.
   78     pub(super) unsize: Option<vir::MethodIdn<'vir, (vir::CSnap, vir::ManyTyVal, vir::ManyCSnap)>>,
   79     pub(super) undo: Option<vir::MethodIdn<'vir, (vir::CSnap, vir::ManyTyVal, vir::ManyCSnap)>>,
   80     /// The type arguments that `cast`, `unsize` and `undo` take.
   81     pub(super) ty_args: CastTyArgs,
   82 }

   84 /// The Viper definitions produced for a builtin cast: always the cast function,
   85 /// plus the side-effecting `unsize`/`undo` methods for `&mut` unsize coercions.
   86 #[derive(Debug, Clone, Copy)]
   87 pub(super) struct MirBuiltinCastLocal<'vir> {
   88     cast: vir::Function<'vir>,
   89     unsize: Option<vir::Method<'vir>>,
   90     undo: Option<vir::Method<'vir>>,
   91     /// Whether this encodes something the user writes rather than something
   92     /// rustc or the encoding adds, which the simplifier may fold away (see
   93     /// `add_user_adt`).
   94     user_level: bool,
   95 }

   97 impl TaskEncoder for MirBuiltinCastEnc {
   98     task_encoder::encoder_cache!(MirBuiltinCastEnc);
   99     const ENCODER_NAME: &'static str = "MIR builtin cast encoder";
  100 
  101     type TaskDescription<'vir> = MirBuiltinCastTask<'vir>;
  102 
  103     type OutputFullDependency<'vir> = MirBuiltinCastOutput<'vir>;
  104     type OutputFullLocal<'vir> = MirBuiltinCastLocal<'vir>;

  106     fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
  107         *task
  108     }
  109 
  110     fn do_encode_full<'vir>(
  111         task_key: &Self::TaskKey<'vir>,
  112         deps: &mut TaskEncoderDependencies<'vir, Self>,
  113     ) -> EncodeFullResult<'vir, Self> {
  114         deps.emit_output_ref(*task_key, ())?;
  115         let MirBuiltinCastTask {
  116             result_ty,
  117             kind,
  118             operand_ty,
  119         } = *task_key;
  120         let op_ty = RustTyDecomposition::identity(operand_ty);
  121         let op_ty = deps.require_dep::<TyUsePureEnc>(op_ty)?;
  122         let op_ty_snap = op_ty.snapshot.downcast_ty::<vir::CSnap>();
  123         let res_ty = RustTyDecomposition::identity(result_ty);
  124         let res_ty = deps.require_dep::<TyUsePureEnc>(res_ty)?;
  125         let res_ty_snap = res_ty.snapshot.downcast_ty::<vir::CSnap>();
  126 
  127         let (name, ty_args, user_level) = match kind {
  128             mir::CastKind::PointerCoercion(ty::adjustment::PointerCoercion::Unsize, ..) => {
  129                 ("unsize", CastTyArgs::Referents, false)
  130             }
  131             mir::CastKind::IntToInt => ("i2i", CastTyArgs::None, true),
  132             mir::CastKind::PtrToPtr => ("p2p", CastTyArgs::Referents, true),
  133             mir::CastKind::IntToFloat => ("i2f", CastTyArgs::None, true),
  134             mir::CastKind::FloatToInt => ("f2i", CastTyArgs::None, true),
  135             mir::CastKind::PointerCoercion(
  136                 ty::adjustment::PointerCoercion::MutToConstPointer,
  137                 ..,
  138             ) => ("mut2const", CastTyArgs::None, false),
  139             mir::CastKind::PointerCoercion(coercion, _) => {
  140                 return Err(EncodeFullError::unsupported(
  141                     format!("the pointer coercion `{coercion:?}`"),
  142                     Vec::new(),
  143                 ));
  144             }
  145             other => {
  146                 return Err(EncodeFullError::unsupported(
  147                     format!("the cast `{other:?}`"),
  148                     Vec::new(),
  149                 ));
  150             }
  151         };
  152         // The referent types are not part of the task key: a cast that needs
  153         // them takes them as type arguments (`CastTyArgs`), so the name only
  154         // uses the pointer types and one task serves every referent.
  155         let name = vir::with_vcx(|vcx| {
  156             vir::vir_format_identifier!(
  157                 vcx,
  158                 "mir_cast_{name}_{}_to_{}",
  159                 operand_ty.name(),
  160                 result_ty.name(),
  161             )
  162         });
  163         let gparams = vir::with_vcx(|vcx| ty_args.params(vcx));
  164         let params = deps.require_dep::<GenericParamsEnc>(gparams)?;
  165         vir::with_vcx(|vcx| {
  166             let arg_decl = vcx.mk_local_decl("arg", op_ty_snap);
  167             let arg_ex = vcx.mk_local_ex(arg_decl);
  168             // The body of the cast function, and the `unsize`/`undo` methods
  169             // with their identifiers if the cast has them.
  170             let (expr, methods) = match kind {
  171                 mir::CastKind::IntToInt => {
  172                     let e_op_ty = op_ty.expect_primitive();
  173                     let e_res_ty = res_ty.expect_primitive();
  174                     let result_prim = *result_ty.expect_primitive();
  175                     let operand_prim = *operand_ty.expect_primitive();
  176 
  177                     // An integer `as` cast never panics: when every value of the
  178                     // source type is representable in the target the value is
  179                     // preserved unchanged; otherwise it is truncated to the target
  180                     // width and reinterpreted with the target's signedness. Either
  181                     // way any input is valid, so the cast function needs NO
  182                     // precondition.
  183                     let wrapped = if matches!(operand_prim.kind(), ty::TyKind::Bool) {
  184                         // `bool as <int>` is `b ? 1 : 0`; always lossless.
  185                         vcx.mk_ternary_expr(
  186                             arg_ex.downcast_ty(),
  187                             vcx.mk_int::<1>(),
  188                             vcx.mk_int::<0>(),
  189                         )
  190                         .upcast_ty()
  191                     } else {
  192                         let (to_bits, to_signed) = vcx.get_int_data(result_prim);
  193                         let (from_bits, from_signed) = vcx.get_int_data(operand_prim);
  194 
  195                         let arg_prim = e_op_ty.snap_to_prim(arg_ex);
  196 
  197                         let lossless = match (from_signed, to_signed) {
  198                             (false, false) | (true, true) => from_bits <= to_bits,
  199                             (false, true) => from_bits < to_bits, // one bit reserved for the sign
  200                             (true, false) => false,               // a negative source never fits
  201                         };
  202                         if lossless {
  203                             arg_prim
  204                         } else {
  205                             // Truncate to the target width and reinterpret with its
  206                             // signedness (two's complement); the shared helper wraps
  207                             // `((x [+ 2^(N-1)]) mod 2^N) [- 2^(N-1)]`.
  208                             vcx.get_wrapped_val(arg_prim.downcast_ty(), result_prim)
  209                                 .upcast_ty()
  210                         }
  211                     };
  212                     let expr = e_res_ty.prim_to_snap(wrapped);
  213 
  214                     (expr, None)
  215                 }
  216                 mir::CastKind::PtrToPtr => {
  217                     let e_op_ty = op_ty.expect_raw();
  218                     let e_res_ty = res_ty.expect_raw();
  219                     let metadata_cast = deps.require_dep::<MetadataCastEnc>(())?;
  220                     let metadata = metadata_cast(
  221                         e_op_ty.metadata_access(arg_ex).downcast_ty(),
  222                         params.ty_exprs()[0],
  223                         params.ty_exprs()[1],
  224                     );
  225                     let expr =
  226                         e_res_ty.prim_to_snap(e_op_ty.address_access(arg_ex), metadata.upcast_ty());
  227                     (expr, None)
  228                 }
  229                 // `*mut T` to `*const T` keeps the address and metadata.
  230                 mir::CastKind::PointerCoercion(
  231                     ty::adjustment::PointerCoercion::MutToConstPointer,
  232                     _,
  233                 ) => {
  234                     let e_op_ty = op_ty.expect_raw();
  235                     let e_res_ty = res_ty.expect_raw();
  236                     let expr = e_res_ty.prim_to_snap(
  237                         e_op_ty.address_access(arg_ex),
  238                         e_op_ty.metadata_access(arg_ex),
  239                     );
  240 
  241                     (expr, None)
  242                 }
  243                 mir::CastKind::IntToFloat => {
  244                     let e_op_ty = op_ty.expect_primitive();
  245                     let (bits, signed) = vcx.get_int_data(*operand_ty.expect_primitive());
  246                     // Convert at the width of the source type, which keeps `int2bv` cheaper for Z3.
  247                     let ty::Float(float) = *result_ty.expect_primitive().kind() else {
  248                         unreachable!()
  249                     };
  250                     let conv = deps
  251                         .require_dep::<FloatBitVecConvEnc>((float, BitVecSize::from_bits(bits)))?;
  252                     let arg_prim = e_op_ty.snap_to_prim(arg_ex);
  253                     let expr = if signed {
  254                         (conv.from_sint)(arg_prim)
  255                     } else {
  256                         (conv.from_uint)(arg_prim)
  257                     };
  258 
  259                     (expr, None)
  260                 }
  261                 mir::CastKind::FloatToInt => {
  262                     let e_op_ty = op_ty.expect_float();
  263                     let e_res_ty = res_ty.expect_primitive();
  264                     let result_prim = *result_ty.expect_primitive();
  265                     let ty::Float(float) = *operand_ty.expect_primitive().kind() else {
  266                         unreachable!()
  267                     };
  268                     let (bits, signed) = vcx.get_int_data(result_prim);
  269                     // Convert at the width of the target type, so that every
  270                     // in-range value is representable.
  271                     let conv = deps
  272                         .require_dep::<FloatBitVecConvEnc>((float, BitVecSize::from_bits(bits)))?;
  273                     let float_const = |bits: u128| {
  274                         (e_op_ty.prim_to_snap)(vcx.mk_const_expr(ConstData::Int(bits)))
  275                     };
  276 
  277                     // `as` rounds towards zero, which the `RTZ` conversions
  278                     // already do. They are unspecified for values whose rounded
  279                     // value is out of range, so saturate first. The bounds
  280                     // `-2^(N-1)`/`0` and `2^(N-1)`/`2^N` are integers, so
  281                     // comparing the untruncated value against them gives the
  282                     // same result as comparing the truncated one. They are also
  283                     // powers of two and therefore exact (or infinite if out of
  284                     // the float's range, in which case only the infinities reach
  285                     // them). The bounds are compared as floats: going via
  286                     // `fp.to_real` makes symbolic casts unprovable with Z3's
  287                     // arithmetic solver.
  288                     let (lower, upper, in_range) = if signed {
  289                         (
  290                             float_pow2_bits(float, bits - 1, true),
  291                             float_pow2_bits(float, bits - 1, false),
  292                             (conv.bitvec.sbv_to_int)((conv.to_sbv)(arg_ex)),
  293                         )
  294                     } else {
  295                         (
  296                             0,
  297                             float_pow2_bits(float, bits, false),
  298                             (conv.bitvec.ubv_to_int)((conv.to_ubv)(arg_ex)),
  299                         )
  300                     };
  301                     let expr = vcx.mk_ternary_expr(
  302                         (e_op_ty.fp_is_nan)(arg_ex),
  303                         e_res_ty.prim_to_snap(vcx.mk_int::<0>().upcast_ty()),
  304                         vcx.mk_ternary_expr(
  305                             (e_op_ty.fp_leq)(arg_ex, float_const(lower)),
  306                             e_res_ty.prim_to_snap(vcx.get_min_int(result_prim).upcast_ty()),
  307                             vcx.mk_ternary_expr(
  308                                 (e_op_ty.fp_geq)(arg_ex, float_const(upper)),
  309                                 e_res_ty.prim_to_snap(vcx.get_max_int(result_prim).upcast_ty()),
  310                                 e_res_ty.prim_to_snap(in_range.upcast_ty()),
  311                             ),
  312                         ),
  313                     );
  314 
  315                     (expr, None)
  316                 }
  317                 mir::CastKind::PointerCoercion(ty::adjustment::PointerCoercion::Unsize, _) => {
  318                     // Unsizing preserves the (wide) reference snapshot type and only
  319                     // rewrites the pointer metadata for the new referent. The cast
  320                     // (like the methods below) is parameterized by the operand and
  321                     // result referent types `U`, `V` (`CastTyArgs::Referents`),
  322                     // which `MirBuiltinUseCastEnc` supplies from each reference's
  323                     // referent arg.
  324                     assert_eq!(op_ty_snap, res_ty_snap);
  325                     let u = params.ty_exprs()[0];
  326                     let v = params.ty_exprs()[1];
  327                     let value_cast = deps.require_dep::<ValueCastEnc>(())?;
  328                     let metadata_cast = deps.require_dep::<MetadataCastEnc>(())?;
  329                     let value_cast = |value| value_cast(value, u, v);
  330                     let metadata_cast = |metadata| metadata_cast(metadata, u, v);
  331                     let (is_mut, expr) = match &op_ty.specifics {
  332                         TySpecifics::ImmRef(data) => {
  333                             let res_data = res_ty.expect_immref();
  334                             let expr = res_data.prim_to_snap(
  335                                 data.addr_access(arg_ex),
  336                                 metadata_cast(data.metadata_access(arg_ex).downcast_ty())
  337                                     .upcast_ty(),
  338                                 value_cast(data.value_access(arg_ex).downcast_ty()).upcast_ty(),
  339                             );
  340                             (false, expr)
  341                         }
  342                         TySpecifics::MutRef(data) => {
  343                             let res_data = res_ty.expect_mutref();
  344                             let expr = res_data.prim_to_snap(
  345                                 data.deref_access(arg_ex),
  346                                 metadata_cast(data.metadata_access(arg_ex).downcast_ty())
  347                                     .upcast_ty(),
  348                                 value_cast(data.value_access(arg_ex).downcast_ty()).upcast_ty(),
  349                             );
  350                             (true, expr)
  351                         }
  352                         // A `Box` (e.g. `Box<[T; N]> -> Box<[T]>`): rebuild the
  353                         // snapshot with the coerced pointer metadata and value
  354                         // (the address is unchanged). Unlike `&mut`, no
  355                         // side-effecting methods are needed: the box owns its
  356                         // referent, so the whole predicate is exhaled/inhaled
  357                         // by the surrounding assignment.
  358                         TySpecifics::StructLike(data)
  359                             if operand_ty.special == RustTySpecial::Box =>
  360                         {
  361                             assert_eq!(operand_ty, result_ty);
  362                             let expr = data.box_unsize_snap(
  363                                 arg_ex,
  364                                 metadata_cast(data.box_metadata_access(arg_ex).downcast_ty())
  365                                     .upcast_ty(),
  366                                 value_cast(data.box_value_access(arg_ex).downcast_ty()).upcast_ty(),
  367                             );
  368                             (false, expr)
  369                         }
  370                         _ => {
  371                             return Err(EncodeFullError::unsupported(
  372                                 format!(
  373                                     "unsizing of `{}` (only references and `Box` can be unsized)",
  374                                     operand_ty.name()
  375                                 ),
  376                                 Vec::new(),
  377                             ));
  378                         }
  379                     };
  380                     // For `&mut` coercions, generate the side-effecting
  381                     // `unsize`/`undo` methods that move the (generic) `p_Param`
  382                     // predicate of the referent between the operand and result type
  383                     // at the same address (the coercion preserves it). The methods
  384                     // are declared over the same `[U, V]` type params (`params`),
  385                     // which `MirBuiltinUseCastEnc` instantiates with the operand
  386                     // referent `U` and result referent `V`.
  387                     // `new_param_ty(i).decompose(gparams)` builds a `Param`
  388                     // decomposition whose arg is `gparams`' `i`th type
  389                     // variable, so `ref_to_pred`/`ref_to_snap` yield
  390                     // `p_Param(addr, U)` / `p_Param(addr, V)` directly; the referent
  391                     // snapshot is transferred via `value_cast` (whose concrete-type
  392                     // axioms are added in `MirBuiltinUseCastEnc`).
  393                     let methods = if is_mut {
  394                         let u_impure = deps.require_dep::<TyUseImpureEnc>(
  395                             LazyRustTy::new_param_ty(0).decompose(gparams),
  396                         )?;
  397                         let v_impure = deps.require_dep::<TyUseImpureEnc>(
  398                             LazyRustTy::new_param_ty(1).decompose(gparams),
  399                         )?;
  400 
  401                         let src_mutref = op_ty.expect_mutref();
  402                         let src_decl = vcx.mk_local_decl("src", op_ty_snap);
  403                         let src_ex = vcx.mk_local_ex(src_decl);
  404                         let addr = src_mutref.deref_access(src_ex);
  405 
  406                         let u_pred = u_impure.ref_to_pred(vcx, addr, None);
  407                         let v_pred = v_impure.ref_to_pred(vcx, addr, None);
  408                         let u_snap = u_impure.ref_to_snap(addr).downcast_ty::<vir::PSnap>();
  409                         let v_snap = v_impure.ref_to_snap(addr).downcast_ty::<vir::PSnap>();
  410 
  411                         // `unsize`: operand referent `U` -> result referent `V`; the
  412                         // new `V` value is `value_cast(old(U), U, V)`. `undo` is the
  413                         // *reverse* equation: the old `V` value is `value_cast` of the
  414                         // recovered `U` value, `old(V) == value_cast(U, U, V)`. Both
  415                         // use the same `value_cast` direction (`U, V`), so a single
  416                         // axiom per coercion (added in `MirBuiltinUseCastEnc`) suffices.
  417                         let unsize_value =
  418                             vcx.mk_eq_expr(v_snap, value_cast(vcx.mk_old_expr(u_snap)));
  419                         let undo_value =
  420                             vcx.mk_eq_expr(vcx.mk_old_expr(v_snap), value_cast(u_snap));
  421 
  422                         let unsize_idn = MethodIdn::new(
  423                             vir::vir_format_identifier!(
  424                                 vcx,
  425                                 "mir_unsize_{}_to_{}",
  426                                 operand_ty.name(),
  427                                 result_ty.name()
  428                             ),
  429                             (op_ty_snap, params.ty_args(), params.const_args()),
  430                         );
  431                         let undo_idn = MethodIdn::new(
  432                             vir::vir_format_identifier!(
  433                                 vcx,
  434                                 "mir_undo_unsize_{}_to_{}",
  435                                 operand_ty.name(),
  436                                 result_ty.name()
  437                             ),
  438                             (op_ty_snap, params.ty_args(), params.const_args()),
  439                         );
  440                         let unsize_method = vcx.mk_method(
  441                             unsize_idn,
  442                             (src_decl, params.ty_decls(), params.const_decls()),
  443                             &[],
  444                             vcx.alloc_slice(&[u_pred]),
  445                             vcx.alloc_slice(&[v_pred, unsize_value]),
  446                             None,
  447                         );
  448                         let undo_method = vcx.mk_method(
  449                             undo_idn,
  450                             (src_decl, params.ty_decls(), params.const_decls()),
  451                             &[],
  452                             vcx.alloc_slice(&[v_pred]),
  453                             vcx.alloc_slice(&[u_pred, undo_value]),
  454                             None,
  455                         );
  456                         Some(((unsize_method, unsize_idn), (undo_method, undo_idn)))
  457                     } else {
  458                         None
  459                     };
  460                     (expr, methods)
  461                 }
  462                 _ => unreachable!("unsupported cast kinds are rejected above"),
  463             };
  464 
  465             let cast_idn = FunctionIdn::new(
  466                 name,
  467                 (op_ty_snap, params.ty_args(), params.const_args()),
  468                 res_ty_snap,
  469             );
  470             let cast = vcx.mk_function(
  471                 cast_idn,
  472                 (arg_decl, params.ty_decls(), params.const_decls()),
  473                 &[],
  474                 &[],
  475                 None,
  476                 Some(expr),
  477             );
  478             let (unsize, undo) = methods.unzip();
  479             let (unsize, unsize_idn) = unsize.unzip();
  480             let (undo, undo_idn) = undo.unzip();
  481             Ok((
  482                 MirBuiltinCastLocal {
  483                     cast,
  484                     unsize,
  485                     undo,
  486                     user_level,
  487                 },
  488                 MirBuiltinCastOutput {
  489                     cast: cast_idn,
  490                     unsize: unsize_idn,
  491                     undo: undo_idn,
  492                     ty_args,
  493                 },
  494             ))
  495         })
  496     }

  498     fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
  499         for local in Self::all_outputs_local(program) {
  500             if local.user_level {
  501                 program.add_user_function(local.cast);
  502             } else {
  503                 program.add_function(local.cast);
  504             }
  505             if let Some(unsize) = local.unsize {
