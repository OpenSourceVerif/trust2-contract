   24 #[derive(Debug, Clone)]
   25 pub struct MethodCallEncOutput<'vir> {
   26     method: MethodEncOutputRef<'vir>,
   27     ty_args: GArgsTy<'vir>,
   28     inputs: Vec<GArgCaster<'vir, Impure>>,
   29     output: GArgCaster<'vir, Impure>,

   32 impl<'vir> MethodCallEncOutput<'vir> {
   33     pub fn call(
   34         &self,
   35         mut args: Vec<vir::ExprRef<'vir>>,
   36         ret: vir::ExprRef<'vir>,
   37     ) -> Vec<vir::Stmt<'vir>> {
   38         assert_eq!(self.inputs.len(), args.len());
   39         let generics = args.iter().zip(self.inputs.iter());
   40         let mut stmts: Vec<_> = generics
   41             .filter_map(|(arg, caster)| caster.cast_to_callee_ctx(arg))
   42             .collect();
   43 
   44         args.insert(0, ret);
   45         let call = (self.method.method_ref)(&args, self.ty_args.get_ty(), self.ty_args.get_const())
   46             .alloc();
   47         stmts.push(call);
   48 
   49         let result = self.output.cast_to_caller_ctx(ret);
   50         if let Some(result) = result {
   51             stmts.push(result);
   52         }
   53         stmts
   54     }
   55 }

   57 impl TaskEncoder for MethodCallEnc {
   58     task_encoder::encoder_cache!(MethodCallEnc);
   59     type TaskDescription<'tcx> = CallTaskDescription<'tcx>;
   60     type OutputFullDependency<'vir> = MethodCallEncOutput<'vir>;
   61 
   62     const ENCODER_NAME: &'static str = "method call encoder";
   63 
   64     fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
   65         *task
   66     }
   67 
   68     fn do_encode_full<'vir>(
   69         task_key: &Self::TaskKey<'vir>,
   70         deps: &mut TaskEncoderDependencies<'vir, Self>,
   71     ) -> EncodeFullResult<'vir, Self> {
   72         deps.emit_output_ref(*task_key, ())?;
   73         let (callee_def_id, assoc_enc) = task_key.trait_call(deps)?;
   74         let method_ref = if let Some(assoc_enc) = assoc_enc {
   75             MethodEncOutputRef {
   76                 method_ref: assoc_enc.call_stub_impure,
   77             }
   78         } else {
   79             let method_ref = deps.require_ref::<MethodEnc>(task_key.callee)?;
   80             // The method of a callee whose wands cannot be generated is not
   81             // emitted (see `MethodEnc`), so a call to it cannot be encoded.
   82             let wands_task = WandEncTask {
   83                 data: FunctionData::new(task_key.callee),
   84             };
   85             deps.require_dep::<WandEnc>(wands_task)?;
   86             method_ref
   87         };
   88         let signature = RustSignature::new(callee_def_id);
   89         let ty_args = deps.require_dep::<GArgsTyEnc>(task_key.gargs)?;
   90         let inputs = signature
   91             .inputs
   92             .iter()
   93             .map(|ty| {
   94                 let normalized = ty.decompose_compare_normalize(signature.gparams, task_key.gargs);
   95                 deps.require_dep::<GArgsCastEnc<Impure>>(normalized)
   96             })
   97             .collect::<Result<Vec<_>, _>>()?;
   98         let normalized = signature
   99             .output
  100             .decompose_compare_normalize(signature.gparams, task_key.gargs);
  101         let output = deps.require_dep::<GArgsCastEnc<Impure>>(normalized)?;
  102         Ok((
  103             (),
  104             MethodCallEncOutput {
  105                 method: method_ref,
  106                 ty_args,
  107                 inputs,
  108                 output,
  109             },
  110         ))
  111     }
  112 
  113     fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
  114         MethodEnc::emit_outputs(program);
  115     }
  116 }

  134 impl TaskEncoder for MethodEnc {
  135     task_encoder::encoder_cache!(MethodEnc);
  136     const ENCODER_NAME: &'static str = "method encoder";
  137     type TaskDescription<'tcx> = DefId;
  138 
  139     type OutputRef<'vir> = MethodEncOutputRef<'vir>;
  140     type OutputFullLocal<'vir> = MethodEncOutput<'vir>;
  141 
  142     fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
  143         *task
  144     }
  145 
  146     fn do_encode_full<'vir>(
  147         task_key: &Self::TaskKey<'vir>,
  148         deps: &mut TaskEncoderDependencies<'vir, Self>,
  149     ) -> EncodeFullResult<'vir, Self> {
  150         let def_id = *task_key;
  151         vir::with_vcx(|vcx| {
  152             let span = vcx.tcx().def_span(def_id);
  153 
  154             let arg_defs = deps.require_ref_spanned::<MirLocalDefEnc>(
  155                 MirLocalDefEncTask::Local {
  156                     def_id,
  157                     all_locals: false,
  158                 },
  159                 span,
  160             )?;
  161 
  162             // Argument count for the Viper method:
  163             // - one (`Ref`) for the return place;
  164             // - one (`Ref`) for each MIR argument.
  165             //
  166             // Note that the return place is modelled as an argument of the
  167             // Viper method. This corresponds to an execution model where the
  168             // method can return data to the caller without a copy--it directly
  169             // modifies a place provided by the caller.
  170             //
  171             // TODO: type parameters: for generic methods we will want to pass
  172             //   values of type `Type` as well`
  173             let arg_count = arg_defs.arg_count + 1;
  174 
  175             // Create the identifier and use it as an output ref. This is what
  176             // is used when other methods call this one.
  177             let method_name =
  178                 vir::vir_format_identifier!(vcx, "m_{}", vir::ViperIdent::from_def_id(vcx, def_id));
  179             let ref_args = vcx.alloc_slice(&vec![vir::TYPE_REF; arg_count]);
  180             let params = GParams::from(def_id);
  181             let generics = deps.require_dep_spanned::<GenericParamsEnc>(params, span)?;
  182             let method_ref = MethodIdn::new(
  183                 method_name,
  184                 (ref_args, generics.ty_args(), generics.const_args()),
  185             );
  186             deps.emit_output_ref(def_id, MethodEncOutputRef { method_ref })?;
  187 
  188             let arg_defs = deps.require_dep_spanned::<MirLocalDefEnc>(
  189                 MirLocalDefEncTask::Local {
  190                     def_id,
  191                     all_locals: false,
  192                 },
  193                 span,
  194             )?;
  195 
  196             // Method contract. We will need to emit pre- and postconditions for
  197             // the permissions, the functional spec, and (in the postcondition)
  198             // wands in case of a reborrowing function.
  199             let mut pres = Vec::new();
  200             let mut posts = Vec::new();
  201             let spec = deps.require_dep_spanned::<MirSpecEnc>(
  202                 (def_id, def_id, MirSpecEncMode::Impure),
  203                 span,
  204             )?;
  205             let function_data = FunctionData::new(def_id);
  206             let wands_task = WandEncTask {
  207                 data: function_data,
  208             };
  209             let wands = deps
  210                 .require_dep_spanned::<WandEnc>(wands_task, span)
  211                 .map_err(|err| match err {
  212                     EncodeFullError::AlreadyEncoded => err,
  213                     // Without its wands, the method has no sound contract, so
  214                     // it is not emitted. This is reported here, once; each call
  215                     // to it reports its own failure (see `MethodCallEnc`).
  216                     err => EncodeFullError::Reported(WandEnc::report_failure(vcx, def_id, &err)),
  217                 })?;
  218 
  219             // Add direct resources for inputs and outputs to the pre- and
  220             // postconditions, respectively. "Direct" here refers to owned
  221             // Viper resources that must be passed in/out given the signature,
  222             // without going through any dereferences.
  223             let mut args = Vec::with_capacity(arg_count + params.count());
  224             for arg_idx in (0..arg_count).map(mir::Local::from) {
  225                 let name_p = arg_defs[arg_idx].local.name;
  226                 args.push(vir::vir_local_decl! { vcx; [name_p] : Ref });
  227                 if arg_idx != mir::RETURN_PLACE {
  228                     pres.push(arg_defs[arg_idx].impure_pred);
  229                 }
  230             }
  231             posts.push(arg_defs[mir::RETURN_PLACE].impure_pred);
  232 
  233             // ..
  234             pres.extend(wands.indirect_pres(vcx, &arg_defs, deps));
  235             posts.extend(wands.indirect_posts(vcx, &arg_defs, deps));
  236             posts.extend(wands.wand_posts(vcx, &arg_defs, deps));
  237 
  238             // The method of a pure function `f` gets the postcondition
  239             //     ensures snap(result) == f_(old(snap(args)))
  240             // with the addresses of `&T` arguments nulled, exactly as a call
  241             // `f(args)` in a specification is encoded. So an impure call
  242             // learns the same value; the body proves it like any other
  243             // postcondition.
  244             let tie = if crate::encoders::is_function_pure(
  245                 def_id,
  246                 GArgs::new(params, params.rust_params()),
  247             ) {
  248                 let pure_func = deps.require_dep_spanned::<FunctionCallEnc>(
  249                     CallTaskDescription::new(def_id, params.rust_params(), def_id)
  250                         .resolve_trait_calls(false),
  251                     span,
  252                 )?;
  253                 let arg_snaps = (1..arg_count)
  254                     .map(mir::Local::from)
  255                     .map(|arg_idx| vcx.mk_old_expr(arg_defs[arg_idx].impure_snap))
  256                     .collect::<Vec<_>>();
  257                 let name = vcx.tcx().def_path_str(def_id);
  258                 Some(vcx.with_span(span, |vcx| {
  259                     vcx.add_error_taint(pure_func.body_taint());
  260                     vcx.handle_error("postcondition.violated:assertion.false", move |_| {
  261                         Some(vec![PrustiError::verification(
  262                             format!(
  263                                 "the result of pure function `{name}` might not match its \
  264                                  definition with the addresses of its shared-reference \
  265                                  arguments ignored: either the result depends on those \
  266                                  addresses, which a pure function must not, or Prusti's \
  267                                  impure and pure encodings of the function disagree"
  268                             ),
  269                             span.into(),
  270                         )])
  271                     });
  272                     vcx.mk_eq_expr(
  273                         arg_defs[mir::RETURN_PLACE].impure_snap,
  274                         pure_func.call_pure(arg_snaps),
  275                     )
  276                 }))
  277             } else {
  278                 None
  279             };
  280 
  281             // Trusted functions, call stubs, external functions and trait
  282             // functions without a default implementation have no body to
  283             // encode; only their contract is emitted.
  284             let blocks = if let Some(body_with_facts) =
  285                 crate::encoders::impure_body_with_facts(def_id)
  286             {
  287                 let body = &body_with_facts.body;
  288                 let local_defs = deps.require_dep_spanned::<MirLocalDefEnc>(
  289                     MirLocalDefEncTask::Local {
  290                         def_id,
  291                         all_locals: true,
  292                     },
  293                     span,
  294                 )?;
  295 
  296                 let pcg_creator = pcg::PcgCtxtCreator::new(vcx.tcx());
  297                 let pcg_ctxt = pcg_creator.new_nll_ctxt(&body_with_facts);
  298                 let fpcs_analysis = pcg::run_pcg(pcg_ctxt);
  299                 pcg_ctxt.update_debug_visualization_metadata();
  300 
  301                 let block_count = body.basic_blocks.len();
  302 
  303                 let mut encoded_blocks = Vec::with_capacity(
  304                     // extra blocks: Start, End
  305                     2 + block_count,
  306                 );
  307                 let mut start_stmts = TraitImplEnc::assume_context_bounds(vcx, deps, params)?;
  308                 for local in (arg_count..body.local_decls.len()).map(mir::Local::from) {
  309                     // Spec-only locals have no definition.
  310                     let Some(local_def) = local_defs.get(local) else {
  311                         continue;
  312                     };
  313                     let name_p = local_def.local.name;
  314                     start_stmts.push(
  315                         vcx.mk_local_decl_stmt(vir::vir_local_decl! { vcx; [name_p] : Ref }, None),
  316                     )
  317                 }
  318                 // This will be overwritten later.
  319                 encoded_blocks.push(vcx.mk_cfg_block(
  320                     &vir::CfgBlockLabelData::Start,
  321                     &[],
  322                     &[],
  323                     vcx.mk_goto_stmt(&vir::CfgBlockLabelData::BasicBlock(0)),
  324                 ));
  325 
  326                 let spec_blocks = SpecBlocks::new(
  327                     deps.require_dep::<SpecBlocksEnc>(def_id)?,
  328                     body,
  329                     fpcs_analysis.analysis().loop_analysis(),
  330                 );
  331 
  332                 deps.check_cycle()?;
  333                 let mut visitor = ImpureEncVisitor {
  334                     vcx,
  335                     deps,
  336                     def_id,
  337                     local_decls: &body.local_decls,
  338                     fpcs_analysis,
  339                     local_defs,
  340                     spec_blocks,
  341                     body,
  342 
  343                     wands,
  344 
  345                     tmp_ctr: 0,
  346                     label_ctr: 0,
  347                     call_labels: Default::default(),
  348                     wandless_calls: Default::default(),
  349                     from_to_vars: Default::default(),
  350 
  351                     current_block: None,
  352                     current_block_pres: None,
  353                     current_block_succs: None,
  354                     current_block_label: None,
  355                     current_fpcs: None,
  356 
  357                     current_stmts: None,
  358                     current_terminator: None,
  359                     encoded_blocks,
  360                 };
  361                 // if we encountered an error/cycle during encoding, we don't
  362                 // emit a method body; encoding errors additionally surface as
  363                 // early errors rather than silently degrading to a stub
  364                 match visitor.visit_body(body) {
  365                     Ok(()) => {
  366                         start_stmts.extend(
  367                             visitor
  368                                 .from_to_vars
  369                                 .decls()
  370                                 .map(|v| vcx.mk_local_decl_stmt(v, Some(vcx.mk_bool::<false>()))),
  371                         );
  372                         visitor.encoded_blocks[0] = vcx.mk_cfg_block(
  373                             &vir::CfgBlockLabelData::Start,
  374                             &[],
  375                             vcx.alloc_slice(&start_stmts),
  376                             vcx.mk_goto_stmt(&vir::CfgBlockLabelData::BasicBlock(0)),
  377                         );
  378 
  379                         visitor.encoded_blocks.push(vcx.mk_cfg_block(
  380                             vcx.alloc(vir::CfgBlockLabelData::End),
  381                             &[],
  382                             &[],
  383                             vcx.alloc(vir::TerminatorStmtData::Exit),
  384                         ));
  385 
  386                         visitor.deps.check_cycle()?;
  387 
  388                         Some(visitor.encoded_blocks)
  389                     }
  390                     Err(EncodeFullError::AlreadyEncoded) => None,
  391                     Err(err) => {
  392                         super::report_dep_error(vcx, &err, |err| {
  393                             let span = err.span.unwrap_or_else(|| vcx.tcx().def_span(def_id));
  394                             // Worded like `FunctionEnc`'s error, so that when both
  395                             // encodings of a pure function's body fail in the same
  396                             // way, rustc reports the identical error only once.
  397                             let name = vcx.tcx().def_path_str(def_id);
  398                             err.report(format!("function body `{name}`"), span)
  399                         });
  400                         None
  401                     }
  402                 }
  403             } else {
  404                 None
  405             };
  406 
  407             // Add functional specification as the last pre- and postconditions.
  408             pres.extend(spec.pre_exprs());
  409             posts.extend(spec.post_exprs());
  410             // After the functional postconditions: applying `f_` assumes its
  411             // (unchecked) postconditions, so they must be checked first.
  412             // Proving the tie still assumes them on the dummied arguments.
  413             // A postcondition that holds on the real arguments but not on the
  414             // dummied ones (e.g. `x as *const i32 != null`) is then assumed
  415             // false, which makes the tie vacuously provable and lets callers
  416             // derive false.
  417             posts.extend(tie);
  418 
  419             Ok((
  420                 MethodEncOutput {
  421                     method: vcx.mk_method(
  422                         method_ref,
  423                         (&args, generics.ty_decls(), generics.const_decls()),
  424                         &[],
  425                         vcx.alloc_slice(&pres),
  426                         vcx.alloc_slice(&posts),
  427                         blocks.map(|blocks| vcx.alloc_slice(&blocks)),
  428                     ),
  429                 },
  430                 (),
  431             ))
  432         })
  433     }
  434 
  435     fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
  436         for output in Self::all_outputs_local(program) {
  437             program.add_method(output.method);
  438         }
  439     }
