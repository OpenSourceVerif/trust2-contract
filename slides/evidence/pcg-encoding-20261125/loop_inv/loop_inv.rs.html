    1 use prusti_contracts::*;
    2 
    3 fn main() {
    4     let mut i = 1000;
    5 
    6     while i > 2 {
    7         // Note: the invariant is evaluated in the body, after the loop guard.
    8         // So, it is not `i >= 2` because inside the loop `i` is never `2`.
    9         // This design choice works well with complex loop guards, which can have side effects and can
   10         // even contain `return` statements.
   11         body_invariant!(i > 2);
   12         i -= 1;
   13     }
   14 
   15     assert!(i == 2);
   16 }
